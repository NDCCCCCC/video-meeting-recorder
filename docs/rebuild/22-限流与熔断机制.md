# 限流与熔断机制

## 概述

本文档描述了系统的限流和熔断机制，用于保护系统免受过载冲击，确保在高负载情况下的稳定性和可用性。

## 核心概念

### 限流 (Rate Limiting)

限制系统处理请求的速率，防止系统过载。

**应用场景**：
- API接口限流
- 防止恶意请求
- 保护下游服务
- 实现公平使用

### 熔断 (Circuit Breaker)

当检测到下游服务故障时，快速失败，避免级联故障。

**应用场景**：
- 保护调用链路
- 防止雪崩效应
- 快速失败响应
- 自动恢复检测

### 降级 (Degradation)

当系统负载过高时，暂时关闭非核心功能。

**应用场景**：
- 高峰期保护核心功能
- 资源不足时简化响应
- 维护模式

### 1. 限流器接口

**文件位置**: `internal/middleware/ratelimit/limiter.go`

```go
package ratelimit

import (
    "context"
    "time"
)

// Limiter 限流器接口
type Limiter interface {
    // Allow 检查是否允许请求
    Allow() bool

    // AllowN 检查是否允许n个请求
    AllowN(n int) bool

    // Wait 等待直到允许请求
    Wait(ctx context.Context) error

    // WaitN 等待直到允许n个请求
    WaitN(ctx context.Context, n int) error

    // Reset 重置限流器
    Reset()

    // GetStatus 获取当前状态
    GetStatus() Status
}

// Status 限流器状态
type Status struct {
    Allowed     bool      `json:"allowed"`
    Remaining   int       `json:"remaining"`
    ResetTime   time.Time `json:"reset_time"`
    RetryAfter  time.Duration `json:"retry_after,omitempty"`
}

// LimiterType 限流器类型
type LimiterType string

const (
    TokenBucketType   LimiterType = "token_bucket"   // 令牌桶
    LeakyBucketType   LimiterType = "leaky_bucket"   // 漏桶
    SlidingWindowType LimiterType = "sliding_window" // 滑动窗口
    FixedWindowType   LimiterType = "fixed_window"   // 固定窗口
)

// Config 限流器配置
type Config struct {
    Type     LimiterType `json:"type" yaml:"type"`
    Rate     int         `json:"rate" yaml:"rate"`         // 速率（请求/秒）
    Burst    int         `json:"burst" yaml:"burst"`       // 突发容量
    Capacity int         `json:"capacity" yaml:"capacity"` // 容量
}
```

### 2. 令牌桶算法

**文件位置**: `internal/middleware/ratelimit/token_bucket.go`

```go
package ratelimit

import (
    "context"
    "sync"
    "time"

    "golang.org/x/time/rate"
)

// TokenBucketLimiter 令牌桶限流器
type TokenBucketLimiter struct {
    limiter *rate.Limiter
    mu      sync.RWMutex
    config  *Config
}

// NewTokenBucketLimiter 创建令牌桶限流器
func NewTokenBucketLimiter(config *Config) *TokenBucketLimiter {
    // 计算每秒限制
    r := rate.Limit(float64(config.Rate))

    // 创建限流器，burst为突发容量
    limiter := rate.NewLimiter(r, config.Burst)

    return &TokenBucketLimiter{
        limiter: limiter,
        config:  config,
    }
}

// Allow 检查是否允许请求
func (l *TokenBucketLimiter) Allow() bool {
    l.mu.RLock()
    defer l.mu.RUnlock()
    return l.limiter.Allow()
}

// AllowN 检查是否允许n个请求
func (l *TokenBucketLimiter) AllowN(n int) bool {
    l.mu.RLock()
    defer l.mu.RUnlock()
    return l.limiter.AllowN(time.Now(), n)
}

// Wait 等待直到允许请求
func (l *TokenBucketLimiter) Wait(ctx context.Context) error {
    l.mu.RLock()
    defer l.mu.RUnlock()
    return l.limiter.Wait(ctx)
}

// WaitN 等待直到允许n个请求
func (l *TokenBucketLimiter) WaitN(ctx context.Context, n int) error {
    l.mu.RLock()
    defer l.mu.RUnlock()
    return l.limiter.WaitN(ctx, n)
}

// Reset 重置限流器
func (l *TokenBucketLimiter) Reset() {
    l.mu.Lock()
    defer l.mu.Unlock()

    r := rate.Limit(float64(l.config.Rate))
    l.limiter = rate.NewLimiter(r, l.config.Burst)
}

// GetStatus 获取当前状态
func (l *TokenBucketLimiter) GetStatus() Status {
    l.mu.RLock()
    defer l.mu.RUnlock()

    return Status{
        Allowed:   l.limiter.Allow(),
        Remaining: int(l.limiter.Tokens()),
        ResetTime: time.Now().Add(time.Duration(float64(time.Second) / float64(l.config.Rate))),
    }
}
```

### 3. 滑动窗口算法

**文件位置**: `internal/middleware/ratelimit/sliding_window.go`

```go
package ratelimit

import (
    "sync"
    "time"
)

// SlidingWindowLimiter 滑动窗口限流器
type SlidingWindowLimiter struct {
    window    time.Duration // 窗口大小
    limit     int           // 窗口内允许的请求数
    requests  []time.Time   // 请求时间戳
    mu        sync.RWMutex  // 读写锁
    lastClean time.Time     // 上次清理时间
}

// NewSlidingWindowLimiter 创建滑动窗口限流器
func NewSlidingWindowLimiter(window time.Duration, limit int) *SlidingWindowLimiter {
    return &SlidingWindowLimiter{
        window:    window,
        limit:     limit,
        requests:  make([]time.Time, 0),
        lastClean: time.Now(),
    }
}

// Allow 检查是否允许请求
func (l *SlidingWindowLimiter) Allow() bool {
    l.mu.Lock()
    defer l.mu.Unlock()

    now := time.Now()

    // 清理过期请求
    l.cleanExpiredRequests(now)

    // 检查是否超过限制
    if len(l.requests) >= l.limit {
        return false
    }

    // 记录当前请求
    l.requests = append(l.requests, now)
    return true
}

// AllowN 检查是否允许n个请求
func (l *SlidingWindowLimiter) AllowN(n int) bool {
    l.mu.Lock()
    defer l.mu.Unlock()

    now := time.Now()
    l.cleanExpiredRequests(now)

    if len(l.requests)+n > l.limit {
        return false
    }

    for i := 0; i < n; i++ {
        l.requests = append(l.requests, now)
    }
    return true
}

// cleanExpiredRequests 清理过期请求
func (l *SlidingWindowLimiter) cleanExpiredRequests(now time.Time) {
    // 定期清理
    if now.Sub(l.lastClean) < l.window/2 {
        return
    }

    cutoff := now.Add(-l.window)
    valid := 0

    for _, req := range l.requests {
        if req.After(cutoff) {
            l.requests[valid] = req
            valid++
        }
    }

    l.requests = l.requests[:valid]
    l.lastClean = now
}

// Reset 重置限流器
func (l *SlidingWindowLimiter) Reset() {
    l.mu.Lock()
    defer l.mu.Unlock()
    l.requests = make([]time.Time, 0)
}

// GetStatus 获取当前状态
func (l *SlidingWindowLimiter) GetStatus() Status {
    l.mu.RLock()
    defer l.mu.RUnlock()

    now := time.Now()
    cutoff := now.Add(-l.window)

    validCount := 0
    for _, req := range l.requests {
        if req.After(cutoff) {
            validCount++
        }
    }

    return Status{
        Allowed:   validCount < l.limit,
        Remaining: l.limit - validCount,
        ResetTime: l.requests[0].Add(l.window),
    }
}
```

```go
package ratelimit

import (
    "fmt"
    "net/http"
    "strconv"
    "sync"
    "time"

    "github.com/gin-gonic/gin"
    "go.uber.org/zap"
    "golang.org/x/time/rate"
)

// RateLimiter 限流器管理器
type RateLimiter struct {
    limiters map[string]Limiter
    mu       sync.RWMutex
    logger   *zap.Logger
}

// NewRateLimiter 创建限流器管理器
func NewRateLimiter(logger *zap.Logger) *RateLimiter {
    return &RateLimiter{
        limiters: make(map[string]Limiter),
        logger:   logger,
    }
}

// getLimiter 获取或创建限流器
func (rl *RateLimiter) getLimiter(key string, config *Config) Limiter {
    rl.mu.Lock()
    defer rl.mu.Unlock()

    if limiter, exists := rl.limiters[key]; exists {
        return limiter
    }

    var limiter Limiter
    switch config.Type {
    case TokenBucketType:
        limiter = NewTokenBucketLimiter(config)
    case SlidingWindowType:
        limiter = NewSlidingWindowLimiter(time.Second, config.Rate)
    default:
        limiter = NewTokenBucketLimiter(config)
    }

    rl.limiters[key] = limiter
    return limiter
}

// IPMiddleware 基于IP的限流中间件
func (rl *RateLimiter) IPMiddleware(config *Config) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 获取客户端IP
        ip := c.ClientIP()

        // 获取限流器
        limiter := rl.getLimiter("ip:"+ip, config)

        // 检查是否允许
        if !limiter.Allow() {
            status := limiter.GetStatus()

            rl.logger.Warn("IP限流触发",
                zap.String("ip", ip),
                zap.String("path", c.Request.URL.Path),
            )

            c.Header("X-RateLimit-Limit", strconv.Itoa(config.Rate))
            c.Header("X-RateLimit-Remaining", "0")
            c.Header("X-RateLimit-Reset", status.ResetTime.Format(time.RFC3339))
            c.Header("Retry-After", strconv.Itoa(int(status.RetryAfter.Seconds())))

            c.JSON(http.StatusTooManyRequests, gin.H{
                "code":    "RATE_LIMIT_EXCEEDED",
                "message": "请求过于频繁，请稍后再试",
                "details": gin.H{
                    "retry_after": status.RetryAfter.String(),
                },
            })
            c.Abort()
            return
        }

        c.Next()
    }
}

// UserMiddleware 基于用户的限流中间件
func (rl *RateLimiter) UserMiddleware(config *Config) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 获取用户ID（从JWT或Session中）
        userID, exists := c.Get("user_id")
        if !exists {
            c.Next()
            return
        }

        // 获取限流器
        key := fmt.Sprintf("user:%v", userID)
        limiter := rl.getLimiter(key, config)

        // 检查是否允许
        if !limiter.Allow() {
            status := limiter.GetStatus()

            rl.logger.Warn("用户限流触发",
                zap.Any("user_id", userID),
                zap.String("path", c.Request.URL.Path),
            )

            c.JSON(http.StatusTooManyRequests, gin.H{
                "code":    "USER_RATE_LIMIT_EXCEEDED",
                "message": "您的请求过于频繁，请稍后再试",
                "details": gin.H{
                    "retry_after": status.RetryAfter.String(),
                },
            })
            c.Abort()
            return
        }

        c.Next()
    }
}

// APIMiddleware 基于API的限流中间件
func (rl *RateLimiter) APIMiddleware(config *Config) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 获取API路径
        path := c.FullPath()

        // 获取限流器
        limiter := rl.getLimiter("api:"+path, config)

        // 检查是否允许
        if !limiter.Allow() {
            rl.logger.Warn("API限流触发",
                zap.String("path", path),
            )

            c.JSON(http.StatusTooManyRequests, gin.H{
                "code":    "API_RATE_LIMIT_EXCEEDED",
                "message": "该接口请求过于频繁",
            })
            c.Abort()
            return
        }

        c.Next()
    }
}

// CombinedMiddleware 组合限流中间件
// 按顺序检查：IP限流 -> 用户限流 -> API限流
func (rl *RateLimiter) CombinedMiddleware(ipConfig, userConfig, apiConfig *Config) gin.HandlerFunc {
    return func(c *gin.Context) {
        // IP限流
        if ipConfig != nil {
            ip := c.ClientIP()
            limiter := rl.getLimiter("ip:"+ip, ipConfig)
            if !limiter.Allow() {
                c.JSON(http.StatusTooManyRequests, gin.H{
                    "code":    "IP_RATE_LIMIT_EXCEEDED",
                    "message": "IP请求过于频繁",
                })
                c.Abort()
                return
            }
        }

        // 用户限流
        if userConfig != nil {
            if userID, exists := c.Get("user_id"); exists {
                key := fmt.Sprintf("user:%v", userID)
                limiter := rl.getLimiter(key, userConfig)
                if !limiter.Allow() {
                    c.JSON(http.StatusTooManyRequests, gin.H{
                        "code":    "USER_RATE_LIMIT_EXCEEDED",
                        "message": "用户请求过于频繁",
                    })
                    c.Abort()
                    return
                }
            }
        }

        // API限流
        if apiConfig != nil {
            path := c.FullPath()
            limiter := rl.getLimiter("api:"+path, apiConfig)
            if !limiter.Allow() {
                c.JSON(http.StatusTooManyRequests, gin.H{
                    "code":    "API_RATE_LIMIT_EXCEEDED",
                    "message": "接口请求过于频繁",
                })
                c.Abort()
                return
            }
        }

        c.Next()
    }
}
```

### 2. 熔断器实现

**文件位置**: `internal/middleware/circuitbreaker/breaker.go`

```go
package circuitbreaker

import (
    "errors"
    "sync"
    "time"

    "go.uber.org/zap"
)

// Config 熔断器配置
type Config struct {
    // 触发熔断的失败阈值
    Threshold int `json:"threshold" yaml:"threshold"`

    // 统计时间窗口
    Interval time.Duration `json:"interval" yaml:"interval"`

    // 熔断后等待时间
    Timeout time.Duration `json:"timeout" yaml:"timeout"`

    // 半开状态下的尝试次数
    HalfOpenAttempts int `json:"half_open_attempts" yaml:"half_open_attempts"`

    // 半开状态下的成功率阈值（0-1）
    HalfOpenSuccessRate float64 `json:"half_open_success_rate" yaml:"half_open_success_rate"`
}

// CircuitBreaker 熔断器
type CircuitBreaker struct {
    name    string
    config  *Config
    logger  *zap.Logger
    mu      sync.RWMutex

    // 状态相关
    state          State
    stateStartTime time.Time

    // 统计相关
    failureCount   int
    successCount   int
    requestCount   int
    lastResetTime  time.Time

    // 半开状态计数
    halfOpenCount  int
    halfOpenSuccess int
}

// NewCircuitBreaker 创建熔断器
func NewCircuitBreaker(name string, config *Config, logger *zap.Logger) *CircuitBreaker {
    if config.Threshold <= 0 {
        config.Threshold = 5 // 默认5次失败触发熔断
    }
    if config.Interval <= 0 {
        config.Interval = time.Minute // 默认1分钟
    }
    if config.Timeout <= 0 {
        config.Timeout = 30 * time.Second // 默认30秒
    }
    if config.HalfOpenAttempts <= 0 {
        config.HalfOpenAttempts = 3 // 默认尝试3次
    }
    if config.HalfOpenSuccessRate <= 0 {
        config.HalfOpenSuccessRate = 0.5 // 默认50%成功率
    }

    return &CircuitBreaker{
        name:           name,
        config:         config,
        logger:         logger,
        state:          StateClosed,
        stateStartTime: time.Now(),
        lastResetTime:  time.Now(),
    }
}

// Allow 检查是否允许请求
func (cb *CircuitBreaker) Allow() error {
    cb.mu.Lock()
    defer cb.mu.Unlock()

    now := time.Now()

    // 检查是否需要重置计数器
    if now.Sub(cb.lastResetTime) > cb.config.Interval {
        cb.resetCounters()
    }

    // 检查是否需要转换状态
    cb.checkStateTransition(now)

    // 根据状态决定是否允许
    switch cb.state {
    case StateOpen:
        // 熔断开启，拒绝请求
        return &CircuitBreakerError{
            Name:   cb.name,
            State:  cb.state,
            Reason: "熔断器开启，请求被拒绝",
        }
    case StateHalfOpen:
        // 半开状态，允许试探
        cb.halfOpenCount++
    case StateClosed:
        // 正常状态，允许请求
    }

    cb.requestCount++
    return nil
}

// RecordSuccess 记录成功
func (cb *CircuitBreaker) RecordSuccess() {
    cb.mu.Lock()
    defer cb.mu.Unlock()

    cb.successCount++
    if cb.state == StateHalfOpen {
        cb.halfOpenSuccess++
    }
}

// RecordFailure 记录失败
func (cb *CircuitBreaker) RecordFailure() {
    cb.mu.Lock()
    defer cb.mu.Unlock()

    cb.failureCount++

    // 检查是否需要触发熔断
    if cb.state == StateClosed && cb.failureCount >= cb.config.Threshold {
        cb.toOpen()
    } else if cb.state == StateHalfOpen {
        // 半开状态失败，直接回到熔断
        cb.toOpen()
    }
}

// checkStateTransition 检查状态转换
func (cb *CircuitBreaker) checkStateTransition(now time.Time) {
    switch cb.state {
    case StateOpen:
        // 检查是否可以尝试恢复
        if now.Sub(cb.stateStartTime) >= cb.config.Timeout {
            cb.toHalfOpen()
        }
    case StateHalfOpen:
        // 检查半开状态是否完成
        if cb.halfOpenCount >= cb.config.HalfOpenAttempts {
            successRate := float64(cb.halfOpenSuccess) / float64(cb.halfOpenCount)
            if successRate >= cb.config.HalfOpenSuccessRate {
                cb.toClosed()
            } else {
                cb.toOpen()
            }
        }
    }
}

// toOpen 转换到开启状态
func (cb *CircuitBreaker) toOpen() {
    cb.logger.Warn("熔断器开启",
        zap.String("name", cb.name),
        zap.Int("failure_count", cb.failureCount),
    )

    cb.state = StateOpen
    cb.stateStartTime = time.Now()
    cb.halfOpenCount = 0
    cb.halfOpenSuccess = 0
}

// toHalfOpen 转换到半开状态
func (cb *CircuitBreaker) toHalfOpen() {
    cb.logger.Info("熔断器进入半开状态",
        zap.String("name", cb.name),
    )

    cb.state = StateHalfOpen
    cb.stateStartTime = time.Now()
    cb.halfOpenCount = 0
    cb.halfOpenSuccess = 0
}

// toClosed 转换到关闭状态
func (cb *CircuitBreaker) toClosed() {
    cb.logger.Info("熔断器恢复",
        zap.String("name", cb.name),
    )

    cb.state = StateClosed
    cb.stateStartTime = time.Now()
    cb.resetCounters()
}

// resetCounters 重置计数器
func (cb *CircuitBreaker) resetCounters() {
    cb.failureCount = 0
    cb.successCount = 0
    cb.requestCount = 0
    cb.lastResetTime = time.Now()
}

// GetState 获取当前状态
func (cb *CircuitBreaker) GetState() State {
    cb.mu.RLock()
    defer cb.mu.RUnlock()
    return cb.state
}

// GetStats 获取统计信息
func (cb *CircuitBreaker) GetStats() Stats {
    cb.mu.RLock()
    defer cb.mu.RUnlock()

    return Stats{
        Name:            cb.name,
        State:           cb.state.String(),
        RequestCount:    cb.requestCount,
        SuccessCount:    cb.successCount,
        FailureCount:    cb.failureCount,
        StateDuration:   time.Since(cb.stateStartTime).String(),
    }
}

// Stats 统计信息
type Stats struct {
    Name          string `json:"name"`
    State         string `json:"state"`
    RequestCount  int    `json:"request_count"`
    SuccessCount  int    `json:"success_count"`
    FailureCount  int    `json:"failure_count"`
    StateDuration string `json:"state_duration"`
}

// CircuitBreakerError 熔断器错误
type CircuitBreakerError struct {
    Name   string
    State  State
    Reason string
}

func (e *CircuitBreakerError) Error() string {
    return e.Reason
}

// IsCircuitBreakerError 检查是否为熔断器错误
func IsCircuitBreakerError(err error) bool {
    var cbErr *CircuitBreakerError
    return errors.As(err, &cbErr)
}
```

```go
package circuitbreaker

import (
    "net/http"

    "github.com/gin-gonic/gin"
    "go.uber.org/zap"
)

// Manager 熔断器管理器
type Manager struct {
    breakers map[string]*CircuitBreaker
    logger   *zap.Logger
}

// NewManager 创建熔断器管理器
func NewManager(logger *zap.Logger) *Manager {
    return &Manager{
        breakers: make(map[string]*CircuitBreaker),
        logger:   logger,
    }
}

// GetOrCreateBreaker 获取或创建熔断器
func (m *Manager) GetOrCreateBreaker(name string, config *Config) *CircuitBreaker {
    if breaker, exists := m.breakers[name]; exists {
        return breaker
    }

    breaker := NewCircuitBreaker(name, config, m.logger)
    m.breakers[name] = breaker
    return breaker
}

// Middleware 创建熔断中间件
func (m *Manager) Middleware(name string, config *Config) gin.HandlerFunc {
    breaker := m.GetOrCreateBreaker(name, config)

    return func(c *gin.Context) {
        // 检查是否允许请求
        if err := breaker.Allow(); err != nil {
            if IsCircuitBreakerError(err) {
                m.logger.Warn("熔断器拦截请求",
                    zap.String("name", name),
                    zap.String("path", c.Request.URL.Path),
                )

                c.JSON(http.StatusServiceUnavailable, gin.H{
                    "code":    "SERVICE_UNAVAILABLE",
                    "message": "服务暂时不可用，请稍后再试",
                    "details": gin.H{
                        "breaker": name,
                        "state":   breaker.GetState().String(),
                    },
                })
                c.Abort()
                return
            }
        }

        // 记录开始时间
        // 执行请求
        c.Next()

        // 根据响应状态记录结果
        if c.Writer.Status() >= 500 {
            breaker.RecordFailure()
        } else {
            breaker.RecordSuccess()
        }
    }
}

// BreakerFunc 包装函数为熔断保护
func (m *Manager) BreakerFunc(name string, config *Config, fn func() error) error {
    breaker := m.GetOrCreateBreaker(name, config)

    if err := breaker.Allow(); err != nil {
        return err
    }

    if err := fn(); err != nil {
        breaker.RecordFailure()
        return err
    }

    breaker.RecordSuccess()
    return nil
}
```

## 降级策略

```go
package common

import (
    "context"
    "sync"

    "github.com/gin-gonic/gin"
    "go.uber.org/zap"
)

// DegradationLevel 降级级别
type DegradationLevel int

const (
    Normal     DegradationLevel = iota // 正常
    Minor                               // 轻度降级
    Moderate                            // 中度降级
    Severe                              // 重度降级
)

// DegradationStrategy 降级策略
type DegradationStrategy struct {
    Name            string
    Level           DegradationLevel
    DisabledFeatures []string
    CacheOnly       bool
    DefaultResponse interface{}
}

// DegradationManager 降级管理器
type DegradationManager struct {
    mu       sync.RWMutex
    level    DegradationLevel
    strategies map[string]*DegradationStrategy
    logger   *zap.Logger
}

// NewDegradationManager 创建降级管理器
func NewDegradationManager(logger *zap.Logger) *DegradationManager {
    return &DegradationManager{
        level:       Normal,
        strategies: make(map[string]*DegradationStrategy),
        logger:     logger,
    }
}

// SetLevel 设置降级级别
func (dm *DegradationManager) SetLevel(level DegradationLevel) {
    dm.mu.Lock()
    defer dm.mu.Unlock()

    oldLevel := dm.level
    dm.level = level

    dm.logger.Warn("降级级别变更",
        zap.String("from", oldLevel.String()),
        zap.String("to", level.String()),
    )
}

// GetLevel 获取当前降级级别
func (dm *DegradationManager) GetLevel() DegradationLevel {
    dm.mu.RLock()
    defer dm.mu.RUnlock()
    return dm.level
}

// IsFeatureEnabled 检查功能是否启用
func (dm *DegradationManager) IsFeatureEnabled(feature string) bool {
    dm.mu.RLock()
    defer dm.mu.RUnlock()

    for _, strategy := range dm.strategies {
        if strategy.Level <= dm.level {
            for _, disabled := range strategy.DisabledFeatures {
                if disabled == feature {
                    return false
                }
            }
        }
    }

    return true
}

// RegisterStrategy 注册降级策略
func (dm *DegradationManager) RegisterStrategy(strategy *DegradationStrategy) {
    dm.mu.Lock()
    defer dm.mu.Unlock()
    dm.strategies[strategy.Name] = strategy
}

// Middleware 降级中间件
func (dm *DegradationManager) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        level := dm.GetLevel()

        // 根据降级级别返回不同响应
        switch level {
        case Severe:
            c.JSON(503, gin.H{
                "code":    "SERVICE_DEGRADED",
                "message": "服务正在降级运行，部分功能不可用",
            })
            c.Abort()
            return
        case Moderate:
            // 禁用某些非关键接口
            if !dm.IsFeatureEnabled(c.FullPath()) {
                c.JSON(503, gin.H{
                    "code":    "FEATURE_DISABLED",
                    "message": "该功能暂时不可用",
                })
                c.Abort()
                return
            }
        case Minor:
            // 只记录日志
        case Normal:
            // 正常运行
        }

        c.Next()
    }
}

func (d DegradationLevel) String() string {
    switch d {
    case Normal:
        return "normal"
    case Minor:
        return "minor"
    case Moderate:
        return "moderate"
    case Severe:
        return "severe"
    default:
        return "unknown"
    }
}
```
