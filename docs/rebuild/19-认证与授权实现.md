# 认证与授权实现

## 一、认证系统概述

### 1.1 认证架构

```
┌─────────────────────────────────────────────────────────────────┐
│                        认证流程                                  │
└─────────────────────────────────────────────────────────────────┘
                              │
    ┌─────────────────────────┼─────────────────────────┐
    │                         │                         │
    ▼                         ▼                         ▼
┌─────────────┐      ┌──────────────┐      ┌──────────────┐
│   Basic     │      │    JWT       │      │   API Key    │
│   Auth      │      │    Auth       │      │   Auth       │
└─────────────┘      └──────────────┘      └──────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                     认证中间件层                                 │
│  - JWTMiddleware              - APIKeyMiddleware                │
│  - BasicAuthMiddleware         - SessionMiddleware              │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                     授权中间件层                                 │
│  - RBACMiddleware (基于角色)                                  │
│  - PermissionMiddleware (基于权限)                             │
│  - DataScopeMiddleware (数据范围)                              │
└─────────────────────────────────────────────────────────────────┘
```

### 1.2 认证方式对比

| 认证方式 | 适用场景 | Token位置 | 优势 | 劣势 |
|---------|---------|----------|------|------|
| **JWT** | 前端用户登录 | Authorization Header | 无状态，跨服务 | 无法主动失效 |
| **API Key** | 第三方集成 | X-API-Key Header | 简单，可限制作用域 | 需要安全管理 |
| **Session** | 传统Web应用 | Cookie | 服务端控制，易失效 | 需要Cookie支持 |

### 2.2 JWT Service实现

```go
// internal/services/auth/jwt_service.go
package auth

import (
    "crypto/rand"
    encoding/hex"
    "errors"
    "time"

    "github.com/golang-jwt/jwt/v5"
    "github.com/spf13/viper"
    "go.uber.org/zap"
)

type JWTService struct {
    secretKey     []byte
    issuer        string
    expireHours   int
    refreshExpire int
    logger        *zap.Logger
}

type Claims struct {
    UserID      uint   `json:"user_id"`
    Username    string `json:"username"`
    RoleID      uint   `json:"role_id"`
    TokenType   string `json:"token_type"` // access | refresh
    jwt.RegisteredClaims
}

// NewJWTService 创建JWT服务
func NewJWTService(config *viper.Viper, logger *zap.Logger) *JWTService {
    return &JWTService{
        secretKey:     []byte(config.GetString("auth.jwt.secret")),
        issuer:        config.GetString("auth.jwt.issuer"),
        expireHours:   config.GetInt("auth.jwt.expire_hours"),
        refreshExpire: config.GetInt("auth.jwt.refresh_expire_hours"),
        logger:        logger,
    }
}

// GenerateTokenPair 生成Token对
func (s *JWTService) GenerateTokenPair(user *models.User) (accessToken, refreshToken string, err error) {
    // 生成Access Token
    accessToken, err = s.generateToken(user, "access", time.Duration(s.expireHours)*time.Hour)
    if err != nil {
        return "", "", err
    }

    // 生成Refresh Token
    refreshToken, err = s.generateToken(user, "refresh", time.Duration(s.refreshExpire)*time.Hour)
    if err != nil {
        return "", "", err
    }

    return accessToken, refreshToken, nil
}

// generateToken 生成Token
func (s *JWTService) generateToken(user *models.User, tokenType string, duration time.Duration) (string, error) {
    now := time.Now()

    claims := &Claims{
        UserID:    user.ID,
        Username:  user.Username,
        RoleID:    user.RoleID,
        TokenType: tokenType,
        RegisteredClaims: jwt.RegisteredClaims{
            Issuer:    s.issuer,
            Subject:   user.Username,
            ExpiresAt: now.Add(duration).Unix(),
            IssuedAt:  now.Unix(),
            NotBefore: now.Unix(),
        },
    }

    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    tokenString, err := token.SignedString(s.secretKey)
    if err != nil {
        return "", err
    }

    return tokenString, nil
}

// ValidateToken 验证Token
func (s *JWTService) ValidateToken(tokenString string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenString, &Claims{}, func(token *jwt.Token) (interface{}, error) {
        if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
            return nil, errors.New("unexpected signing method")
        }
        return s.secretKey, nil
    })

    if err != nil {
        return nil, err
    }

    if claims, ok := token.Claims.(*Claims); ok && token.Valid {
        // 检查token类型
        if claims.TokenType != "access" {
            return nil, errors.New("invalid token type")
        }
        return claims, nil
    }

    return nil, errors.New("invalid token")
}

// RefreshAccessToken 刷新Access Token
func (s *JWTService) RefreshAccessToken(refreshToken string) (string, error) {
    // 验证refresh token
    token, err := jwt.ParseWithClaims(refreshToken, &Claims{}, func(token *jwt.Token) (interface{}, error) {
        if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
            return nil, errors.New("unexpected signing method")
        }
        return s.secretKey, nil
    })

    if err != nil {
        return "", err
    }

    claims, ok := token.Claims.(*Claims)
    if !ok || !token.Valid {
        return "", errors.New("invalid refresh token")
    }

    if claims.TokenType != "refresh" {
        return "", errors.New("invalid token type")
    }

    // 加载用户信息
    var user models.User
    if err := db.First(&user, claims.UserID).Error; err != nil {
        return "", errors.New("user not found")
    }

    // 生成新的access token
    return s.generateToken(&user, "access", time.Duration(s.expireHours)*time.Hour)
}

// GenerateRandomSecret 生成随机密钥
func (s *JWTService) GenerateRandomSecret() (string, error) {
    bytes := make([]byte, 32)
    if err := rand.Read(bytes); err != nil {
        return "", err
    }
    return hex.EncodeToString(bytes), nil
}
```

### 2.3 认证中间件

```go
// internal/middleware/auth.go
package middleware

import (
    "strings"
    "record_go/internal/services/auth"

    "github.com/gin-gonic/gin"
    "go.uber.org/zap"
)

// JWTAuth JWT认证中间件
func JWTAuth(jwtService *auth.JWTService) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 1. 从Header获取token
        authHeader := c.GetHeader("Authorization")
        if authHeader == "" {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})
            c.Abort()
            return
        }

        // 2. 解析Bearer token
        parts := strings.SplitN(authHeader, " ", 2)
        if len(parts) != 2 || parts[0] != "Bearer" {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})
            c.Abort()
            return
        }

        // 3. 验证token
        claims, err := jwtService.ValidateToken(parts[1])
        if err != nil {
            c.JSON(401, gin.H{"code": 1005, "message": "Token无效或已过期"})
            c.Abort()
            return
        }

        // 4. 将用户信息存入context
        c.Set("user_id", claims.UserID)
        c.Set("username", claims.Username)
        c.Set("role_id", claims.RoleID)

        c.Next()
    }
}

// OptionalAuth 可选认证中间件（允许未认证用户访问）
func OptionalAuth(jwtService *auth.JWTService) gin.HandlerFunc {
    return func(c *gin.Context) {
        authHeader := c.GetHeader("Authorization")
        if authHeader == "" {
            c.Next()
            return
        }

        parts := strings.SplitN(authHeader, " ", 2)
        if len(parts) != 2 || parts[0] != "Bearer" {
            c.Next()
            return
        }

        if claims, err := jwtService.ValidateToken(parts[1]); err == nil {
            c.Set("user_id", claims.UserID)
            c.Set("username", claims.Username)
            c.Set("role_id", claims.RoleID)
        }

        c.Next()
    }
}
```

```go
// internal/services/auth/password_validator.go
package auth

import (
    "regexp"
    "unicode"
)

type PasswordValidator struct {
    minLength     int
    requireUpper  bool
    requireLower  bool
    requireNumber bool
    requireSpecial bool
}

type ValidationResult struct {
    Valid  bool     `json:"valid"`
    Errors []string `json:"errors,omitempty"`
}

// NewPasswordValidator 创建密码验证器
func NewPasswordValidator(config *Config) *PasswordValidator {
    return &PasswordValidator{
        minLength:     config.Int("auth.password.min_length"),
        requireUpper:  config.Bool("auth.password.require_upper"),
        requireLower:  config.Bool("auth.password.require_lower"),
        requireNumber: config.Bool("auth.password.require_number"),
        requireSpecial: config.Bool("auth.password.require_special"),
    }
}

// Validate 验证密码强度
func (v *PasswordValidator) Validate(password string) *ValidationResult {
    result := &ValidationResult{
        Valid:  true,
        Errors: make([]string, 0),
    }

    // 检查长度
    if len(password) < v.minLength {
        result.Valid = false
        result.Errors = append(result.Errors,
            fmt.Sprintf("密码长度不能少于%d位", v.minLength))
    }

    // 检查大写字母
    if v.requireUpper {
        hasUpper := false
        for _, c := range password {
            if unicode.IsUpper(c) {
                hasUpper = true
                break
            }
        }
        if !hasUpper {
            result.Valid = false
            result.Errors = append(result.Errors, "密码必须包含至少一个大写字母")
        }
    }

    // 检查小写字母
    if v.requireLower {
        hasLower := false
        for _, c := range password {
            if unicode.IsLower(c) {
                hasLower = true
                break
            }
        }
        if !hasLower {
            result.Valid = false
            result.Errors = append(result.Errors, "密码必须包含至少一个小写字母")
        }
    }

    // 检查数字
    if v.requireNumber {
        hasNumber := false
        for _, c := range password {
            if unicode.IsDigit(c) {
                hasNumber = true
                break
            }
        }
        if !hasNumber {
            result.Valid = false
            result.Errors = append(result.Errors, "密码必须包含至少一个数字")
        }
    }

    // 检查特殊字符
    if v.requireSpecial {
        hasSpecial := false
        specialChars := regexp.MustCompile(`[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]`)
        if specialChars.MatchString(password) {
            hasSpecial = true
        }
        if !hasSpecial {
            result.Valid = false
            result.Errors = append(result.Errors, "密码必须包含至少一个特殊字符")
        }
    }

    return result
}

// HashPassword 哈希密码
func HashPassword(password string) (string, error) {
    bytes, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
    return string(bytes), err
}

// CheckPassword 检查密码
func CheckPassword(hashedPassword, password string) bool {
    err := bcrypt.CompareHashAndPassword([]byte(hashedPassword), []byte(password))
    return err == nil
}
```

### 3.2 用户认证服务

```go
// internal/services/core/user_service.go
type UserService struct {
    db     *gorm.DB
    logger *zap.Logger
    jwt    *auth.JWTService
}

// Login 用户登录
func (s *UserService) Login(req *LoginRequest) (*LoginResponse, error) {
    // 1. 查找用户
    var user models.User
    err := s.db.Preload("Role").Preload("Role.Permissions").Where("username = ?", req.Username).First(&user).Error
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, errors.New("用户名或密码错误")
        }
        return nil, err
    }

    // 2. 检查密码
    if !user.CheckPassword(req.Password) {
        return nil, errors.New("用户名或密码错误")
    }

    // 3. 检查用户状态
    if !user.IsActive {
        return nil, errors.New("用户已被禁用")
    }

    // 4. 生成JWT token
    accessToken, refreshToken, err := s.jwt.GenerateTokenPair(&user)
    if err != nil {
        return nil, fmt.Errorf("生成token失败: %w", err)
    }

    // 5. 更新最后登录时间
    now := time.Now()
    user.LastLoginAt = &now
    s.db.Save(&user)

    // 6. 创建session记录（可选，用于token撤销）
    session := &models.Session{
        UserID:    user.ID,
        Token:     accessToken,  // 存储access token hash
        ExpiresAt: now.Add(time.Duration(s.jwt.expireHours) * time.Hour),
        IPAddress: c.ClientIP(),
        UserAgent: c.Request.UserAgent(),
    }
    s.db.Create(session)

    // 7. 返回响应
    return &LoginResponse{
        AccessToken:  accessToken,
        RefreshToken: refreshToken,
        ExpiresIn:     s.jwt.expireHours * 3600,
        User:          user.ToDTO(),
    }, nil
}

// RefreshToken 刷新Token
func (s *UserService) RefreshToken(refreshToken string) (*RefreshTokenResponse, error) {
    // 1. 验证refresh token
    accessToken, err := s.jwt.RefreshAccessToken(refreshToken)
    if err != nil {
        return nil, err
    }

    // 2. 重新生成token对
    claims, _ := s.jwt.ValidateToken(refreshToken)
    var user models.User
    if err := s.db.First(&user, claims.UserID).Error; err != nil {
        return nil, err
    }

    newAccessToken, newRefreshToken, err := s.jwt.GenerateTokenPair(&user)
    if err != nil {
        return nil, err
    }

    // 3. 撤销旧token
    s.revokeToken(accessToken)

    return &RefreshTokenResponse{
        AccessToken:  newAccessToken,
        RefreshToken: newRefreshToken,
        ExpiresIn:     s.jwt.expireHours * 3600,
    }, nil
}

// Logout 用户登出
func (s *UserService) Logout(c *gin.Context) error {
    userID := c.GetUint("user_id")

    // 1. 撤销当前token
    token := c.GetHeader("Authorization")
    if token != "" {
        parts := strings.Split(token, " ")
        if len(parts) == 2 && parts[0] == "Bearer" {
            s.revokeToken(parts[1])
        }
    }

    // 2. 删除所有session
    s.db.Where("user_id = ?", userID).Delete(&models.Session{})

    return nil
}
```

```go
package middleware

import (

// RequirePermission 需要特定权限
func RequirePermission(resource, action string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetUint("user_id")
        if userID == 0 {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})

            return

        user, err := loadUserWithPermissions(c, userID)

            c.Abort()
            return
        }

        // 检查权限
        if !user.HasPermission(resource, action) {
            c.JSON(403, gin.H{"code": 1003, "message": "权限不足"})
            c.Abort()
            return
        }

        c.Next()
    }
}

// RequireRole 需要特定角色
func RequireRole(roles ...string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetUint("user_id")
        if userID == 0 {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})

            return

            return

        hasRole := false
        for _, role := range roles {
            if user.Role.Name == role {
                hasRole = true
                break
            }

            return
        }

        c.Next()
```

### 4.3 数据范围权限

```go
// internal/middleware/data_scope.go
package middleware

import (
    "github.com/gin-gonic/gin"
    "go.uber.org/zap"
)

// DataScopeType 数据范围类型
type DataScopeType string

const (
    DataScopeAll      DataScopeType = "all"      // 全部数据
    DataScopeCustom   DataScopeType = "custom"   // 自定义
    DataScopeDept     DataScopeType = "dept"     // 部门数据
    DataScopeDeptAndSub   DataScopeType = "dept_and_sub" // 部门及子部门
    DataScopeSelf     DataScopeType = "self"     // 仅本人数据
)

// DataScopeMiddleware 数据范围中间件
type DataScopeMiddleware struct {
    db     *gorm.DB
    logger *zap.Logger
}

// RequireDataScope 要求特定数据范围权限
func (m *DataScopeMiddleware) RequireDataScope(minScope DataScopeType) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetUint("user_id")
        if userID == 0 {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})
            c.Abort()
            return
        }

        // 加载用户数据范围
        userScope, err := m.getUserDataScope(userID)
        if err != nil {
            c.JSON(500, gin.H{"code": 1005, "message": "加载数据范围失败"})
            c.Abort()
            return
        }

        // 检查范围权限
        if !m.checkScopePermission(userScope, minScope) {
            c.JSON(403, gin.H{"code": 1003, "message": "数据范围权限不足"})
            c.Abort()
            return
        }

        // 将数据范围存入context
        c.Set("data_scope", userScope)

        c.Next()
    }
}

// getUserDataScope 获取用户数据范围
func (m *DataScopeMiddleware) getUserDataScope(userID uint) (*DataScope, error) {
    var user models.User
    if err := m.db.Preload("Role").First(&user, userID).Error; err != nil {
        return nil, err
    }

    // 从角色获取数据范围
    scope := &DataScope{
        Type:     DataScopeSelf,
        UserIDs:  []uint{userID},
        DeptIDs:  []uint{},
    }

    // 管理员全部数据
    if user.Role.Name == models.RoleAdmin {
        scope.Type = DataScopeAll
    }

    // 查询自定义数据范围
    if user.Role.DataScope != "" {
        scope.Type = DataScopeType(user.Role.DataScope)
        // 加载具体的部门或用户列表
        // ...
    }

    return scope, nil
}

// checkScopePermission 检查范围权限
func (m *DataScopeMiddleware) checkScopePermission(userScope *DataScope, requiredScope DataScopeType) bool {
    scopePriority := map[DataScopeType]int{
        DataScopeAll:          5,
        DataScopeCustom:       4,
        DataScopeDept:         3,
        DataScopeDeptAndSub:   2,
        DataScopeSelf:         1,
    }

    return scopePriority[userScope.Type] >= scopePriority[requiredScope]
}

// ApplyDataScope 应用数据范围到查询
func (m *DataScopeMiddleware) ApplyDataScope(query *gorm.DB, c *gin.Context) *gorm.DB {
    scope, exists := c.Get("data_scope")
    if !exists {
        return query
    }

    dataScope := scope.(*DataScope)

    switch dataScope.Type {
    case DataScopeAll:
        // 不过滤，返回所有数据
    case DataScopeSelf:
        query = query.Where("created_by = ?", c.GetUint("user_id"))
    case DataScopeDept:
        query = query.Where("dept_id IN ?", dataScope.DeptIDs)
    case DataScopeDeptAndSub:
        // 需要查询子部门
        subDeptIDs := m.getSubDeptIDs(dataScope.DeptIDs)
        allDeptIDs := append(dataScope.DeptIDs, subDeptIDs...)
        query = query.Where("dept_id IN ?", allDeptIDs)
    case DataScopeCustom:
        query = query.Where("created_by IN ?", dataScope.UserIDs)
    }

    return query
}
```

```go
type APIKeyService struct {

// GenerateAPIKey 生成API Key
func (s *APIKeyService) GenerateAPIKey(userID uint, name string, scopes []string) (*models.APIKey, error) {
    // 1. 生成随机key
    key, err := generateSecureKey()
    if err != nil {
        return nil, err
    }

    // 2. 创建API Key记录
    apiKey := &models.APIKey{
        Name:      name,
        Key:       key,
        UserID:    userID,
        ExpiresAt: time.Now().Add(365 * 24 * time.Hour), // 默认1年
        IsActive:  true,
        Scopes:    strings.Join(scopes, ","),
    }

    if err := s.db.Create(apiKey).Error; err != nil {
        return nil, err
    }

    return apiKey, nil
}

// generateSecureKey 生成安全的Key
func generateSecureKey() (string, error) {
    bytes := make([]byte, 32)
    if _, err := rand.Read(bytes); err != nil {
        return "", err
    }
    return "sk_" + hex.EncodeToString(bytes), nil
}
```

### 5.2 API Key中间件

```go
// internal/middleware/apikey.go
func APIKeyAuth(db *gorm.DB) gin.HandlerFunc {
    return func(c *gin.Context) {
        apiKey := c.GetHeader("X-API-Key")
        if apiKey == "" {
            c.JSON(401, gin.H{"code": 1002, "message": "未授权"})
            c.Abort()
            return
        }

        // 查询API Key
        var key models.APIKey
        err := db.Preload("User").Preload("User.Role").Where("key = ?", apiKey).First(&key).Error
        if err != nil {
            c.JSON(401, gin.H{"code": 1002, "message": "无效的API Key"})
            c.Abort()
            return
        }

        // 检查状态
        if !key.IsActive || key.IsExpired() {
            c.JSON(401, gin.H{"code": 1002, "message": "API Key已失效"})
            c.Abort()
            return
        }

        // 更新最后使用时间
        key.LastUsedAt = &time.Now{}
        db.Save(&key)

        // 将用户信息存入context
        c.Set("user_id", key.UserID)
        c.Set("api_key_id", key.ID)
        c.Set("auth_type", "apikey")

        c.Next()
    }
}
```
