# 前端架构设计

## 一、技术栈选型

### 1.1 核心技术

| 技术 | 版本 | 说明 |
|------|------|------|
| React | 18.x | UI框架 |
| TypeScript | 5.x | 类型系统 |
| Vite | 5.x | 构建工具 |
| React Router | 6.x | 路由管理 |
| Zustand | 4.x | 状态管理 |
| TanStack Query | 5.x | 服务端状态管理 |
| Axios | 1.x | HTTP客户端 |
| Ant Design | 5.x | UI组件库 |

### 1.2 技术选型理由

**React + TypeScript**
- TypeScript提供类型安全，减少运行时错误
- React生态成熟，组件化开发高效
- 虚拟DOM性能优秀

**Vite**
- 开发启动速度快（基于ESM）
- HMR热更新迅速
- 内置TS支持，无需额外配置

**Zustand**
- 轻量级状态管理（~1KB）
- API简洁，学习成本低
- 无需Provider包裹

**TanStack Query (React Query)**
- 自动处理缓存、重试、轮询
- 优雅的loading/error状态管理
- 减少样板代码

**Ant Design**
- 企业级UI组件库
- 丰富的组件和主题定制
- 中文文档完善

## 二、架构设计原则

### 2.1 分层架构

```
┌─────────────────────────────────────────────────────────┐
│                      UI Layer                           │
│                 (Pages + Components)                    │
└─────────────────────────────────────────────────────────┘
                           │
                           ▼
┌─────────────────────────────────────────────────────────┐
│                    Business Layer                       │
│              (Custom Hooks + State Management)           │
└─────────────────────────────────────────────────────────┘
                           │
                           ▼
┌─────────────────────────────────────────────────────────┐
│                     Data Layer                           │
│              (API Client + TanStack Query)               │
└─────────────────────────────────────────────────────────┘
```

### 2.2 数据流向

```
用户交互 → 触发Action → 调用API → 更新状态 → 重新渲染
         ↑                              │
         └──────── 响应反馈 ←──────────┘
```

### 2.3 关键设计模式

| 模式 | 应用场景 |
|------|----------|
| Container/Presentational | 组件分离 |
| Custom Hooks | 逻辑复用 |
| Higher-Order Components | 横切关注点 |
| Render Props | 组件组合 |
| Compound Components | 复杂组件API |

### 4.3 TanStack Query配置

```typescript
// api/queryClient.ts
import { QueryClient } from '@tanstack/react-query';

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      // 缓存时间
      staleTime: 5 * 60 * 1000,      // 5分钟
      gcTime: 10 * 60 * 1000,       // 10分钟

      // 重试配置
      retry: (failureCount, error) => {
        // 401错误不重试
        if (error.status === 401) return false;
        // 最多重试3次
        return failureCount < 3;
      },

      // 轮询配置
      refetchOnWindowFocus: false,
      refetchOnReconnect: true,
      refetchOnMount: true,
    },
    mutations: {
      // 变更重试
      retry: 1,
    },
  },
});

// 查询键工厂
export const queryKeys = {
  auth: ['auth'] as const,
  users: {
    all: ['users'] as const,
    lists: () => [...queryKeys.users.all, 'list'] as const,
    list: (filters: string) => [...queryKeys.users.lists(), { filters }] as const,
    details: () => [...queryKeys.users.all, 'detail'] as const,
    detail: (id: number) => [...queryKeys.users.details(), id] as const,
  },
  tasks: {
    all: ['tasks'] as const,
    lists: () => [...queryKeys.tasks.all, 'list'] as const,
    list: (filters: TaskFilters) => [...queryKeys.tasks.lists(), filters] as const,
    details: () => [...queryKeys.tasks.all, 'detail'] as const,
    detail: (id: number) => [...queryKeys.tasks.details(), id] as const,
  },
};

// components/common/Button/Button.tsx
import { ButtonHTMLAttributes, forwardRef } from 'react';
import { cn } from '@/utils/classNames';

// 组件Props类型定义
export interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost' | 'link';
  size?: 'small' | 'medium' | 'large';
  loading?: boolean;
  block?: boolean;
  icon?: React.ReactNode;
  children: React.ReactNode;
}

// 使用forwardRef支持ref传递
export const Button = forwardRef<HTMLButtonElement, ButtonProps>(
  (
    {
      variant = 'primary',
      size = 'medium',
      loading = false,
      block = false,
      icon,
      disabled,
      className,
      children,
      ...props
    },
    ref
  ) => {
    return (
      <button
        ref={ref}
        disabled={disabled || loading}
        className={cn(
          'btn',
          `btn-${variant}`,
          `btn-${size}`,
          block && 'btn-block',
          loading && 'btn-loading',
          className
        )}
        {...props}
      >
        {loading && <span className="btn-spinner" />}
        {icon && <span className="btn-icon">{icon}</span>}
        {children}
      </button>
    );
  }
);

Button.displayName = 'Button';
```

// components/business/TaskCard/TaskCard.tsx
import { Card, Tag, Button, Space } from 'antd';
import { useTask } from './useTask';
import type { Task } from '@/types/models';

interface TaskCardProps {
  task: Task;
  onEdit?: (task: Task) => void;
  onDelete?: (id: number) => void;
}

export function TaskCard({ task, onEdit, onDelete }: TaskCardProps) {
  const { status, statusColor, canEdit, canDelete, handleStart, handleStop } = useTask(task);

  return (
    <Card
      title={task.name}
      extra={<Tag color={statusColor}>{status}</Tag>}
      actions={[
        canEdit && <Button key="edit" size="small" onClick={() => onEdit?.(task)}>
          编辑
        </Button>,
        canDelete && <Button key="delete" size="small" danger onClick={() => onDelete?.(task.id)}>
          删除
        </Button>,
      ]}
    >
      <Space direction="vertical" style={{ width: '100%' }}>
        <div>{task.description}</div>
        <div>
          <span>开始时间: {task.startTime}</span>
        </div>
        <div>
          <span>结束时间: {task.endTime}</span>
        </div>
        <Space>
          {task.status === 'pending' && (
            <Button type="primary" size="small" onClick={handleStart}>
              启动
            </Button>
          )}
          {task.status === 'running' && (
            <Button danger size="small" onClick={handleStop}>
              停止
            </Button>
          )}
        </Space>
      </Space>
    </Card>
  );
}

### 6.2 路由守卫

```typescript
// router/guards.ts
import { redirect } from 'react-router-dom';
import { authStore } from '@/stores/authStore';

// 认证守卫
export function authLoader() {
  if (!authStore.isAuthenticated) {
    return redirect('/auth/login');
  }
  return null;
}

// 权限守卫
export function permissionLoader(requiredPermission: string) {
  return () => {
    if (!authStore.hasPermission(requiredPermission)) {
    return redirect('/403');
    }
    return null;
  };
}
```

### 6.3 路由配置实现

```typescript
// router/index.tsx
import { createBrowserRouter, RouterProvider } from 'react-router-dom';
import { routes } from './routes';
import { guards } from './guards';

// 创建路由
const router = createBrowserRouter(
  routes.map(route => {
    // 添加loader守卫
    if (route.loader) {
      return route;
    }

    // 处理需要权限的路由
    if (route.permission) {
      return {
        ...route,
        loader: guards.permissionLoader(route.permission),
      };
    }

    return route;
  })
);

export function AppRouter() {
  return <RouterProvider router={router} />;
}
```

// 使用虚拟滚动处理大列表
import { useVirtualizer } from '@tanstack/react-virtual';

function TaskList({ tasks }: { tasks: Task[] }) {
  const parentRef = useRef<HTMLDivElement>(null);

  const virtualizer = useVirtualizer({
    count: tasks.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 150, // 预估每项高度
    overscan: 5, // 额外渲染项数
  });

  return (
    <div ref={parentRef} style={{ height: '600px', overflow: 'auto' }}>
      <div style={{ height: `${virtualizer.getTotalSize()}px` }}>
        {virtualizer.getVirtualItems().forEach((virtualItem) => (
          <div
            key={virtualItem.key}
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              width: '100%',
              height: `${virtualItem.size}px`,
              transform: `translateY(${virtualItem.start}px)`,
            }}
          >
            <TaskCard task={tasks[virtualItem.index]} />
          </div>
