# 数据库迁移策略

## 概述

本文档描述了项目的数据库迁移管理策略，包括迁移工具选择、迁移脚本编写、版本控制和回滚机制。

## 核心原则

1. **增量式迁移** - 每次迁移只做小改动
2. **可回滚** - 所有迁移必须提供回滚脚本
3. **幂等性** - 迁移可以安全重复执行
4. **向前兼容** - 新版本支持旧数据格式
5. **零停机** - 支持在线迁移，无需停机

## 迁移工具选择

### 工具对比

| 工具 | 优点 | 缺点 | 推荐度 |
|------|------|------|--------|
| GORM AutoMigrate | 简单，与 ORM 集成 | 缺少版本控制，无法回滚 | ⭐⭐ |
| golang-migrate | 功能完整，支持回滚 | 配置复杂 | ⭐⭐⭐⭐⭐ |
| goose | 轻量，SQL + Go 支持 | 社区较小 | ⭐⭐⭐⭐ |
| sql-migrate | 简单易用 | 功能较少 | ⭐⭐⭐ |

### 选择：golang-migrate

**原因**：
- 支持 SQL 和 Go 两种迁移方式
- 完善的版本控制
- 支持回滚
- 多数据库支持
- 活跃的社区

## 迁移命名规范

### 文件命名

```
{VERSION}_{NAME}_{DIRECTION}.sql
```

- **VERSION**: 6位数字，从 000001 开始
- **NAME**: 描述性名称，使用下划线分隔
- **DIRECTION**: up (应用) 或 down (回滚)

### 示例

```
000001_create_users_table.up.sql
000001_create_users_table.down.sql
000002_add_email_index.up.sql
000002_add_email_index.down.sql
000003_add_role_column.up.sql
000003_add_role_column.down.sql
```

## 迁移执行器

**文件**: `internal/database/migrate.go`

```go
package database

import (
    "database/sql"
    "embed"
    "fmt"
    "sort"
    "time"

    "github.com/golang-migrate/migrate/v4"
    "github.com/golang-migrate/migrate/v4/database/sqlite3"
    "github.com/golang-migrate/migrate/v4/source/iofs"
    "go.uber.org/zap"
)

//go:embed migrations
var migrationsFS embed.FS

// Migrator 迁移器
type Migrator struct {
    db     *sql.DB
    logger *zap.Logger
}

// NewMigrator 创建迁移器
func NewMigrator(db *sql.DB, logger *zap.Logger) *Migrator {
    return &Migrator{
        db:     db,
        logger: logger,
    }
}

// Up 执行所有待执行的迁移
func (m *Migrator) Up() error {
    m.logger.Info("开始数据库迁移")

    // 创建迁移实例
    mInstance, err := m.createMigrate()
    if err != nil {
        return fmt.Errorf("创建迁移实例失败: %w", err)
    }
    defer mInstance.Close()

    // 获取当前版本
    currentVersion, dirty, err := mInstance.Version()
    if err != nil && err != migrate.ErrNilVersion {
        return fmt.Errorf("获取当前版本失败: %w", err)
    }

    if dirty {
        m.logger.Warn("数据库处于脏状态，需要修复",
            zap.Uint("current_version", currentVersion),
        )
        return fmt.Errorf("数据库迁移状态异常，请手动修复")
    }

    m.logger.Info("当前数据库版本",
        zap.Uint("version", currentVersion),
    )

    // 执行迁移
    if err := mInstance.Up(); err != nil && err != migrate.ErrNoChange {
        return fmt.Errorf("执行迁移失败: %w", err)
    }

    // 获取新版本
    newVersion, _, _ := mInstance.Version()
    m.logger.Info("数据库迁移完成",
        zap.Uint("old_version", currentVersion),
        zap.Uint("new_version", newVersion),
    )

    return nil
}

// Down 回滚最近一次迁移
func (m *Migrator) Down() error {
    m.logger.Info("回滚最后一次迁移")

    mInstance, err := m.createMigrate()
    if err != nil {
        return err
    }
    defer mInstance.Close()

    if err := mInstance.Steps(-1); err != nil && err != migrate.ErrNoChange {
        return fmt.Errorf("回滚迁移失败: %w", err)
    }

    m.logger.Info("迁移回滚完成")
    return nil
}

// DownTo 回滚到指定版本
func (m *Migrator) DownTo(version uint) error {
    m.logger.Info("回滚迁移",
        zap.Uint("target_version", version),
    )

    mInstance, err := m.createMigrate()
    if err != nil {
        return err
    }
    defer mInstance.Close()

    if err := mInstance.Migrate(version); err != nil {
        return fmt.Errorf("回滚到指定版本失败: %w", err)
    }

    m.logger.Info("迁移回滚完成",
        zap.Uint("target_version", version),
    )
    return nil
}

// Version 获取当前迁移版本
func (m *Migrator) Version() (uint, bool, error) {
    mInstance, err := m.createMigrate()
    if err != nil {
        return 0, false, err
    }
    defer mInstance.Close()

    return mInstance.Version()
}

// createMigrate 创建迁移实例
func (m *Migrator) createMigrate() (*migrate.Migrate, error) {
    // 从 embed.FS 创建源
    d, err := ifs.New(migrationsFS, "migrations")
    if err != nil {
        return nil, fmt.Errorf("创建迁移源失败: %w", err)
    }

    // 创建驱动
    driver, err := sqlite3.WithInstance(m.db, &sqlite3.Config{
        NoForeignKeys: true, // 迁移期间禁用外键
    })
    if err != nil {
        return nil, fmt.Errorf("创建数据库驱动失败: %w", err)
    }

    // 创建迁移实例
    mInstance, err := migrate.NewWithInstance(
        "iofs",
        d,
        "sqlite3",
        driver,
    )
    if err != nil {
        return nil, fmt.Errorf("创建迁移实例失败: %w", err)
    }

    return mInstance, nil
}

// Force 强制设置版本（慎用）
func (m *Migrator) Force(version uint) error {
    m.logger.Warn("强制设置数据库版本",
        zap.Uint("version", version),
    )

    mInstance, err := m.createMigrate()
    if err != nil {
        return err
    }
    defer mInstance.Close()

    return mInstance.Force(version)
}

// ListMigrationFiles 列出所有迁移文件
func (m *Migrator) ListMigrationFiles() ([]string, error) {
    entries, err := migrationsFS.ReadDir("migrations/schema")
    if err != nil {
        return nil, err
    }

    files := make([]string, 0, len(entries))
    for _, entry := range entries {
        if !entry.IsDir() && entry.Name()[len(entry.Name())-6:] == ".up.sql" {
            files = append(files, entry.Name())
        }
    }

    sort.Strings(files)
    return files, nil
}
```

## 种子数据执行器

**文件**: `internal/database/seeder.go`

```go
package database

import (
    "context"
    "embed"
    "encoding/json"
    "fmt"
    "sort"
    "strings"

    "go.uber.org/zap"
    "gorm.io/gorm"
)

//go:embed migrations/seed
var seedFS embed.FS

// Seeder 种子数据执行器
type Seeder struct {
    db     *gorm.DB
    logger *zap.Logger
}

// NewSeeder 创建种子执行器
func NewSeeder(db *gorm.DB, logger *zap.Logger) *Seeder {
    return &Seeder{
        db:     db,
        logger: logger,
    }
}

// Seed 执行所有种子数据
func (s *Seeder) Seed(ctx context.Context) error {
    s.logger.Info("开始执行种子数据")

    // 读取种子目录
    entries, err := seedFS.ReadDir("migrations/seed")
    if err != nil {
        return fmt.Errorf("读取种子目录失败: %w", err)
    }

    // 筛选并排序 up 文件
    files := make([]string, 0)
    for _, entry := range entries {
        name := entry.Name()
        if strings.HasSuffix(name, ".up.sql") {
            files = append(files, name)
        }
    }
    sort.Strings(files)

    // 执行每个种子文件
    for _, file := range files {
        if err := s.executeSeed(ctx, file); err != nil {
            return fmt.Errorf("执行种子文件 %s 失败: %w", file, err)
        }
    }

    s.logger.Info("种子数据执行完成")
    return nil
}

// executeSeed 执行单个种子文件
func (s *Seeder) executeSeed(ctx context.Context, file string) error {
    // 读取文件内容
    path := fmt.Sprintf("migrations/seed/%s", file)
    content, err := seedFS.ReadFile(path)
    if err != nil {
        return err
    }

    // 执行 SQL
    if err := s.db.Exec(string(content)).Error; err != nil {
        return fmt.Errorf("执行 SQL 失败: %w", err)
    }

    s.logger.Debug("种子文件执行成功",
        zap.String("file", file),
    )

    return nil
}

// SeedFromFile 从 JSON 文件导入种子数据
func (s *Seeder) SeedFromFile(ctx context.Context, model interface{}, filePath string) error {
    s.logger.Info("从文件导入种子数据",
        zap.String("file", filePath),
    )

    // 读取 JSON 文件
    data, err := seedFS.ReadFile(filePath)
    if err != nil {
        return fmt.Errorf("读取文件失败: %w", err)
    }

    // 解析 JSON
    records := make([]map[string]interface{}, 0)
    if err := json.Unmarshal(data, &records); err != nil {
        return fmt.Errorf("解析 JSON 失败: %w", err)
    }

    // 批量插入
    if len(records) > 0 {
        if err := s.db.Table(fmt.Sprintf("%t", model)).Create(records).Error; err != nil {
            return fmt.Errorf("插入数据失败: %w", err)
        }
    }

    s.logger.Info("种子数据导入完成",
        zap.Int("count", len(records)),
    )

    return nil
}
```

## 辅助脚本

**文件**: `scripts/migrate.sh`

```bash
#!/bin/bash

# 数据库迁移辅助脚本

set -e

# 配置
DB_PATH="./data/app.db"
MIGRATIONS_DIR="./migrations"

# 颜色输出
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 检查数据库文件
check_db() {
    if [ ! -f "$DB_PATH" ]; then
        log_error "数据库文件不存在: $DB_PATH"
        exit 1
    fi
}

# 创建迁移
create_migration() {
    local name=$1
    local type=${2:-schema}

    if [ -z "$name" ]; then
        log_error "请提供迁移名称"
        echo "用法: ./migrate.sh create <名称> [类型]"
        echo "类型: schema (默认), seed, data"
        exit 1
    fi

    # 查找下一个版本号
    local version=$(find "$MIGRATIONS_DIR/$type" -name "*.up.sql" 2>/dev/null | wc -l)
    version=$(printf "%06d" $((version + 1)))

    local up_file="$MIGRATIONS_DIR/$type/${version}_${name}.up.sql"
    local down_file="$MIGRATIONS_DIR/$type/${version}_${name}.down.sql"

    # 创建文件
    mkdir -p "$MIGRATIONS_DIR/$type"

    cat > "$up_file" <<EOF
-- +migrate Up
-- $name

EOF

    cat > "$down_file" <<EOF
-- +migrate Down
-- 回滚: $name

EOF

    log_info "迁移文件已创建:"
    echo "  - $up_file"
    echo "  - $down_file"
}

# 执行迁移
migrate_up() {
    log_info "开始执行数据库迁移..."
    go run cmd/server/main.go --migrate-up
}

# 回滚迁移
migrate_down() {
    local steps=${1:-1}
    log_info "回滚最近 $steps 次迁移..."
    go run cmd/server/main.go --migrate-down --steps=$steps
}

# 回滚到指定版本
migrate_down_to() {
    local version=$1
    if [ -z "$version" ]; then
        log_error "请提供目标版本"
        exit 1
    fi
    log_info "回滚到版本 $version..."
    go run cmd/server/main.go --migrate-down-to --version=$version
}

# 查看迁移状态
migrate_status() {
    log_info "数据库迁移状态:"
    go run cmd/server/main.go --migrate-status
}

# 查看迁移版本
migrate_version() {
    log_info "当前迁移版本:"
    go run cmd/server/main.go --migrate-version
}

# 强制设置版本
migrate_force() {
    local version=$1
    if [ -z "$version" ]; then
        log_error "请提供目标版本"
        exit 1
    fi
    log_warn "强制设置版本到 $version"
    read -p "确定要继续吗? (y/N) " -n 1 -r
    echo
    if [[ $REPLY =~ ^[Yy]$ ]]; then
        go run cmd/server/main.go --migrate-force --version=$version
    else
        log_info "已取消"
    fi
}

# 验证迁移
validate_migrations() {
    log_info "验证迁移文件..."

    local errors=0

    # 检查 up/down 文件成对
    for up_file in "$MIGRATIONS_DIR"/schema/*.up.sql; do
        if [ -f "$up_file" ]; then
            local down_file="${up_file%.up.sql}.down.sql"
            if [ ! -f "$down_file" ]; then
                log_error "缺少回滚文件: $down_file"
                errors=$((errors + 1))
            fi
        fi
    done

    if [ $errors -eq 0 ]; then
        log_info "所有迁移文件验证通过"
    else
        log_error "发现 $errors 个错误"
        exit 1
    fi
}

# 显示帮助
show_help() {
    cat <<EOF
数据库迁移辅助脚本

用法: ./migrate.sh <命令> [参数]

命令:
  create <名称> [类型]  创建新的迁移文件
                        类型: schema (默认), seed, data

  up                   执行所有待执行的迁移
  down [步骤数]        回滚最近的一次迁移 (默认 1 步)
  down-to <版本>       回滚到指定版本
  status               查看迁移状态
  version              查看当前迁移版本
  force <版本>         强制设置版本 (慎用)
  validate             验证迁移文件
  help                 显示此帮助信息

示例:
  ./migrate.sh create add_user_preferences
  ./migrate.sh create seed_admin_user seed
  ./migrate.sh up
  ./migrate.sh down
  ./migrate.sh down 3
  ./migrate.sh down-to 000005
  ./migrate.sh status
  ./migrate.sh validate

EOF
}

# 主函数
main() {
    case "${1:-help}" in
        create)
            create_migration "$2" "$3"
            ;;
        up)
            check_db
            migrate_up
            ;;
        down)
            check_db
            migrate_down "$2"
            ;;
        down-to)
            check_db
            migrate_down_to "$2"
            ;;
        status)
            check_db
            migrate_status
            ;;
        version)
            check_db
            migrate_version
            ;;
        force)
            check_db
            migrate_force "$2"
            ;;
        validate)
            validate_migrations
            ;;
        help|--help|-h)
            show_help
            ;;
        *)
            log_error "未知命令: $1"

}

main "$@"
```

## 最佳实践

### 1. 迁移设计原则

```sql
-- ✅ 好的迁移：小步增量
-- 000001_add_users_table.up.sql
CREATE TABLE users (...);

-- 000002_add_user_email.up.sql
ALTER TABLE users ADD COLUMN email VARCHAR(100);

-- 000003_add_user_phone.up.sql
ALTER TABLE users ADD COLUMN phone VARCHAR(20);

-- ❌ 不好的迁移：一次做太多
-- 000001_add_users_and_all_fields.up.sql
CREATE TABLE users (
    -- 50+ 个字段...
);
```

### 2. SQLite 特殊处理

```sql
-- SQLite 不支持直接 ALTER COLUMN
-- ❌ 不支持
ALTER TABLE users ALTER COLUMN username SET DATA TYPE VARCHAR(100);

-- ✅ 正确做法：重建表
CREATE TABLE users_new (...);
INSERT INTO users_new SELECT * FROM users;
DROP TABLE users;
ALTER TABLE users_new RENAME TO users;
```

### 3. 数据迁移安全

```sql
-- ✅ 数据迁移前备份
CREATE TABLE users_backup AS SELECT * FROM users;

-- 执行迁移
UPDATE users SET new_column = old_column;

-- 验证后删除备份
-- DROP TABLE users_backup;
```

### 4. 索引创建时机

```sql
-- ✅ 小表：直接创建
CREATE INDEX idx_users_email ON users(email);

-- ✅ 大表：分批创建（应用层处理）
-- 先创建无索引表
-- 导入数据
-- 再创建索引
```

### 5. 外键处理

```sql
-- SQLite 外键默认禁用
-- 迁移期间禁用外键检查
PRAGMA foreign_keys = OFF;

-- 执行迁移...
-- ALTER TABLE ...

-- 迁移完成后启用
PRAGMA foreign_keys = ON;
```
