# 前端状态管理

## 一、状态管理概述

### 1.1 状态分类

```
┌─────────────────────────────────────────────────────────────┐
│                      前端状态分类                             │
└─────────────────────────────────────────────────────────────┘
    │
    ├── Server State (服务端状态) - TanStack Query
    │   ├── 数据来自API
    │   ├── 需要缓存和同步
    │   └── 示例：用户列表、任务列表
    │
    ├── Client State (客户端状态) - Zustand
    │   ├── 仅存在于客户端
    │   ├── 由用户交互产生
    │   └── 示例：UI状态、表单状态、主题
    │
    └── URL State (URL状态) - React Router
        ├── 存储在URL中
        ├── 可分享和刷新
        └── 示例：搜索参数、分页、过滤
```

### 1.2 状态管理工具对比

| 工具 | 适用场景 | 优点 |
|------|----------|------|
| Zustand | 客户端状态 | 轻量、简洁、无Provider |
| TanStack Query | 服务端状态 | 自动缓存、重试、去重 |
| React Router | URL状态 | 与URL同步、可分享 |

## 二、Zustand状态管理

### 2.1 AuthStore（认证状态）

```typescript
// src/stores/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { authApi } from '@/api/modules/auth';
import type { User, LoginRequest } from '@/types/models';

interface AuthState {
  // 状态
  token: string | null;
  refreshToken: string | null;
  isAuthenticated: boolean;
  user: User | null;
  permissions: string[];

  // Actions
  login: (credentials: LoginRequest) => Promise<void>;
  logout: () => void;
  refreshAccessToken: () => Promise<void>;
  setUser: (user: User) => void;
  setPermissions: (permissions: string[]) => void;
  hasPermission: (permission: string) => boolean;
  hasAnyPermission: (permissions: string[]) => boolean;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      // 初始状态
      token: null,
      refreshToken: null,
      isAuthenticated: false,
      user: null,
      permissions: [],

      // 登录
      login: async (credentials) => {
        const response = await authApi.login(credentials);
        const { token, refresh_token, user, permissions } = response.data;

        set({
          token,
          refreshToken: refresh_token,
          isAuthenticated: true,
          user,
          permissions,
        });
      },

      // 登出
      logout: () => {
        set({
          token: null,
          refreshToken: null,
          isAuthenticated: false,
          user: null,
          permissions: [],
        });
      },

      // 刷新Token
      refreshAccessToken: async () => {
        const { refreshToken } = get();
        if (!refreshToken) {
          throw new Error('No refresh token available');
        }

        const response = await authApi.refreshToken(refreshToken);
        const { token, refresh_token } = response.data;

        set({
          token,
          refreshToken: refresh_token,
        });
      },

      // 设置用户信息
      setUser: (user) => {
        set({ user });
      },

      // 设置权限
      setPermissions: (permissions) => {
        set({ permissions });
      },

      // 检查单个权限
      hasPermission: (permission) => {
        const { permissions } = get();
        return permissions.includes(permission) || permissions.includes('*');
      },

      // 检查是否有任意一个权限
      hasAnyPermission: (permissions) => {
        const { permissions: userPermissions } = get();
        return permissions.some(p => userPermissions.includes(p)) ||
               userPermissions.includes('*');
      },
    }),
    {
      name: 'auth-storage', // localStorage key
      partialize: (state) => ({
        token: state.token,
        refreshToken: state.refreshToken,
        user: state.user,
        permissions: state.permissions,
      }),
    }
  )
);
```

### 2.2 UserStore（用户状态）

```typescript
// src/stores/userStore.ts
import { create } from 'zustand';
import { userApi } from '@/api/modules/user';
import type { User, CreateUserData, UpdateUserData } from '@/types/models';

interface UserState {
  users: User[];
  currentUser: User | null;
  loading: boolean;
  error: string | null;

  // Actions
  fetchUsers: (params?: UserQueryParams) => Promise<void>;
  createUser: (data: CreateUserData) => Promise<void>;
  updateUser: (id: number, data: UpdateUserData) => Promise<void>;
  deleteUser: (id: number) => Promise<void>;
  setCurrentUser: (user: User | null) => void;
  clearError: () => void;
}

interface UserQueryParams {
  page?: number;
  pageSize?: number;
  keyword?: string;
  status?: string;
}

export const useUserStore = create<UserState>((set, get) => ({
  users: [],
  currentUser: null,
  loading: false,
  error: null,

  // 获取用户列表
  fetchUsers: async (params) => {
    set({ loading: true, error: null });
    try {
      const response = await userApi.getUsers(params);
      set({ users: response.data.items, loading: false });
    } catch (error: any) {
      set({
        error: error.message || '获取用户列表失败',
        loading: false
      });
    }
  },

  // 创建用户
  createUser: async (data) => {
    set({ loading: true, error: null });
    try {
      const response = await userApi.createUser(data);
      set((state) => ({
        users: [...state.users, response.data],
        loading: false
      }));
    } catch (error: any) {
      set({
        error: error.message || '创建用户失败',
        loading: false
      });
      throw error;
    }
  },

  // 更新用户
  updateUser: async (id, data) => {
    set({ loading: true, error: null });
    try {
      const response = await userApi.updateUser(id, data);
      set((state) => ({
        users: state.users.map(u => u.id === id ? response.data : u),
        loading: false
      }));
    } catch (error: any) {
      set({
        error: error.message || '更新用户失败',
        loading: false
      });
      throw error;
    }
  },

  // 删除用户
  deleteUser: async (id) => {
    set({ loading: true, error: null });
    try {
      await userApi.deleteUser(id);
      set((state) => ({
        users: state.users.filter(u => u.id !== id),
        loading: false
      }));
    } catch (error: any) {
      set({
        error: error.message || '删除用户失败',
        loading: false
      });
      throw error;
    }
  },

  // 设置当前用户
  setCurrentUser: (user) => {
    set({ currentUser: user });
  },

  // 清除错误
  clearError: () => {
    set({ error: null });
  },
}));
```

```typescript
interface TaskFilterState {
  status?: string;
  keyword?: string;
  dateRange?: [Date, Date];
  sortBy?: string;
  sortOrder?: 'asc' | 'desc';
}

interface TaskUIState {
  // 筛选状态
  filters: TaskFilterState;

  // 选择状态
  selectedTaskIds: number[];

  // 视图状态
  viewMode: 'list' | 'card' | 'timeline';

  // 编辑状态
  editingTask: Task | null;

  // Actions
  setFilters: (filters: TaskFilterState) => void;
  clearFilters: () => void;
  setSelectedTasks: (ids: number[]) => void;
  toggleTaskSelection: (id: number) => void;
  clearSelection: () => void;
  setViewMode: (mode: 'list' | 'card' | 'timeline') => void;
  setEditingTask: (task: Task | null) => void;
}

export const useTaskStore = create<TaskUIState>((set) => ({
  filters: {},
  selectedTaskIds: [],
  viewMode: 'list',
  editingTask: null,

  setFilters: (filters) => set({ filters }),
  clearFilters: () => set({ filters: {} }),

  setSelectedTasks: (ids) => set({ selectedTaskIds: ids }),
  toggleTaskSelection: (id) => {
    set((state) => {
      const index = state.selectedTaskIds.indexOf(id);
      if (index > -1) {
        return { selectedTaskIds: state.selectedTaskIds.filter(i => i !== id) };
      } else {
        return { selectedTaskIds: [...state.selectedTaskIds, id] };
      }
    });
  },
  clearSelection: () => set({ selectedTaskIds: [] }),

  setViewMode: (viewMode) => set({ viewMode }),
  setEditingTask: (editingTask) => set({ editingTask }),
}));
```

### 3.1 QueryClient配置

```typescript
// src/api/queryClient.ts
import { QueryClient } from '@tanstack/react-query';

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      // 缓存配置
      staleTime: 5 * 60 * 1000,       // 5分钟内数据视为新鲜
      gcTime: 30 * 60 * 1000,         // 30分钟后清理缓存

      // 重试配置
      retry: (failureCount, error: any) => {
        // 认证错误不重试
        if (error?.status === 401) return false;
        // 404不重试
        if (error?.status === 404) return false;
        // 最多重试2次
        return failureCount < 2;
      },

      // 窗口聚焦行为
      refetchOnWindowFocus: false,
      refetchOnReconnect: 'always',

      // 轮询配置
      refetchInterval: false,
    },

    mutations: {
      // 变更重试1次
      retry: 1,

      // 变更失败时自动回滚
      onError: (error) => {
        console.error('Mutation error:', error);
      },

      // 变更成功后自动刷新相关查询
      onSuccess: async (data, variables, context) => {
        // 可以在这里执行额外的刷新逻辑
        await queryClient.invalidateQueries({
          predicate: (query) => {
            // 刷新相关的查询
            return context?.relatedQueries?.includes(query.queryKey[0] as string);
          },
        });
      },
    },
  },
});
```

### 3.3 Hooks封装

```typescript
// src/hooks/useTasks.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { taskApi } from '@/api/modules/task';
import { queryKeys } from '@/api/queryKeys';
import type { Task, TaskQueryFilters, CreateTaskData, UpdateTaskData } from '@/types/models';

// 获取任务列表
export function useTasks(filters?: TaskQueryFilters) {
  return useQuery({
    queryKey: queryKeys.tasks.list(filters || {}),
    queryFn: () => taskApi.getTasks(filters),
  });
}

// 获取任务详情
export function useTask(id: number) {
  return useQuery({
    queryKey: queryKeys.tasks.detail(id),
    queryFn: () => taskApi.getTask(id),
    enabled: !!id, // id存在时才查询
  });
}

// 创建任务
export function useCreateTask() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (data: CreateTaskData) => taskApi.createTask(data),
    onSuccess: () => {
      // 刷新任务列表
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    },
  });
}

// 更新任务
export function useUpdateTask() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ id, data }: { id: number; data: UpdateTaskData }) =>
      taskApi.updateTask(id, data),
    onSuccess: (_, variables) => {
      // 刷新任务详情
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.detail(variables.id) });
      // 刷新任务列表
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    },
  });
}

// 删除任务
export function useDeleteTask() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (id: number) => taskApi.deleteTask(id),
    onSuccess: () => {
      // 刷新任务列表
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    },
  });
}

// 启动任务
export function useStartTask() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (id: number) => taskApi.startTask(id),
    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.detail(variables.id) });
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    },
  });
}

// 停止任务
export function useStopTask() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (id: number) => taskApi.stopTask(id),
    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.detail(variables.id) });
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    },
  });
}
```

### 3.4 轮询Hook

```typescript
// src/hooks/usePolling.ts
import { useEffect, useRef } from 'react';
import { useQuery } from '@tanstack/react-query';

interface UsePollingOptions<T> {
  queryKey: any[];
  queryFn: () => Promise<T>;
  enabled?: boolean;
  interval?: number; // 轮询间隔（毫秒）
  onStop?: (data: T) => boolean; // 返回true停止轮询
}

export function usePolling<T>({
  queryKey,
  queryFn,
  enabled = true,
  interval = 5000,
  onStop,
}: UsePollingOptions<T>) {
  const intervalRef = useRef<NodeJS.Timeout>();

  const query = useQuery({
    queryKey,
    queryFn,
    enabled,
    refetchInterval: (data) => {
      // 检查是否应该停止轮询
      if (data && onStop?.(data)) {
        return false;
      }
      return interval;
    },
  });

  // 清理定时器
  useEffect(() => {
    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, []);

  return query;
}

// 使用示例：轮询任务状态直到完成
function TaskStatus({ taskId }: { taskId: number }) {
  const { data: task } = usePolling({
    queryKey: queryKeys.tasks.detail(taskId),
    queryFn: () => taskApi.getTask(taskId),
    interval: 3000, // 每3秒轮询
    onStop: (task) => {
      // 任务完成或失败时停止轮询
      return ['completed', 'failed', 'cancelled'].includes(task.status);
    },
  });

  return (
    <Tag color={getStatusColor(task?.status)}>
      {task?.status}
    </Tag>
  );
}
```

### 4.3 状态同步策略

```typescript
// src/hooks/useTaskRealtime.ts
import { useEffect } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import { useWebSocket } from './useWebSocket';

// WebSocket实时更新同步到TanStack Query缓存
export function useTaskRealtime() {
  const queryClient = useQueryClient();
  const { lastMessage } = useWebSocket(`${WS_URL}/tasks`);

  useEffect(() => {
    if (lastMessage?.type === 'task_updated') {
      const { task_id } = lastMessage;

      // 使相关查询失效，触发重新获取
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.detail(task_id) });
      queryClient.invalidateQueries({ queryKey: queryKeys.tasks.lists() });
    }
  }, [lastMessage, queryClient]);
}
```
