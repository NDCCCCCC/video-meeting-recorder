# 前端认证流程

## 一、认证流程概述

### 1.1 认证流程图

```
┌─────────────────────────────────────────────────────────────┐
│                       认证流程                                │
└─────────────────────────────────────────────────────────────┘
    │
    ├── 登录流程
    │   ├── 用户输入凭证 → API验证 → 获取Token → 存储Token → 跳转首页
    │   └── 失败处理 → 显示错误信息
    │
    ├── Token刷新流程
    │   ├── 检测Token即将过期 → 自动刷新 → 更新存储中的Token
    │   └── 刷新失败 → 强制登出
    │
    └── 登出流程
        ├── 清除本地Token → 调用登出API → 跳转登录页
```

### 1.2 Token管理策略

| 策略 | 说明 |
|------|------|
| Access Token | 短期有效（默认2小时），用于API调用 |
| Refresh Token | 长期有效（默认7天），用于刷新Access Token |
| 自动刷新 | Token过期前5分钟自动刷新 |
| 刷新失败 | 强制用户重新登录 |

### 2.2 登录状态管理

```typescript
// src/stores/authStore.ts (关键部分)
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { authApi } from '@/api/modules/auth';

interface AuthState {
  token: string | null;
  refreshToken: string | null;
  isAuthenticated: boolean;
  user: User | null;
  permissions: string[];

  // Actions
  login: (credentials: LoginRequest) => Promise<void>;
  logout: () => void;
  refreshAccessToken: () => Promise<void>;
  updateUser: (user: User) => void;
  setPermissions: (permissions: string[]) => void;
  hasPermission: (permission: string) => boolean;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      token: null,
      refreshToken: null,
      isAuthenticated: false,
      user: null,
      permissions: [],

      login: async (credentials) => {
        try {
          const response = await authApi.login(credentials);
          const { token, refresh_token, user, permissions } = response.data;

          set({
            token,
            refreshToken: refresh_token,
            isAuthenticated: true,
            user,
            permissions,
          });

          // 启动Token自动刷新
          startTokenRefresh();
        } catch (error: any) {
          // 清除可能存在的旧token
          set({
            token: null,
            refreshToken: null,
            isAuthenticated: false,
            user: null,
            permissions: [],
          });
          throw error;
        }
      },

      logout: () => {
        // 调用登出API
        authApi.logout().catch(console.error);

        // 清除状态
        set({
          token: null,
          refreshToken: null,
          isAuthenticated: false,
          user: null,
          permissions: [],
        });

        // 停止Token刷新
        stopTokenRefresh();

        // 跳转到登录页
        window.location.href = '/auth/login';
      },

      refreshAccessToken: async () => {
        const { refreshToken } = get();
        if (!refreshToken) {
          throw new Error('No refresh token available');
        }

        try {
          const response = await authApi.refreshToken(refreshToken);
          const { token, refresh_token } = response.data;

          set({
            token,
            refreshToken: refresh_token,
          });
        } catch (error) {
          // 刷新失败，强制登出
          get().logout();
          throw error;
        }
      },
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({
        token: state.token,
        refreshToken: state.refreshToken,
        user: state.user,
        permissions: state.permissions,
      }),
    }
  )
);

// Token刷新定时器
let tokenRefreshTimer: NodeJS.Timeout | null = null;

function startTokenRefresh() {
  // 清除现有定时器
  if (tokenRefreshTimer) {
    clearTimeout(tokenRefreshTimer);
  }

  // 检查Token有效期，在过期前5分钟刷新
  const checkAndRefresh = () => {
    const token = useAuthStore.getState().token;
    if (!token) {
      return;
    }

    try {
      const payload = parseJWT(token);
      const expiresIn = payload.exp * 1000 - Date.now();

      // 如果Token在5分钟内过期，刷新
      if (expiresIn < 5 * 60 * 1000) {
        useAuthStore.getState().refreshAccessToken();
      }

      // 设置下次检查
      const nextCheck = Math.max(expiresIn - 5 * 60 * 1000, 60000); // 最少1分钟
      tokenRefreshTimer = setTimeout(checkAndRefresh, nextCheck);
    } catch (error) {
      console.error('Token refresh check error:', error);
      // Token解析失败，强制登出
      useAuthStore.getState().logout();
    }
  };

  // 立即检查一次
  checkAndRefresh();
}

function stopTokenRefresh() {
  if (tokenRefreshTimer) {
    clearTimeout(tokenRefreshTimer);
    tokenRefreshTimer = null;
  }
}

// 解析JWT Token
function parseJWT(token: string) {
  const base64Url = token.split('.')[1];
  const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
  const jsonPayload = decodeURIComponent(atob(base64));
  return JSON.parse(jsonPayload);
}
```

```typescript
// 密码强度检查
export function checkPasswordStrength(password: string): {
  strength: 'weak' | 'medium' | 'strong';
  score: number;
  suggestions: string[];
} {
  let score = 0;
  const suggestions: string[] = [];

  // 长度检查
  if (password.length >= 8) {
    score += 1;
  } else {
    suggestions.push('密码至少8个字符');
  }

  if (password.length >= 12) {
    score += 1;
  }

  // 包含小写字母
  if (/[a-z]/.test(password)) {
    score += 1;
  } else {
    suggestions.push('包含小写字母');
  }

  // 包含大写字母
  if (/[A-Z]/.test(password)) {
    score += 1;
  } else {
    suggestions.push('包含大写字母');
  }

  // 包含数字
  if (/\d/.test(password)) {
    score += 1;
  } else {
    suggestions.push('包含数字');
  }

  // 包含特殊字符
  if (/[@$!%*?&]/.test(password)) {
    score += 1;
  } else {
    suggestions.push('包含特殊字符 (@$!%*?&)');
  }

  // 计算强度
  let strength: 'weak' | 'medium' | 'strong';
  if (score >= 5) {
    strength = 'strong';
  } else if (score >= 3) {
    strength = 'medium';
  }

  return { strength, score, suggestions };
}
```

## 三、路由认证保护

### 3.1 认证Provider

```typescript
// src/providers/AuthProvider.tsx
import { useEffect, useState } from 'react';
import { useAuthStore } from '@/stores/authStore';
import { useNavigate, useLocation } from 'react-router-dom';
import { Spin } from 'antd';

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const navigate = useNavigate();
  const location = useLocation();
  const { isAuthenticated, token, refreshAccessToken } = useAuthStore();
  const [isInitialized, setIsInitialized] = useState(false);

  useEffect(() => {
    const initAuth = async () => {
      // 如果有token但未初始化，尝试验证并刷新
      if (token && !isInitialized) {
        try {
          // 验证token是否有效
          await refreshAccessToken();
          setIsInitialized(true);
        } catch (error) {
          // Token无效，清除状态
          console.error('Token validation failed:', error);
        }
      } else {
        setIsInitialized(true);
      }
    };

    initAuth();
  }, [token]);

  // 保护路由：未登录时重定向
  useEffect(() => {
    if (isInitialized && !isAuthenticated) {
      // 当前页面不是登录页时，记录并跳转
      if (location.pathname !== '/auth/login') {
        navigate(`/auth/login?redirect=${encodeURIComponent(location.pathname)}`, {
          replace: true,
        });
      }
    }
  }, [isInitialized, isAuthenticated, location, navigate]);

  // 显示加载状态
  if (!isInitialized) {
    return (
      <div style={{
        display: 'flex',
        justifyContent: 'center',
        alignItems: 'center',
        height: '100vh',
      }}>
        <Spin size="large" />
      </div>
    );
  }

  return <>{children}</>;
}
```

```typescript
// src/components/ProtectedRoute.tsx
import { Navigate } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';

interface ProtectedRouteProps {
  children: React.ReactNode;
  permission?: string | string[];
  fallback?: React.ReactNode;
}

export function ProtectedRoute({ children, permission, fallback }: ProtectedRouteProps) {
  const { isAuthenticated, hasPermission, hasAnyPermission } = useAuthStore();

  // 检查认证状态
  if (!isAuthenticated) {
    return <Navigate to="/auth/login" replace />;
  }

  // 检查权限
  if (permission) {
    const hasAccess = Array.isArray(permission)
      ? hasAnyPermission(permission)
      : hasPermission(permission);

    if (!hasAccess) {
      return fallback || <Navigate to="/403" replace />;
    }
  }

  return <>{children}</>;
}

// 使用示例
function AppRoutes() {
  return (
    <Routes>
      {/* 公开路由 */}

          <ProtectedRoute>
            <Dashboard />
          </ProtectedRoute>
```

### 4.2 自动登出（Token过期）

```typescript
// src/api/interceptors.ts (在响应拦截器中)
import { message } from 'antd';
import { useAuthStore } from '@/stores/authStore';

instance.interceptors.response.use(
  (response) => response,
  (error) => {
    const originalRequest = error.config;

    // 401错误处理
    if (error.response?.status === 401 && !originalRequest._retry) {
      const { refreshToken, logout } = useAuthStore.getState();

      // 如果有refresh token，尝试刷新
      if (refreshToken) {
        return useAuthStore.getState().refreshAccessToken()
          .then(() => {
            // 刷新成功，重试原请求
            originalRequest._retry = true;
            return instance(originalRequest);
          })
          .catch(() => {
            // 刷新失败，强制登出
            logout();
            message.error('登录已过期，请重新登录');
            return Promise.reject(error);
          });
      } else {
        // 没有refresh token，直接登出
        logout();
        message.error('登录已过期，请重新登录');
      }
    }

    return Promise.reject(error);
  }
);
```

## 五、会话超时处理

### 5.1 会话超时Hook

```typescript
// src/hooks/useSessionTimeout.ts
import { useEffect, useRef } from 'react';
import { useAuthStore } from '@/stores/authStore';
import { Modal } from 'antd';

// 会话超时时间（分钟）
const SESSION_TIMEOUT_MINUTES = 30;
// 警告提前时间（分钟）
const WARNING_ADVANCE_MINUTES = 5;

export function useSessionTimeout() {
  const { logout, isAuthenticated } = useAuthStore();
  const activityTimerRef = useRef<NodeJS.Timeout>();
  const warningTimerRef = useRef<NodeJS.Timeout>();
  const timeoutTimerRef = useRef<NodeJS.Timeout>();

  // 重置活动计时器
  const resetActivityTimer = () => {
    // 清除所有定时器
    if (activityTimerRef.current) {
      clearTimeout(activityTimerRef.current);
    }
    if (warningTimerRef.current) {
      clearTimeout(warningTimerRef.current);
    }
    if (timeoutTimerRef.current) {
      clearTimeout(timeoutTimerRef.current);
    }

    if (!isAuthenticated) {
      return;
    }

    // 设置活动计时器
    activityTimerRef.current = setTimeout(() => {
      showWarningModal();
    }, (SESSION_TIMEOUT_MINUTES - WARNING_ADVANCE_MINUTES) * 60 * 1000);
  };

  // 显示警告对话框
  const showWarningModal = () => {
    let secondsRemaining = WARNING_ADVANCE_MINUTES * 60;

    const modal = Modal.warning({
      title: '会话即将过期',
      content: `由于长时间未操作，您的会话将在 ${secondsRemaining} 秒后过期。是否继续？`,
      okText: '继续使用',
      cancelText: '退出登录',
      onOk: () => {
        // 用户选择继续，重置计时器
        resetActivityTimer();
      },
      onCancel: () => {
        // 用户选择退出
        logout();
      },
    });

    // 倒计时更新
    const countdownInterval = setInterval(() => {
      secondsRemaining--;
      if (secondsRemaining <= 0) {
        clearInterval(countdownInterval);
        modal.destroy();
        logout();
      } else {
        modal.update({
          content: `由于长时间未操作，您的会话将在 ${secondsRemaining} 秒后过期。是否继续？`,
        });
      }
    }, 1000);

    // 设置过期计时器
    timeoutTimerRef.current = setTimeout(() => {
      clearInterval(countdownInterval);
      modal.destroy();
      logout();
    }, WARNING_ADVANCE_MINUTES * 60 * 1000);
  };

  // 监听用户活动
  useEffect(() => {
    if (!isAuthenticated) {
      return;
    }

    const activityEvents = [
      'mousedown',
      'mousemove',
      'keypress',
      'scroll',
      'touchstart',
    ];

    const handleActivity = () => {
      resetActivityTimer();
    };

    // 添加事件监听
    activityEvents.forEach(event => {
      document.addEventListener(event, handleActivity);
    });

    // 初始化计时器
    resetActivityTimer();

    // 清理函数
    return () => {
      activityEvents.forEach(event => {
        document.removeEventListener(event, handleActivity);
      });
      if (activityTimerRef.current) {
        clearTimeout(activityTimerRef.current);
      }
      if (warningTimerRef.current) {
        clearTimeout(warningTimerRef.current);
      }
      if (timeoutTimerRef.current) {
        clearTimeout(timeoutTimerRef.current);
      }
    };
  }, [isAuthenticated]);
}
```

### 6.1 OAuth2.0登录流程

```typescript
// src/pages/auth/OAuthCallback.tsx
import { useEffect } from 'react';
import { useNavigate, useSearchParams } from 'react-router-dom';
import { Spin, Result } from 'antd';
import { useAuthStore } from '@/stores/authStore';
import { oauthApi } from '@/api/modules/oauth';

export default function OAuthCallback() {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();
  const { login } = useAuthStore();

  const [status, setStatus] = useState<'loading' | 'success' | 'error'>('loading');
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const handleOAuthCallback = async () => {
      const code = searchParams.get('code');
      const state = searchParams.get('state');
      const provider = searchParams.get('provider');

      if (!code) {
        setStatus('error');
        setError('缺少授权码');
        return;
      }

      try {
        // 调用后端OAuth回调接口
        const response = await oauthApi.callback({
          provider,
          code,
          state,
        });

        const { token, refresh_token, user, permissions } = response.data;

        // 保存认证信息
        useAuthStore.getState().setAuthInfo({
          token,
          refreshToken: refresh_token,
          user,
          permissions,
        });

        setStatus('success');

        // 跳转到首页或原页面
        const redirect = searchParams.get('redirect') || '/';
        navigate(redirect);
      } catch (err: any) {
        setStatus('error');
        setError(err.message || '登录失败');
      }
    };

    handleOAuthCallback();
  }, [searchParams]);

  if (status === 'loading') {
    return (
      <div style={{ textAlign: 'center', padding: '50px' }}>
        <Spin size="large" tip="登录中..." />
      </div>
    );
  }

  if (status === 'error') {
    return (
      <Result
        status="error"
        title="登录失败"
        subTitle={error}
        extra={
          <Button type="primary" onClick={() => navigate('/auth/login')}>
          返回登录
          </Button>
        }
      />
    );
  }

  return null;
}

function getOAuthAuthUrl(provider: string): string {
  const baseUrl = import.meta.env.VITE_API_BASE_URL;
  const redirectUri = `${window.location.origin}/auth/callback`;

  switch (provider) {
    case 'dingtalk':
      return `${baseUrl}/oauth/dingtalk/authorize?redirect_uri=${encodeURIComponent(redirectUri)}`;
    case 'wechat':
      return `${baseUrl}/oauth/wechat/authorize?redirect_uri=${encodeURIComponent(redirectUri)}`;
    case 'feishu':
      return `${baseUrl}/oauth/feishu/authorize?redirect_uri=${encodeURIComponent(redirectUri)}`;
    default:
      throw new Error(`Unsupported OAuth provider: ${provider}`);
  }
}
```
