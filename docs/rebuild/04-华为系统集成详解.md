# 华为系统集成详解

## 一、华为集成概述

### 1.1 华为视频会议系统

系统通过华为会议系统的REST API进行集成，实现：

- **终端呼叫控制** - 自动呼叫会议号加入会议
- **会议状态监控** - 获取会议状态和参与者信息
- **RTSP流获取** - 获取会议直播流地址
- **终端设备管理** - 管理华为终端设备的连接状态

## 四、HuaweiClient 实现

### 4.1 会话管理

```go
// internal/huawei/client.go
type HuaweiClient struct {
    config     *config.HuaweiConfig
    httpClient *http.Client
    session    *Session
    logger     *zap.Logger
    mu         sync.RWMutex
}

type Session struct {
    ID        string
    ExpiresAt time.Time
    mu        sync.RWMutex
}

// 登录获取会话
func (c *HuaweiClient) Login(ctx context.Context) error {
    url := fmt.Sprintf("%s/sessions", c.config.APIBase)

    req := map[string]string{
        "username": c.config.Username,
        "password": c.config.Password,
    }

    resp, err := c.httpClient.Post(ctx, url, req)
    if err != nil {
        return fmt.Errorf("登录失败: %w", err)
    }

    if resp.Code != 0 {
        return fmt.Errorf("登录失败: %s", resp.Msg)
    }

    c.mu.Lock()
    c.session = &Session{
        ID:        resp.Data.SessionID,
        ExpiresAt: time.Now().Add(time.Duration(resp.Data.ExpiresIn) * time.Second),
    }
    c.mu.Unlock()

    return nil
}

// 保活会话
func (c *HuaweiClient) KeepAlive(ctx context.Context) error {
    c.mu.RLock()
    if c.session == nil {
        c.mu.RUnlock()
        return errors.New("未登录")
    }

    sessionID := c.session.ID
    c.mu.RUnlock()

    url := fmt.Sprintf("%s/sessions/%s/keep-alive", c.config.APIBase, sessionID)

    _, err := c.httpClient.Post(ctx, url, nil)
    if err != nil {
        // 保活失败，尝试重新登录
        return c.Login(ctx)
    }

    // 更新过期时间
    c.mu.Lock()
    c.session.ExpiresAt = time.Now().Add(time.Duration(c.config.SessionTimeout) * time.Second)
    c.mu.Unlock()

    return nil
}

// 自动保活
func (c *HuaweiClient) StartKeepAlive(ctx context.Context) {
    ticker := time.NewTicker(time.Duration(c.config.KeepAliveInterval) * time.Second)
    defer ticker.Stop()

    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            if err := c.KeepAlive(ctx); err != nil {
                c.logger.Error("会话保活失败", zap.Error(err))
            }
        }
    }
}
```

```go
// SafeCallConference 先挂断再呼叫，确保连接状态
func (c *HuaweiClient) SafeCallConference(ctx context.Context, req *CallConferenceRequest) error {
    // 1. 先尝试挂断（清理可能的残留连接）
    hangupReq := &HangupConferenceRequest{
        TerminalNumber: req.TerminalNumber,
    }
    _ = c.HangupConference(ctx, hangupReq) // 忽略错误

    // 2. 等待一小段时间
    time.Sleep(1 * time.Second)

    // 3. 再呼叫
    return c.CallConference(ctx, req)
}

// 呼叫会议
func (c *HuaweiClient) CallConference(ctx context.Context, req *CallConferenceRequest) error {
    url := fmt.Sprintf("%s/confctrl/conferences/call", c.config.APIBase)

    // 添加会话ID
    headers := map[string]string{

        return fmt.Errorf("呼叫失败: %w", err)
    }

    if resp.Code != 0 {
        return fmt.Errorf("呼叫失败: %s", resp.Msg)
    }

    return nil
}
```

### 4.3 HTTP客户端封装

```go
// HTTPClient 封装HTTP通信
type HTTPClient struct {
    client *http.Client
    logger *zap.Logger
}

func NewHTTPClient(timeout time.Duration) *HTTPClient {
    return &HTTPClient{
        client: &http.Client{
            Timeout: timeout,
            Transport: &http.Transport{
                TLSClientConfig: &tls.Config{
                    InsecureSkipVerify: true, // 兼容旧设备
                    MinVersion:         tls.VersionTLS10,
                },
                MaxIdleConns:        10,
                MaxIdleConnsPerHost: 5,
                IdleConnTimeout:     90 * time.Second,
            },
        },
    }
}

// Post 发送POST请求
func (c *HTTPClient) Post(ctx context.Context, url string, body interface{}, headers ...map[string]string) (*APIResponse, error) {
    var reqBody io.Reader
    if body != nil {
        jsonBody, err := json.Marshal(body)
        if err != nil {
            return nil, err
        }
        reqBody = bytes.NewReader(jsonBody)
    }

    req, err := http.NewRequestWithContext(ctx, "POST", url, reqBody)
    if err != nil {
        return nil, err
    }

    req.Header.Set("Content-Type", "application/json")
    for _, h := range headers {
        for k, v := range h {
            req.Header.Set(k, v)
        }
    }

    resp, err := c.client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    respBody, err := io.ReadAll(resp.Body)
    if err != nil {
        return nil, err
    }

    var apiResp APIResponse
    if err := json.Unmarshal(respBody, &apiResp); err != nil {
        return nil, err
    }

    return &apiResp, nil
}

// APIResponse 统一API响应格式
type APIResponse struct {
    Code int         `json:"code"`
    Msg  string      `json:"msg"`
    Data interface{} `json:"data"`
}
```

### 5.1 连接器职责

```go
// internal/services/video_recording/huawei_conference_connector.go
type HuaweiConferenceConnector struct {
    huaweiService *huawei.HuaweiService
    logger        *zap.Logger
}

// ConnectToConference 连接华为会议
func (c *HuaweiConferenceConnector) ConnectToConference(task *models.VideoRecordingTask) error {
    c.logger.Info("开始连接华为会议",
        zap.Uint("task_id", task.ID),
        zap.String("conference_number", task.ConferenceNumber),
    )

    // 1. 加载华为配置
    config, err := c.getHuaweiConfig(task.HuaweiConfigID)
    if err != nil {
        return fmt.Errorf("加载华为配置失败: %w", err)
    }

    // 2. 锁定终端
    if err := c.LockTerminal(task.ID, config); err != nil {
        return fmt.Errorf("锁定终端失败: %w", err)
    }

    // 3. 呼叫会议
    callReq := &huawei.CallConferenceRequest{
        ConferenceNumber: task.ConferenceNumber,
        TerminalNumber:   config.TerminalNumber,
        Password:         "", // 如果会议有密码
        Subject:          fmt.Sprintf("录制任务: %s", task.Name),
    }

    if err := c.huaweiService.SafeCallConference(context.Background(), callReq); err != nil {
        c.UnlockTerminal(config) // 释放锁
        return fmt.Errorf("呼叫会议失败: %w", err)
    }

    // 4. 等待连接确认
    if err := c.waitForConnection(task.ConferenceNumber, 30*time.Second); err != nil {
        c.UnlockTerminal(config)
        return fmt.Errorf("等待连接失败: %w", err)
    }

    c.logger.Info("连接华为会议成功",
        zap.Uint("task_id", task.ID),
        zap.String("conference_number", task.ConferenceNumber),
    )

    return nil
}

// DisconnectFromConference 断开华为会议
func (c *HuaweiConferenceConnector) DisconnectFromConference(task *models.VideoRecordingTask) error {
    c.logger.Info("断开华为会议",
        zap.Uint("task_id", task.ID),
        zap.String("conference_number", task.ConferenceNumber),
    )

    // 1. 加载华为配置
    config, err := c.getHuaweiConfig(task.HuaweiConfigID)
    if err != nil {
        return fmt.Errorf("加载华为配置失败: %w", err)
    }

    // 2. 挂断会议
    hangupReq := &huawei.HangupConferenceRequest{
        TerminalNumber: config.TerminalNumber,
    }

    _ = c.huaweiService.HangupConference(context.Background(), hangupReq) // 忽略错误

    // 3. 解锁终端
    if err := c.UnlockTerminal(config); err != nil {
        c.logger.Warn("解锁终端失败", zap.Error(err))
    }

    return nil
}
```

### 5.2 终端锁定机制

```go
// LockTerminal 锁定终端
func (c *HuaweiConferenceConnector) LockTerminal(taskID uint, config *models.HuaweiConfig) error {
    // 检查是否已被锁定
    if config.IsLocked {
        // 检查是否被同一任务锁定
        if config.LockedBy != nil && *config.LockedBy == taskID {
            return nil // 已被本任务锁定
        }
        return fmt.Errorf("终端已被任务 %d 锁定", *config.LockedBy)
    }

    // 获取终端状态
    status, err := c.huaweiService.GetTerminalStatus(context.Background(), config.TerminalNumber)
    if err != nil {
        return fmt.Errorf("获取终端状态失败: %w", err)
    }

    // 检查终端是否空闲
    if status.Status != "idle" {
        return fmt.Errorf("终端状态不是空闲: %s", status.Status)
    }

    // 锁定终端
    config.IsLocked = true
    config.LockedBy = &taskID
    now := time.Now()
    config.LockedAt = &now

    if err := c.db.Save(config).Error; err != nil {
        return fmt.Errorf("保存锁定状态失败: %w", err)
    }

    return nil
}

// UnlockTerminal 解锁终端
func (c *HuaweiConferenceConnector) UnlockTerminal(config *models.HuaweiConfig) error {
    config.IsLocked = false
    config.LockedBy = nil
    config.LockedAt = nil

    if err := c.db.Save(config).Error; err != nil {
        return fmt.Errorf("保存解锁状态失败: %w", err)
    }

    return nil
}
```

### 5.3 等待连接确认

```go
// waitForConnection 等待连接确认
func (c *HuaweiConferenceConnector) waitForConnection(conferenceNumber string, timeout time.Duration) error {
    ctx, cancel := context.WithTimeout(context.Background(), timeout)
    defer cancel()

    ticker := time.NewTicker(2 * time.Second)
    defer ticker.Stop()

    for {
        select {
        case <-ctx.Done():
            return errors.New("等待连接超时")
        case <-ticker.C:
            // 获取会议信息
            info, err := c.huaweiService.GetConferenceInfo(ctx, conferenceNumber)
            if err != nil {
                continue
            }

            // 检查终端是否已加入
            for _, attendee := range info.Attendees {
                if attendee.Status == "connected" {
                    return nil
                }
            }
        }
    }
}
```
