# 操作审计日志系统

## 一、系统概述

### 1.1 设计目标

操作审计日志系统用于记录系统中所有重要操作的完整追踪信息，满足以下需求：

- **合规性要求** - 满足等保2.0、SOX、ISO27001等审计要求
- **安全追溯** - 完整记录用户操作行为，支持安全事件追溯
- **问题排查** - 提供详细的操作记录，辅助问题定位和排查
- **数据范围权限** - 根据用户权限过滤可查看的审计日志

### 1.2 核心特性

| 特性 | 说明 |
|------|------|
| **完整记录** | 记录操作前后数据变化 |
| **高性能** | 异步写入，不阻塞主流程 |
| **可追溯** | 请求链路追踪（Request ID） |
| **可查询** | 多维度查询和筛选 |
| **数据脱敏** | 敏感信息自动脱敏 |
| **数据归档** | 支持日志定期归档和清理 |

### 2.2 数据脱敏规则

```go
// internal/services/audit/sanitizer.go
package audit

import (
    "regexp"
    "strings"
)

// Sanitizer 脱敏器
type Sanitizer struct {
    rules []SanitizeRule
}

// SanitizeRule 脱敏规则
type SanitizeRule struct {
    Field    string
    Pattern  *regexp.Regexp
    Replace  string
}

// NewSanitizer 创建脱敏器
func NewSanitizer() *Sanitizer {
    return &Sanitizer{
        rules: []SanitizeRule{
            // 密码字段
            {Field: "password", Pattern: regexp.MustCompile("."), Replace: "******"},
            {Field: "password_hash", Pattern: regexp.MustCompile("."), Replace: "******"},

            // 手机号
            {Field: "phone", Pattern: regexp.MustCompile(`(\d{3})\d{4}(\d{4})`), Replace: "$1****$2"},
            {Field: "mobile", Pattern: regexp.MustCompile(`(\d{3})\d{4}(\d{4})`), Replace: "$1****$2"},

            // 身份证
            {Field: "id_card", Pattern: regexp.MustCompile(`(\d{6})\d{8}(\d{4})`), Replace: "$1********$2"},

            // 邮箱
            {Field: "email", Pattern: regexp.MustCompile(`(.{2}).*@(.*)`), Replace: "$1***@$2"},

            // IP地址（可选，根据安全要求）
            {Field: "ip_address", Pattern: regexp.MustCompile(`(\d+\.\d+)\.\d+\.\d+`), Replace: "$1.*.*"},
        },
    }
}

// Sanitize 对数据进行脱敏处理
func (s *Sanitizer) Sanitize(data interface{}) interface{} {
    // 实现脱敏逻辑
    switch v := data.(type) {
    case map[string]interface{}:
        return s.sanitizeMap(v)
    case string:
        return s.sanitizeString("", v)
    default:
        return data
    }
}

func (s *Sanitizer) sanitizeMap(data map[string]interface{}) map[string]interface{} {
    result := make(map[string]interface{})
    for key, value := range data {
        if strValue, ok := value.(string); ok {
            result[key] = s.sanitizeString(key, strValue)
        } else if mapValue, ok := value.(map[string]interface{}); ok {
            result[key] = s.sanitizeMap(mapValue)
        } else {
            result[key] = value
        }
    }
    return result
}

func (s *Sanitizer) sanitizeString(field, value string) string {
    // 检查字段名是否需要脱敏
    for _, rule := range s.rules {
        if strings.Contains(strings.ToLower(field), rule.Field) {
            return rule.Pattern.ReplaceAllString(value, rule.Replace)
        }
    }
    return value
}
```

```go
// AuditLogService 审计日志服务
type AuditLogService struct {

    logger     *zap.Logger
    sanitizer  *Sanitizer

// NewAuditLogService 创建审计日志服务
func NewAuditLogService(db *gorm.DB, logger *zap.Logger) *AuditLogService {
    service := &AuditLogService{
        db:         db,
        logger:     logger,
        sanitizer:  NewSanitizer(),
        asyncQueue: make(chan *AuditLog, 1000),
    }

    // 启动异步处理goroutine
    go service.processQueue()

    return service
}

// LogOperation 记录操作日志（异步）
func (s *AuditLogService) LogOperation(ctx context.Context, req *LogOperationRequest) error {
    // 1. 脱敏处理
    if req.OldData != nil {
        req.OldData = s.sanitizer.Sanitize(req.OldData)
    }
    if req.NewData != nil {
        req.NewData = s.sanitizer.Sanitize(req.NewData)
    }

    // 2. 生成差异
    diff := s.generateDiff(req.OldData, req.NewData)

    // 3. 序列化数据
    oldDataJSON, _ := json.Marshal(req.OldData)
    newDataJSON, _ := json.Marshal(req.NewData)
    diffJSON, _ := json.Marshal(diff)

    // 4. 创建日志记录
    log := &AuditLog{
        UserID:     req.UserID,
        Username:   req.Username,
        RoleID:     req.RoleID,
        RoleName:   req.RoleName,
        Action:     req.Action,
        Module:     req.Module,
        Resource:   req.Resource,
        ResourceID: req.ResourceID,
        RequestID:  req.RequestID,
        TraceID:    req.TraceID,
        Method:     req.Method,
        Path:       req.Path,
        OldData:    string(oldDataJSON),
        NewData:    string(newDataJSON),
        DiffData:   string(diffJSON),
        Status:     req.Status,
        ErrorMsg:   req.ErrorMsg,
        ErrorCode:  req.ErrorCode,
        IPAddress:  req.IPAddress,
        UserAgent:  req.UserAgent,
        Duration:   req.Duration,
        CreatedAt:  time.Now(),
    }

    // 5. 放入异步队列
    select {
    case s.asyncQueue <- log:
        return nil
    default:
        // 队列满，同步写入
        return s.db.Create(log).Error
    }
}

// processQueue 异步处理队列
func (s *AuditLogService) processQueue() {
    ticker := time.NewTicker(time.Second)
    defer ticker.Stop()

    batch := make([]*AuditLog, 0, 100)

    for {
        select {
        case log := <-s.asyncQueue:
            batch = append(batch, log)
            if len(batch) >= 100 {
                s.flushBatch(batch)
                batch = make([]*AuditLog, 0, 100)
            }
        case <-ticker.C:
            if len(batch) > 0 {
                s.flushBatch(batch)
                batch = make([]*AuditLog, 0, 100)
            }
        }
    }
}

// flushBatch 批量写入
func (s *AuditLogService) flushBatch(batch []*AuditLog) {
    if err := s.db.CreateInBatches(batch, 100).Error; err != nil {
        s.logger.Error("批量写入审计日志失败",
            zap.Int("count", len(batch)),
            zap.Error(err),
        )
    }
}
```

```go
// QueryRequest 查询请求
type QueryRequest struct {
    // 基础筛选
    Module     string    `form:"module"`
    Action     string    `form:"action"`
    Status     string    `form:"status"`
    UserID     uint      `form:"user_id"`
    Username   string    `form:"username"`

    // 时间范围
    StartTime  time.Time `form:"start_time"`
    EndTime    time.Time `form:"end_time"`

    // 搜索
    Keyword    string    `form:"keyword"`

    // 分页
    Page       int       `form:"page" binding:"min=1"`
    PageSize   int       `form:"page_size" binding:"min=1,max=100"`

    // 排序
    OrderBy    string    `form:"order_by"`
    Order      string    `form:"order" binding:"oneof=asc desc"`
}

// QueryResponse 查询响应
type QueryResponse struct {
    Items      []*AuditLog `json:"items"`
    Total      int64        `json:"total"`
    Page       int          `json:"page"`
    PageSize   int          `json:"page_size"`
    TotalPages int          `json:"total_pages"`
}

// Query 查询审计日志（带数据范围权限）
func (s *AuditLogService) Query(ctx context.Context, req *QueryRequest, user *User) (*QueryResponse, error) {
    query := s.db.Model(&AuditLog{})

    // 应用数据范围权限（个人只能看自己的日志）
    scope := ctx.Value("data_scope").(*DataScope)
    if scope.Type == DataScopeSelf {
        query = query.Where("user_id = ?", user.ID)
    }

    // 应用筛选条件
    if req.Module != "" {
        query = query.Where("module = ?", req.Module)
    }
    if req.Action != "" {
        query = query.Where("action = ?", req.Action)
    }
    if req.Status != "" {
        query = query.Where("status = ?", req.Status)
    }
    if req.Username != "" {
        query = query.Where("username LIKE ?", "%"+req.Username+"%")
    }
    if !req.StartTime.IsZero() {
        query = query.Where("created_at >= ?", req.StartTime)
    }
    if !req.EndTime.IsZero() {
        query = query.Where("created_at <= ?", req.EndTime)
    }
    if req.Keyword != "" {
        query = query.Where("resource LIKE ? OR error_msg LIKE ?",
            "%"+req.Keyword+"%", "%"+req.Keyword+"%")
    }

    // 统计总数
    var total int64
    query.Count(&total)

    // 排序
    orderBy := req.OrderBy
    if orderBy == "" {
        orderBy = "created_at"
    }
    order := req.Order
    if order == "" {
        order = "desc"
    }
    query = query.Order(orderBy + " " + order)

    // 分页
    var logs []*AuditLog
    err := query.Offset((req.Page - 1) * req.PageSize).
        Limit(req.PageSize).
        Find(&logs).Error

    if err != nil {
        return nil, err
    }

    // 计算总页数
    totalPages := int(total) / req.PageSize
    if int(total)%req.PageSize > 0 {
        totalPages++
    }

    return &QueryResponse{
        Items:      logs,
        Total:      total,
        Page:       req.Page,
        PageSize:   req.PageSize,
        TotalPages: totalPages,
    }, nil
}

// GetByID 获取单条日志详情
func (s *AuditLogService) GetByID(ctx context.Context, id uint, user *User) (*AuditLog, error) {
    var log AuditLog
    err := s.db.First(&log, id).Error
    if err != nil {
        return nil, err
    }

    // 检查数据范围权限
    scope := ctx.Value("data_scope").(*DataScope)
    if scope.Type == DataScopeSelf && log.UserID != user.ID {
        return nil, ErrPermissionDenied
    }

    return &log, nil
}
```

```go
package middleware

import (

    "time"

    "github.com/gin-gonic/gin"

)

// AuditMiddleware 审计日志中间件
type AuditMiddleware struct {
    auditService *audit.AuditLogService
    logger       *zap.Logger
}

// AuditOperation 审计操作
func (m *AuditMiddleware) AuditOperation(module, action string) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()

        // 读取请求体
        var requestBody map[string]interface{}
        if c.Request.Method != "GET" && c.Request.ContentLength > 0 {
            body, _ := io.ReadAll(c.Request.Body)
            c.Request.Body = io.NopCloser(bytes.NewBuffer(body))
            json.Unmarshal(body, &requestBody)
        }

        // 处理请求
        c.Next()

        // 构建审计日志请求
        req := &audit.LogOperationRequest{
            UserID:    c.GetUint("user_id"),
            Username:  c.GetString("username"),
            RoleID:    c.GetUint("role_id"),
            Action:    action,
            Module:    module,
            RequestID: c.GetString("request_id"),
            Method:    c.Request.Method,
            Path:      c.Request.URL.Path,
            NewData:   requestBody,
            IPAddress: c.ClientIP(),
            UserAgent: c.Request.UserAgent(),
            Duration:  time.Since(start).Milliseconds(),
        }

        // 根据响应状态设置结果
        if c.Writer.Status() >= 400 {
            req.Status = audit.StatusFailure
            req.ErrorMsg = "请求失败"
        } else {
            req.Status = audit.StatusSuccess
        }

        // 异步记录
        m.auditService.LogOperation(c.Request.Context(), req)
    }
}
```

```go
// internal/handlers/audit_handler.go
package handlers

import (
    "github.com/gin-gonic/gin"

// AuditHandler 审计日志处理器
type AuditHandler struct {
    *BaseHandler
    auditService *audit.AuditLogService
}

// Query 查询审计日志
// @Summary 查询审计日志
// @Tags 审计日志
// @Produce json
// @Param module query string false "模块"
// @Param action query string false "操作"
// @Param status query string false "状态"
// @Param start_time query string false "开始时间"
// @Param end_time query string false "结束时间"
// @Param page query int false "页码" default(1)
// @Param page_size query int false "页大小" default(20)
// @Success 200 {object} Response
// @Router /api/v1/audit/logs [get]
func (h *AuditHandler) Query(c *gin.Context) {
    var req audit.QueryRequest
    if err := c.ShouldBindQuery(&req); err != nil {
        h.ErrorResponse(c, 400, "INVALID_REQUEST", "请求参数错误", err)
        return
    }

    user := h.GetUser(c)
    result, err := h.auditService.Query(c.Request.Context(), &req, user)
    if err != nil {
        h.ErrorResponse(c, 500, "INTERNAL_ERROR", "查询失败", err)
        return
    }

    h.SuccessResponse(c, result)
}

// GetByID 获取日志详情
// @Summary 获取审计日志详情
// @Tags 审计日志
// @Produce json
// @Param id path int true "日志ID"
// @Success 200 {object} Response
// @Router /api/v1/audit/logs/:id [get]

    id, err := strconv.ParseUint(c.Param("id"), 10, 32)

        h.ErrorResponse(c, 404, "NOT_FOUND", "日志不存在", err)
        return
    }

    h.SuccessResponse(c, log)
}

    os.Remove(filepath)

    stats, err := h.auditService.GetStatistics(c.Request.Context(), days)
    if err != nil {
        h.ErrorResponse(c, 500, "INTERNAL_ERROR", "获取统计失败", err)
        return
    }

    h.SuccessResponse(c, stats)
}
```

## 七、数据归档与清理

### 7.1 自动归档

```go
// internal/services/audit/archive.go
package audit

import (
    "time"
    "gorm.io/gorm"
)

// ArchiveService 归档服务
type ArchiveService struct {
    db     *gorm.DB
    logger *zap.Logger
}

// ArchiveOldLogs 归档旧日志
func (s *ArchiveService) ArchiveOldLogs(days int) error {
    cutoffDate := time.Now().AddDate(0, 0, -days)

    // 查询需要归档的日志
    var logs []*AuditLog
    err := s.db.Where("created_at < ?", cutoffDate).Find(&logs).Error
    if err != nil {
        return err
    }

    if len(logs) == 0 {
        return nil
    }

    // 创建归档文件
    archiveFile := fmt.Sprintf("audit_archive_%s.json", time.Now().Format("20060102"))

    // 写入归档文件
    file, err := os.Create(archiveFile)
    if err != nil {
        return err
    }
    defer file.Close()

    encoder := json.NewEncoder(file)
    for _, log := range logs {
        encoder.Encode(log)
    }

    // 删除已归档的日志
    return s.db.Where("created_at < ?", cutoffDate).Delete(&AuditLog{}).Error
}
```

```go
// internal/services/audit/cleanup.go
func (s *CleanupService) CleanupExpiredLogs() error {
    // 1. 清理过期日志（保留1年）
    oneYearAgo := time.Now().AddDate(-1, 0, 0)
    result := s.db.Where("created_at < ? AND status = ?", oneYearAgo, "success").
        Delete(&AuditLog{})

    s.logger.Info("清理过期审计日志",
        zap.Int64("rows", result.RowsAffected),
    )

    return result.Error
}
```
