# 配置管理详解

## 一、配置系统概述

### 1.1 配置工具

系统使用**Viper**作为配置管理工具，支持：

- **多种格式** - JSON, YAML, TOML, HCL
- **环境变量** - 支持环境变量覆盖
- **默认值** - 内置默认值
- **热加载** - 配置文件变更自动重载
- **多环境** - 支持开发、测试、生产环境

### 1.2 配置文件结构

```
config.yaml                 # 主配置文件
config.dev.yaml             # 开发环境配置
config.prod.yaml            # 生产环境配置
.env                        # 环境变量文件
```

## 三、配置加载

### 3.1 配置加载函数

```go
// internal/config/config.go
func LoadConfig(configPath string) (*Config, error) {
    v := viper.New()

    // 1. 设置配置文件
    v.SetConfigFile(configPath)

    // 2. 设置配置文件类型
    v.SetConfigType("yaml")

    // 3. 环境变量前缀
    v.SetEnvPrefix("CONF")
    v.AutomaticEnv()

    // 4. 环境变量分隔符
    v.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))

    // 5. 读取配置文件
    if err := v.ReadInConfig(); err != nil {
        return nil, fmt.Errorf("读取配置文件失败: %w", err)
    }

    // 6. 解析配置
    var config Config
    if err := v.Unmarshal(&config); err != nil {
        return nil, fmt.Errorf("解析配置失败: %w", err)
    }

    // 7. 环境变量替换
    if err := replaceEnvVars(&config); err != nil {
        return nil, fmt.Errorf("环境变量替换失败: %w", err)
    }

    // 8. 验证配置
    if err := config.Validate(); err != nil {
        return nil, fmt.Errorf("配置验证失败: %w", err)
    }

    return &config, nil
}

// replaceEnvVars 替换环境变量
func replaceEnvVars(config *Config) error {
    // 替换华为密码
    if strings.HasPrefix(config.Huawei.Password, "${") {
        envVar := strings.TrimPrefix(strings.TrimSuffix(config.Huawei.Password, "}"), "${")
        if parts := strings.SplitN(envVar, ":", 2); len(parts) == 2 {
            if value := os.Getenv(parts[0]); value != "" {
                config.Huawei.Password = value
            } else {
                config.Huawei.Password = parts[1]
            }
        }
    }

    return nil
}

// Validate 验证配置
func (c *Config) Validate() error {
    // 验证服务器配置
    if c.Server.Port <= 0 || c.Server.Port > 65535 {
        return errors.New("无效的服务器端口")
    }

    // 验证数据库配置
    if c.Database.Database == "" {
        return errors.New("数据库名称不能为空")
    }

    // 验证华为配置
    if c.Huawei.ConferenceServer == "" {
        return errors.New("华为服务器地址不能为空")
    }

    return nil
}
```

## 五、配置热加载

### 5.1 监控配置变更

```go
// WatchConfig 监控配置变更
func WatchConfig(configPath string, callback func(*Config)) error {
    v := viper.New()

    v.SetConfigFile(configPath)
    v.SetConfigType("yaml")

    // 初始读取
    if err := v.ReadInConfig(); err != nil {
        return err
    }

    // 监控配置变更
    v.WatchConfig()

    v.OnConfigChange(func(e fsnotify.Event) {
        log.Info("配置文件已变更", zap.String("file", e.Name))

        var config Config
        if err := v.Unmarshal(&config); err != nil {
            log.Error("解析配置失败", zap.Error(err))
            return
        }

        // 调用回调函数
        callback(&config)
    })

    return nil
}
```

## 六、配置验证

### 6.1 配置验证器

```go
// internal/config/validator.go
type ConfigValidator struct {
    errors []string
}

// ValidateConfig 验证配置
func ValidateConfig(config *Config) error {
    validator := &ConfigValidator{
        errors: make([]string, 0),
    }

    // 验证服务器配置
    validator.validateServerConfig(&config.Server)

    // 验证数据库配置
    validator.validateDatabaseConfig(&config.Database)

    // 验证华为配置
    validator.validateHuaweiConfig(&config.Huawei)

    if len(validator.errors) > 0 {
        return fmt.Errorf("配置验证失败:\n%s", strings.Join(validator.errors, "\n"))
    }

    return nil
}

func (v *ConfigValidator) validateServerConfig(config *ServerConfig) {
    if config.Port <= 0 || config.Port > 65535 {
        v.errors = append(v.errors, "  服务器端口必须在1-65535之间")
    }

    if config.Mode != "debug" && config.Mode != "release" {
        v.errors = append(v.errors, "  运行模式必须是debug或release")
    }
}

func (v *ConfigValidator) validateDatabaseConfig(config *DatabaseConfig) {
    if config.Database == "" {
        v.errors = append(v.errors, "  数据库名称不能为空")
    }

    if config.MaxIdleConns < 0 {
        v.errors = append(v.errors, "  最大空闲连接数不能为负数")
    }

    if config.MaxOpenConns <= 0 {
        v.errors = append(v.errors, "  最大打开连接数必须大于0")
    }
}

func (v *ConfigValidator) validateHuaweiConfig(config *HuaweiConfig) {
    if config.ConferenceServer == "" {
        v.errors = append(v.errors, "  华为服务器地址不能为空")
    }

    if config.Username == "" {
        v.errors = append(v.errors, "  华为用户名不能为空")
    }

    if config.Password == "" {
        v.errors = append(v.errors, "  华为密码不能为空")
    }
}
```
