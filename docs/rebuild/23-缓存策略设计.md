# 缓存策略设计

> **⚠️ 文档状态说明**
>
> 本文档描述的缓存策略**不适用于当前项目架构**。
>
> 根据项目V2.0架构决策：
> - 数据库采用 **SQLite**（纯Go驱动，无需CGO）
> - **不使用 Redis 或任何分布式缓存**
> - **不使用本地缓存层**
>
> 本文档保留作为参考资料，如未来需要引入缓存时可参考。

## 概述

本文档描述了系统的缓存架构设计，包括多级缓存、缓存策略、缓存失效机制以及最佳实践，旨在提高系统性能和减少数据库压力。

> **注意：当前项目不实现缓存策略，以下内容仅供参考。**

## 核心概念

### 缓存类型

| 类型 | 说明 | 速度 | 容量 | 持久化 |
|------|------|------|------|--------|
| 本地缓存 | 进程内内存 | 极快 (ns级) | 小 (MB级) | 否 |
| 分布式缓存 | Redis/Memcached | 快 (μs级) | 大 (GB级) | 可选 |
| CDN缓存 | 边缘节点 | 中等 (ms级) | 极大 (TB级) | 是 |

### 缓存模式

1. **Cache-Aside (旁路缓存)** - 应用程序管理缓存
2. **Read-Through (读穿透)** - 缓存层管理读取
3. **Write-Through (写穿透)** - 写入同时更新缓存和数据库
4. **Write-Behind (写回)** - 异步写入数据库
5. **Refresh-Ahead (预刷新)** - 提前刷新即将过期的缓存

```go
package cache

import (
    "context"
    "time"
)

// Cache 缓存接口
type Cache interface {
    // Get 获取缓存
    Get(ctx context.Context, key string, dest interface{}) error

    // Set 设置缓存
    Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error

    // Delete 删除缓存
    Delete(ctx context.Context, key string) error

    // Exists 检查缓存是否存在
    Exists(ctx context.Context, key string) (bool, error)

    // Clear 清空缓存
    Clear(ctx context.Context) error

    // GetMultiple 批量获取
    GetMultiple(ctx context.Context, keys []string, dest map[string]interface{}) error

    // SetMultiple 批量设置
    SetMultiple(ctx context.Context, items map[string]interface{}, ttl time.Duration) error

    // DeleteMultiple 批量删除
    DeleteMultiple(ctx context.Context, keys []string) error
}

// Stats 缓存统计
type Stats struct {
    Hits       int64 `json:"hits"`
    Misses     int64 `json:"misses"`
    HitRate    float64 `json:"hit_rate"`
    Size       int64 `json:"size"`
    Evictions  int64 `json:"evictions"`
}

// CacheWithStats 支持统计的缓存
type CacheWithStats interface {
    Cache
    GetStats() Stats
    ResetStats()
}
```

## 本地缓存实现

### 1. LRU缓存

**文件位置**: `internal/cache/local/lru.go`

```go
package local

import (
    "container/list"
    "context"
    "sync"
    "time"
)

// LRUCache LRU (Least Recently Used) 缓存
type LRUCache struct {
    capacity int                           // 容量
    items    map[string]*list.Element      // 数据映射
    lru      *list.List                    // LRU链表
    mu       sync.RWMutex                  // 读写锁
    stats    *CacheStats                   // 统计信息
}

// entry 缓存条目
type entry struct {
    key       string
    value     interface{}
    expiredAt time.Time
}

// CacheStats 缓存统计
type CacheStats struct {
    Hits      int64
    Misses    int64
    Evictions int64
}

// NewLRUCache 创建LRU缓存
func NewLRUCache(capacity int) *LRUCache {
    return &LRUCache{
        capacity: capacity,
        items:    make(map[string]*list.Element),
        lru:      list.New(),
        stats:    &CacheStats{},
    }
}

// Get 获取缓存
func (c *LRUCache) Get(ctx context.Context, key string, dest interface{}) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    elem, exists := c.items[key]
    if !exists {
        c.stats.Misses++
        return ErrCacheNotFound
    }

    ent := elem.Value.(*entry)

    // 检查是否过期
    if !ent.expiredAt.IsZero() && time.Now().After(ent.expiredAt) {
        c.removeElement(elem)
        c.stats.Misses++
        return ErrCacheExpired
    }

    // 移动到链表头部
    c.lru.MoveToFront(elem)

    // 设置目标值
    if dest != nil {
        // 使用反射或类型断言设置值
        if v, ok := dest.(*interface{}); ok {
            *v = ent.value
        }
    }

    c.stats.Hits++
    return nil
}

// Set 设置缓存
func (c *LRUCache) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    var expiredAt time.Time
    if ttl > 0 {
        expiredAt = time.Now().Add(ttl)
    }

    // 如果已存在，更新值
    if elem, exists := c.items[key]; exists {
        ent := elem.Value.(*entry)
        ent.value = value
        ent.expiredAt = expiredAt
        c.lru.MoveToFront(elem)
        return nil
    }

    // 检查容量，淘汰最久未使用的
    if c.lru.Len() >= c.capacity {
        c.evict()
    }

    // 添加新条目
    ent := &entry{
        key:       key,
        value:     value,
        expiredAt: expiredAt,
    }
    elem := c.lru.PushFront(ent)
    c.items[key] = elem

    return nil
}

// Delete 删除缓存
func (c *LRUCache) Delete(ctx context.Context, key string) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    if elem, exists := c.items[key]; exists {
        c.removeElement(elem)
    }

    return nil
}

// Exists 检查缓存是否存在
func (c *LRUCache) Exists(ctx context.Context, key string) (bool, error) {
    c.mu.RLock()
    defer c.mu.RUnlock()

    _, exists := c.items[key]
    return exists, nil
}

// Clear 清空缓存
func (c *LRUCache) Clear(ctx context.Context) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    c.items = make(map[string]*list.Element)
    c.lru.Init()
    return nil
}

// evict 淘汰最久未使用的条目
func (c *LRUCache) evict() {
    elem := c.lru.Back()
    if elem != nil {
        c.removeElement(elem)
        c.stats.Evictions++
    }
}

// removeElement 移除元素
func (c *LRUCache) removeElement(elem *list.Element) {
    ent := elem.Value.(*entry)
    delete(c.items, ent.key)
    c.lru.Remove(elem)
}

// GetStats 获取统计信息
func (c *LRUCache) GetStats() Stats {
    c.mu.RLock()
    defer c.mu.RUnlock()

    total := c.stats.Hits + c.stats.Misses
    hitRate := 0.0
    if total > 0 {
        hitRate = float64(c.stats.Hits) / float64(total)
    }

    return Stats{
        Hits:      c.stats.Hits,
        Misses:    c.stats.Misses,
        HitRate:   hitRate,
        Size:      int64(c.lru.Len()),
        Evictions: c.stats.Evictions,
    }
}

// ResetStats 重置统计信息
func (c *LRUCache) ResetStats() {
    c.mu.Lock()
    defer c.mu.Unlock()

    c.stats = &CacheStats{}
}
```

### 2. 多级缓存

**文件位置**: `internal/cache/local/multi_level.go`

```go
package local

import (
    "context"
    "fmt"
    "time"
)

// MultiLevelCache 多级缓存
// L1: 最热数据，小容量，无TTL
// L2: 热数据，中等容量，短TTL
// L3: 温数据，大容量，长TTL
type MultiLevelCache struct {
    l1 *LRUCache // L1缓存
    l2 *LRUCache // L2缓存
    l3 *LRUCache // L3缓存
}

// CacheLevel 缓存级别
type CacheLevel int

const (
    LevelL1 CacheLevel = iota
    LevelL2
    LevelL3
)

// NewMultiLevelCache 创建多级缓存
func NewMultiLevelCache(l1Size, l2Size, l3Size int) *MultiLevelCache {
    return &MultiLevelCache{
        l1: NewLRUCache(l1Size),  // 1000条
        l2: NewLRUCache(l2Size),  // 10000条
        l3: NewLRUCache(l3Size),  // 100000条
    }
}

// Get 获取缓存（从L1到L3查找）
func (m *MultiLevelCache) Get(ctx context.Context, key string, dest interface{}) error {
    // L1查找
    err := m.l1.Get(ctx, key, dest)
    if err == nil {
        return nil
    }

    // L2查找
    err = m.l2.Get(ctx, key, dest)
    if err == nil {
        // 提升到L1
        m.l1.Set(ctx, key, dest, 0)
        return nil
    }

    // L3查找
    err = m.l3.Get(ctx, key, dest)
    if err == nil {
        // 提升到L2
        m.l2.Set(ctx, key, dest, time.Minute*5)
        m.l1.Set(ctx, key, dest, 0)
        return nil
    }

    return ErrCacheNotFound
}

// Set 设置缓存（写入所有级别）
func (m *MultiLevelCache) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    // L1: 无TTL
    m.l1.Set(ctx, key, value, 0)

    // L2: 短TTL
    l2TTL := ttl
    if l2TTL == 0 {
        l2TTL = time.Minute * 5
    }
    m.l2.Set(ctx, key, value, l2TTL)

    // L3: 长TTL
    l3TTL := ttl
    if l3TTL == 0 {
        l3TTL = time.Hour
    }
    m.l3.Set(ctx, key, value, l3TTL)

    return nil
}

// Delete 删除缓存（从所有级别删除）
func (m *MultiLevelCache) Delete(ctx context.Context, key string) error {
    m.l1.Delete(ctx, key)
    m.l2.Delete(ctx, key)
    m.l3.Delete(ctx, key)
    return nil
}

// Clear 清空缓存
func (m *MultiLevelCache) Clear(ctx context.Context) error {
    m.l1.Clear(ctx)
    m.l2.Clear(ctx)
    m.l3.Clear(ctx)
    return nil
}

// GetStats 获取所有级别的统计信息
func (m *MultiLevelCache) GetStats() map[CacheLevel]Stats {
    return map[CacheLevel]Stats{
        LevelL1: m.l1.GetStats(),
        LevelL2: m.l2.GetStats(),
        LevelL3: m.l3.GetStats(),
    }
}

// Exists 检查缓存是否存在
func (m *MultiLevelCache) Exists(ctx context.Context, key string) (bool, error) {
    exists, _ := m.l1.Exists(ctx, key)
    if exists {
        return true, nil
    }

    exists, _ = m.l2.Exists(ctx, key)
    if exists {
        return true, nil
    }

    return m.l3.Exists(ctx, key)
}

// GetMultiple 批量获取
func (m *MultiLevelCache) GetMultiple(ctx context.Context, keys []string, dest map[string]interface{}) error {
    for _, key := range keys {
        var value interface{}
        err := m.Get(ctx, key, &value)
        if err == nil {
            dest[key] = value
        }
    }
    return nil
}

// SetMultiple 批量设置
func (m *MultiLevelCache) SetMultiple(ctx context.Context, items map[string]interface{}, ttl time.Duration) error {
    for key, value := range items {
        m.Set(ctx, key, value, ttl)
    }
    return nil
}

// DeleteMultiple 批量删除
func (m *MultiLevelCache) DeleteMultiple(ctx context.Context, keys []string) error {
    for _, key := range keys {
        m.Delete(ctx, key)
    }
    return nil
}
```

## Redis缓存实现

```go
package redis

import (
    "context"
    "encoding/json"
    "fmt"
    "time"

    "github.com/redis/go-redis/v9"
    "go.uber.org/zap"
)

// RedisCache Redis缓存实现
type RedisCache struct {
    client *redis.Client
    logger *zap.Logger
    stats  *CacheStats
}

// CacheStats 缓存统计
type CacheStats struct {
    Hits   int64
    Misses int64
}

// NewRedisCache 创建Redis缓存
func NewRedisCache(addr, password string, db int, logger *zap.Logger) *RedisCache {
    client := redis.NewClient(&redis.Options{
        Addr:     addr,
        Password: password,
        DB:       db,
        PoolSize: 100,
    })

    // 测试连接
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    if err := client.Ping(ctx).Err(); err != nil {
        logger.Fatal("Redis连接失败", zap.Error(err))
    }

    return &RedisCache{
        client: client,
        logger: logger,
        stats:  &CacheStats{},
    }
}

// Get 获取缓存
func (r *RedisCache) Get(ctx context.Context, key string, dest interface{}) error {
    val, err := r.client.Get(ctx, key).Result()
    if err != nil {
        if err == redis.Nil {
            r.stats.Misses++
            return ErrCacheNotFound
        }
        return fmt.Errorf("redis get error: %w", err)
    }

    r.stats.Hits++

    // 解析JSON
    if dest != nil {
        if err := json.Unmarshal([]byte(val), dest); err != nil {
            return fmt.Errorf("json unmarshal error: %w", err)
        }
    }

    return nil
}

// Set 设置缓存
func (r *RedisCache) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    // 序列化为JSON
    data, err := json.Marshal(value)
    if err != nil {
        return fmt.Errorf("json marshal error: %w", err)
    }

    // 设置缓存
    return r.client.Set(ctx, key, data, ttl).Err()
}

// Delete 删除缓存
func (r *RedisCache) Delete(ctx context.Context, key string) error {
    return r.client.Del(ctx, key).Err()
}

// Exists 检查缓存是否存在
func (r *RedisCache) Exists(ctx context.Context, key string) (bool, error) {
    n, err := r.client.Exists(ctx, key).Result()
    return n > 0, err
}

// Clear 清空缓存（慎用）
func (r *RedisCache) Clear(ctx context.Context) error {
    return r.client.FlushDB(ctx).Err()
}

// GetMultiple 批量获取（MGET）
func (r *RedisCache) GetMultiple(ctx context.Context, keys []string, dest map[string]interface{}) error {
    vals, err := r.client.MGet(ctx, keys...).Result()
    if err != nil {
        return err
    }

    for i, val := range vals {
        if val != nil {
            var value interface{}
            if err := json.Unmarshal([]byte(val.(string)), &value); err == nil {
                dest[keys[i]] = value
            }
        }
    }

    return nil
}

// SetMultiple 批量设置（MSET）
func (r *RedisCache) SetMultiple(ctx context.Context, items map[string]interface{}, ttl time.Duration) error {
    pipe := r.client.Pipeline()

    for key, value := range items {
        data, err := json.Marshal(value)
        if err != nil {
            continue
        }
        pipe.Set(ctx, key, data, ttl)
    }

    _, err := pipe.Exec(ctx)
    return err
}

// DeleteMultiple 批量删除
func (r *RedisCache) DeleteMultiple(ctx context.Context, keys []string) error {
    return r.client.Del(ctx, keys...).Err()
}

// GetStats 获取统计信息
func (r *RedisCache) GetStats() Stats {
    total := r.stats.Hits + r.stats.Misses
    hitRate := 0.0
    if total > 0 {
        hitRate = float64(r.stats.Hits) / float64(total)
    }

    // 获取Redis信息
    ctx := context.Background()
    info, _ := r.client.Info(ctx, "memory").Result()

    return Stats{
        Hits:   r.stats.Hits,
        Misses: r.stats.Misses,
        HitRate: hitRate,
    }
}

// ResetStats 重置统计信息
func (r *RedisCache) ResetStats() {
    r.stats = &CacheStats{}
}

// Incr 自增
func (r *RedisCache) Incr(ctx context.Context, key string) (int64, error) {
    return r.client.Incr(ctx, key).Result()
}

// Decr 自减
func (r *RedisCache) Decr(ctx context.Context, key string) (int64, error) {
    return r.client.Decr(ctx, key).Result()
}

// Expire 设置过期时间
func (r *RedisCache) Expire(ctx context.Context, key string, ttl time.Duration) error {
    return r.client.Expire(ctx, key, ttl).Err()
}

// TTL 获取剩余过期时间
func (r *RedisCache) TTL(ctx context.Context, key string) (time.Duration, error) {
    return r.client.TTL(ctx, key).Result()
}
```

## 缓存策略实现

### 1. Cache-Aside模式

**文件位置**: `internal/cache/strategy/cache_aside.go`

```go
package strategy

import (
    "context"
    "fmt"
    "time"
)

// CacheAside Cache-Aside模式
// 应用程序负责管理缓存，缓存缺失时从数据库加载
type CacheAside struct {
    cache Cache
    db    DataLoader
}

// DataLoader 数据加载器接口
type DataLoader interface {
    Load(ctx context.Context, key string) (interface{}, error)
}

// NewCacheAside 创建Cache-Aside策略
func NewCacheAside(cache Cache, db DataLoader) *CacheAside {
    return &CacheAside{
        cache: cache,
        db:    db,
    }
}

// Get 获取数据（缓存优先）
func (ca *CacheAside) Get(ctx context.Context, key string, dest interface{}) error {
    // 1. 尝试从缓存获取
    err := ca.cache.Get(ctx, key, dest)
    if err == nil {
        return nil // 缓存命中
    }

    // 2. 缓存未命中，从数据库加载
    data, err := ca.db.Load(ctx, key)
    if err != nil {
        return fmt.Errorf("db load error: %w", err)
    }

    // 3. 写入缓存（异步）
    go func() {
        ca.cache.Set(context.Background(), key, data, time.Minute*5)
    }()

    // 4. 设置返回值
    if dest != nil {
        // 使用反射或类型断言设置值
    }

    return nil
}

// Set 设置数据（只更新数据库，删除缓存）
func (ca *CacheAside) Set(ctx context.Context, key string, value interface{}) error {
    // 1. 更新数据库
    err := ca.db.Save(ctx, key, value)
    if err != nil {
        return err
    }

    // 2. 删除缓存（而非更新）
    go func() {
        ca.cache.Delete(context.Background(), key)
    }()

    return nil
}

// Delete 删除数据
func (ca *CacheAside) Delete(ctx context.Context, key string) error {
    // 1. 删除数据库
    err := ca.db.Delete(ctx, key)
    if err != nil {
        return err
    }

    // 2. 删除缓存
    go func() {
        ca.cache.Delete(context.Background(), key)
    }()

    return nil
}
```

```go
package strategy

// WriteThrough Write-Through模式
// 写入同时更新缓存和数据库，保证一致性
type WriteThrough struct {
    cache Cache
    db    DataLoader
}

// NewWriteThrough 创建Write-Through策略
func NewWriteThrough(cache Cache, db DataLoader) *WriteThrough {
    return &WriteThrough{
        cache: cache,
        db:    db,
    }
}

// Set 设置数据（同时写缓存和数据库）
func (wt *WriteThrough) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    // 1. 写入缓存
    if err := wt.cache.Set(ctx, key, value, ttl); err != nil {
        return fmt.Errorf("cache set error: %w", err)
    }

    // 2. 写入数据库
    if err := wt.db.Save(ctx, key, value); err != nil {
        // 数据库写入失败，回滚缓存
        wt.cache.Delete(ctx, key)
        return fmt.Errorf("db save error: %w", err)
    }

    return nil
}

// Get 获取数据（缓存优先）
func (wt *WriteThrough) Get(ctx context.Context, key string, dest interface{}) error {
    // 从缓存获取
    err := wt.cache.Get(ctx, key, dest)
    if err == nil {
        return nil
    }

    // 从数据库加载
    data, err := wt.db.Load(ctx, key)
    if err != nil {
        return err
    }

    // 回写缓存
    go func() {
        wt.cache.Set(context.Background(), key, data, time.Minute*5)
    }()

    return nil
}
```

### 3. Write-Behind模式

**文件位置**: `internal/cache/strategy/write_behind.go`

```go
package strategy

import (
    "context"
    "fmt"
    "sync"
    "time"
)

// WriteBehind Write-Behind模式
// 先写缓存，异步批量写数据库
type WriteBehind struct {
    cache     Cache
    db        DataLoader
    buffer    chan *WriteTask
    wg        sync.WaitGroup
    stopCh    chan struct{}
    batchSize int
    flushTime time.Duration
}

// WriteTask 写任务
type WriteTask struct {
    Key   string
    Value interface{}
}

// NewWriteBehind 创建Write-Behind策略
func NewWriteBehind(cache Cache, db DataLoader, batchSize int, flushTime time.Duration) *WriteBehind {
    wb := &WriteBehind{
        cache:     cache,
        db:        db,
        buffer:    make(chan *WriteTask, batchSize*2),
        stopCh:    make(chan struct{}),
        batchSize: batchSize,
        flushTime: flushTime,
    }

    // 启动异步写入协程
    wb.wg.Add(1)
    go wb.flusher()

    return wb
}

// Set 设置数据（只写缓存）
func (wb *WriteBehind) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    // 1. 写入缓存
    if err := wb.cache.Set(ctx, key, value, ttl); err != nil {
        return err
    }

    // 2. 加入写缓冲队列
    select {
    case wb.buffer <- &WriteTask{Key: key, Value: value}:
        return nil
    default:
        // 缓冲区满，同步写入
        return wb.db.Save(ctx, key, value)
    }
}

// Get 获取数据
func (wb *WriteBehind) Get(ctx context.Context, key string, dest interface{}) error {
    // 从缓存获取
    err := wb.cache.Get(ctx, key, dest)
    if err == nil {
        return nil
    }

    // 从数据库加载
    data, err := wb.db.Load(ctx, key)
    if err != nil {
        return err
    }

    // 回写缓存
    go wb.cache.Set(context.Background(), key, data, time.Minute*5)

    return nil
}

// flusher 异步刷新器
func (wb *WriteBehind) flusher() {
    defer wb.wg.Done()

    ticker := time.NewTicker(wb.flushTime)
    defer ticker.Stop()

    batch := make([]*WriteTask, 0, wb.batchSize)

    for {
        select {
        case task := <-wb.buffer:
            batch = append(batch, task)
            if len(batch) >= wb.batchSize {
                wb.flush(batch)
                batch = batch[:0]
            }

        case <-ticker.C:
            if len(batch) > 0 {
                wb.flush(batch)
                batch = batch[:0]
            }

        case <-wb.stopCh:
            if len(batch) > 0 {
                wb.flush(batch)
            }
            return
        }
    }
}

// flush 批量刷新到数据库
func (wb *WriteBehind) flush(tasks []*WriteTask) {
    ctx := context.Background()
    for _, task := range tasks {
        if err := wb.db.Save(ctx, task.Key, task.Value); err != nil {
            // 写入失败，记录日志
            // 可以选择重试或写入死信队列
        }
    }
}

// Close 关闭写后处理器
func (wb *WriteBehind) Close() error {
    close(wb.stopCh)
    wb.wg.Wait()
    return nil
}
```

```go
    "github.com/redis/go-redis/v9"

// CacheMiddleware 缓存中间件
type CacheMiddleware struct {
    cache  Cache
    logger *zap.Logger
    ttl    time.Duration
}

// NewCacheMiddleware 创建缓存中间件
func NewCacheMiddleware(cache Cache, logger *zap.Logger, ttl time.Duration) *CacheMiddleware {
    return &CacheMiddleware{
        cache:  cache,
        logger: logger,
        ttl:    ttl,
    }
}

// Middleware 返回缓存中间件函数
func (m *CacheMiddleware) Middleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        // 只缓存GET请求
        if c.Request.Method != "GET" {
            c.Next()
            return
        }

        // 生成缓存键
        key := m.generateKey(c)

        // 尝试从缓存获取
        var cachedResponse CachedResponse
        err := m.cache.Get(c.Request.Context(), key, &cachedResponse)
        if err == nil {
            // 缓存命中
            for k, v := range cachedResponse.Headers {
                c.Header(k, v)
            }
            c.Data(cachedResponse.StatusCode, cachedResponse.ContentType, cachedResponse.Body)
            c.Abort()
            return
        }

        // 缓存未命中，继续处理请求
        // 使用响应包装器
        writer := &responseWriter{
            ResponseWriter: c.Writer,
            buffer:        &bytes.Buffer{},
        }
        c.Writer = writer

        c.Next()

        // 只缓存成功响应
        if c.Writer.Status() >= 200 && c.Writer.Status() < 300 {
            cachedResponse := CachedResponse{
                StatusCode:  c.Writer.Status(),
                ContentType: c.Writer.Header().Get("Content-Type"),
                Headers:     make(map[string]string),
                Body:        writer.buffer.Bytes(),
            }

            // 复制响应头
            for k, v := range c.Writer.Header() {
                if len(v) > 0 {
                    cachedResponse.Headers[k] = v[0]
                }
            }

            // 异步写入缓存
            go func() {
                ctx := context.Background()
                if err := m.cache.Set(ctx, key, cachedResponse, m.ttl); err != nil {
                    m.logger.Error("缓存写入失败",
                        zap.String("key", key),
                        zap.Error(err),
                    )
                }
            }()
        }
    }
}

// generateKey 生成缓存键
func (m *CacheMiddleware) generateKey(c *gin.Context) string {
    // 格式: cache:{method}:{path}:{query}
    path := c.Request.URL.Path
    query := c.Request.URL.RawQuery

    key := fmt.Sprintf("cache:%s:%s", c.Request.Method, path)
    if query != "" {
        key += ":" + query
    }

    return key
}

// CachedResponse 缓存响应
type CachedResponse struct {
    StatusCode  int              `json:"status_code"`
    ContentType string           `json:"content_type"`
    Headers     map[string]string `json:"headers"`
    Body        []byte            `json:"body"`
}

// responseWriter 响应包装器
type responseWriter struct {
    gin.ResponseWriter
    buffer *bytes.Buffer
}

func (w *responseWriter) Write(data []byte) (int, error) {
    w.buffer.Write(data)
    return w.ResponseWriter.Write(data)
}

func (w *responseWriter) WriteString(s string) (int, error) {
    w.buffer.WriteString(s)
    return w.ResponseWriter.WriteString(s)
}
```

```go
    InvalidationPattern                        // 模式匹配

// CacheInvalidation 缓存失效管理器
type CacheInvalidation struct {

}

// NewCacheInvalidation 创建缓存失效管理器
func NewCacheInvalidation(cache Cache, logger *zap.Logger) *CacheInvalidation {

        logger: logger,
    }
}

// Invalidate 失效缓存
func (ci *CacheInvalidation) Invalidate(ctx context.Context, key string) error {
    return ci.cache.Delete(ctx, key)
}

// InvalidateByPrefix 按前缀失效缓存
func (ci *CacheInvalidation) InvalidateByPrefix(ctx context.Context, prefix string) error {
    // 如果是Redis，使用SCAN命令
    if redisCache, ok := ci.cache.(*redis.RedisCache); ok {
        // 使用SCAN + DEL删除匹配的键
        iter := redisCache.Client().Scan(ctx, 0, prefix+"*", 0).Iterator()
        keys := []string{}

        for iter.Next(ctx) {
            keys = append(keys, iter.Val())
        }

        if len(keys) > 0 {
            return redisCache.DeleteMultiple(ctx, keys)
        }
    }

    return nil
}

// InvalidateByPattern 按模式失效缓存
func (ci *CacheInvalidation) InvalidateByPattern(ctx context.Context, pattern string) error {
    return ci.InvalidateByPrefix(ctx, pattern)
}

// Middleware 缓存失效中间件
func (ci *CacheInvalidation) Middleware(invalidationType InvalidationType) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 保存原始响应
        c.Next()

        // 只在成功响应时失效缓存
        if c.Writer.Status() >= 200 && c.Writer.Status() < 300 {
            switch c.Request.Method {
            case "POST", "PUT", "PATCH", "DELETE":
                // 根据请求路径确定失效的缓存
                keys := ci.getInvalidationKeys(c)
                for _, key := range keys {
                    go func(k string) {
                        ctx := context.Background()
                        if err := ci.cache.Delete(ctx, k); err != nil {
                            ci.logger.Error("缓存失效失败",
                                zap.String("key", k),
                                zap.Error(err),
                            )
                        }
                    }(key)
                }
            }
        }
    }
}

// getInvalidationKeys 获取需要失效的缓存键
func (ci *CacheInvalidation) getInvalidationKeys(c *gin.Context) []string {
    path := c.FullPath()
    keys := []string{}

    // 会议相关操作
    if strings.Contains(path, "/conferences") {
        // 失效会议列表缓存
        keys = append(keys, "cache:conference:active")
        keys = append(keys, "cache:conference:list:*")
    }

    // 用户相关操作
    if strings.Contains(path, "/users") {
        // 失效用户列表缓存
        keys = append(keys, "cache:user:list:*")
    }

    // 录制相关操作
    if strings.Contains(path, "/recordings") {
        // 失效录制统计缓存
        keys = append(keys, "cache:stats:recording")
    }

    return keys
}
```
