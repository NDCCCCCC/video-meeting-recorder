# 服务层设计

## 一、服务层概述

### 1.1 设计原则

服务层遵循以下设计原则：

1. **统一接口** - 所有服务实现`Service`接口
2. **依赖注入** - 使用依赖注入管理服务依赖
3. **单一职责** - 每个服务只负责一个领域
4. **接口隔离** - 使用接口隔离服务依赖
5. **生命周期管理** - 统一的初始化、启动、停止流程

### 1.2 统一服务接口

```go
// internal/common/interfaces.go
type Service interface {
    // Initialize 初始化服务
    Initialize() error

    // Start 启动服务
    Start() error

    // Stop 停止服务
    Stop() error

    // Restart 重启服务
    Restart() error

    // HealthCheck 健康检查
    HealthCheck() error

    // GetName 获取服务名称
    GetName() string

    // GetType 获取服务类型
    GetType() string

    // GetStatus 获取服务状态
    GetStatus() ServiceStatus

    // GetContext 获取上下文
    GetContext() context.Context

    // SetContext 设置上下文
    SetContext(ctx context.Context)
}

// ServiceStatus 服务状态
type ServiceStatus string

const (
    ServiceStatusStopped   ServiceStatus = "stopped"
    ServiceStatusStarting  ServiceStatus = "starting"
    ServiceStatusRunning   ServiceStatus = "running"
    ServiceStatusStopping  ServiceStatus = "stopping"
    ServiceStatusError     ServiceStatus = "error"
)
```

### 1.3 基础服务实现

```go
// internal/common/base_service.go
type BaseService struct {
    name    string
    sType   string
    status  ServiceStatus
    ctx     context.Context
    cancel  context.CancelFunc
    logger  *zap.Logger
    config  *config.Config
}

func (s *BaseService) Initialize() error {
    s.status = ServiceStatusStopped
    return nil
}

func (s *BaseService) Start() error {
    s.status = ServiceStatusStarting

    // 创建上下文
    s.ctx, s.cancel = context.WithCancel(context.Background())

    s.status = ServiceStatusRunning
    return nil
}

func (s *BaseService) Stop() error {
    s.status = ServiceStatusStopping

    // 取消上下文
    if s.cancel != nil {
        s.cancel()
    }

    s.status = ServiceStatusStopped
    return nil
}

func (s *BaseService) Restart() error {
    if err := s.Stop(); err != nil {
        return err
    }
    return s.Start()
}

func (s *BaseService) HealthCheck() error {
    if s.status != ServiceStatusRunning {
        return fmt.Errorf("服务未运行: %s", s.status)
    }
    return nil
}

func (s *BaseService) GetName() string {
    return s.name
}

func (s *BaseService) GetType() string {
    return s.sType
}

func (s *BaseService) GetStatus() ServiceStatus {
    return s.status
}

func (s *BaseService) GetContext() context.Context {
    return s.ctx
}

func (s *BaseService) SetContext(ctx context.Context) {
    s.ctx = ctx
}
```

### 2.1 VideoRecordingTaskService

```go
// internal/services/video_recording/video_recording_service.go
type VideoRecordingTaskService struct {
    common.BaseService
    db               *gorm.DB
    logger           *zap.Logger
    config           *config.Config
    scheduler        *VideoSimpleScheduler
    huaweiConnector  *HuaweiConferenceConnector
    coordinator      *SimpleRecordingCoordinator
}

// NewVideoRecordingTaskService 创建服务
func NewVideoRecordingTaskService(
    db *gorm.DB,
    logger *zap.Logger,
    config *config.Config,
) *VideoRecordingTaskService {
    return &VideoRecordingTaskService{
        BaseService: common.BaseService{
            name:   "video_recording_task_service",
            sType:  "task_service",
            logger: logger,
            config: config,
        },
        db:     db,
        logger: logger,
        config: config,
    }
}

// Initialize 初始化服务
func (s *VideoRecordingTaskService) Initialize() error {
    if err := s.BaseService.Initialize(); err != nil {
        return err
    }

    // 创建子组件
    s.huaweiConnector = NewHuaweiConferenceConnector(s.db, s.logger, s.config)
    s.coordinator = NewSimpleRecordingCoordinator(s.logger, s.config)
    s.scheduler = NewVideoSimpleScheduler(s, s.huaweiConnector, s.coordinator, s.logger, s.config)

    // 初始化子组件
    if err := s.huaweiConnector.Initialize(); err != nil {
        return fmt.Errorf("初始化华为连接器失败: %w", err)
    }

    if err := s.coordinator.Initialize(); err != nil {
        return fmt.Errorf("初始化录制协调器失败: %w", err)
    }

    if err := s.scheduler.Initialize(); err != nil {
        return fmt.Errorf("初始化调度器失败: %w", err)
    }

    return nil
}

// Start 启动服务
func (s *VideoRecordingTaskService) Start() error {
    if err := s.BaseService.Start(); err != nil {
        return err
    }

    // 启动子组件
    if err := s.huaweiConnector.Start(); err != nil {
        return fmt.Errorf("启动华为连接器失败: %w", err)
    }

    if err := s.coordinator.Start(); err != nil {
        return fmt.Errorf("启动录制协调器失败: %w", err)
    }

    if err := s.scheduler.Start(); err != nil {
        return fmt.Errorf("启动调度器失败: %w", err)
    }

    s.logger.Info("视频录制任务服务启动成功")
    return nil
}

// Stop 停止服务
func (s *VideoRecordingTaskService) Stop() error {
    s.status = ServiceStatusStopping

    // 停止子组件(逆序)
    if err := s.scheduler.Stop(); err != nil {
        s.logger.Error("停止调度器失败", zap.Error(err))
    }

    if err := s.coordinator.Stop(); err != nil {
        s.logger.Error("停止录制协调器失败", zap.Error(err))
    }

    if err := s.huaweiConnector.Stop(); err != nil {
        s.logger.Error("停止华为连接器失败", zap.Error(err))
    }

    return s.BaseService.Stop()
}

// CreateTask 创建任务
func (s *VideoRecordingTaskService) CreateTask(req *CreateTaskRequest, userID uint) (*models.VideoRecordingTask, error) {
    // 1. 解析时间
    startTime, err := time.Parse(time.RFC3339, req.StartTime)
    if err != nil {
        return nil, fmt.Errorf("解析开始时间失败: %w", err)
    }

    endTime, err := time.Parse(time.RFC3339, req.EndTime)
    if err != nil {
        return nil, fmt.Errorf("解析结束时间失败: %w", err)
    }

    // 2. 创建任务实体
    task := &models.VideoRecordingTask{
        Name:               req.Name,
        Description:        req.Description,
        StartTime:          startTime,
        EndTime:            endTime,
        PreJoinMinutes:     req.PreJoinMinutes,
        RecordDelayMinutes: req.RecordDelayMinutes,
        ConferenceNumber:   req.ConferenceNumber,
        HuaweiConfigID:     req.HuaweiConfigID,
        Status:             models.VideoStatusPending,
        CreatedBy:          userID,
    }

    // 3. 验证任务
    if err := task.IsValid(); err != nil {
        return nil, err
    }

    // 4. 保存到数据库
    if err := s.db.Create(task).Error; err != nil {
        return nil, fmt.Errorf("保存任务失败: %w", err)
    }

    // 5. 添加到调度器
    if err := s.scheduler.AddTask(task); err != nil {
        return nil, fmt.Errorf("添加到调度器失败: %w", err)
    }

    s.logger.Info("创建录制任务成功",
        zap.Uint("task_id", task.ID),
        zap.String("name", task.Name),
    )

    return task, nil
}

// GetTask 获取任务
func (s *VideoRecordingTaskService) GetTask(taskID uint) (*models.VideoRecordingTask, error) {
    var task models.VideoRecordingTask
    err := s.db.
        Preload("HuaweiConfig").
        Preload("Creator").
        Preload("ConferenceRecord").
        First(&task, taskID).Error

    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, fmt.Errorf("任务不存在: %d", taskID)
        }
        return nil, err
    }

    return &task, nil
}

// UpdateTask 更新任务
func (s *VideoRecordingTaskService) UpdateTask(taskID uint, req *UpdateTaskRequest) error {
    var task models.VideoRecordingTask
    if err := s.db.First(&task, taskID).Error; err != nil {
        return err
    }

    // 检查状态
    if task.Status != models.VideoStatusPending {
        return errors.New("只能修改待执行状态的任务")
    }

    // 更新字段
    if req.Name != "" {
        task.Name = req.Name
    }
    if req.Description != "" {
        task.Description = req.Description
    }
    if req.StartTime != "" {
        startTime, _ := time.Parse(time.RFC3339, req.StartTime)
        task.StartTime = startTime
    }
    if req.EndTime != "" {
        endTime, _ := time.Parse(time.RFC3339, req.EndTime)
        task.EndTime = endTime
    }

    // 保存
    return s.db.Save(&task).Error
}

// DeleteTask 删除任务
func (s *VideoRecordingTaskService) DeleteTask(taskID uint) error {
    var task models.VideoRecordingTask
    if err := s.db.First(&task, taskID).Error; err != nil {
        return err
    }

    // 检查状态
    if task.Status == models.VideoStatusRecording || task.Status == models.VideoStatusConnecting {
        return errors.New("无法删除正在执行的任务")
    }

    // 从调度器移除
    s.scheduler.RemoveTask(taskID)

    // 删除任务
    return s.db.Delete(&task).Error
}

// UpdateTaskStatus 更新任务状态
func (s *VideoRecordingTaskService) UpdateTaskStatus(taskID uint, status models.VideoRecordingTaskStatus, errorMsg string) error {
    updates := map[string]interface{}{
        "status": status,
    }

    if errorMsg != "" {
        updates["error_msg"] = errorMsg
    }

    return s.db.Model(&models.VideoRecordingTask{}).
        Where("id = ?", taskID).
        Updates(updates).Error
}

// GetTasksByStatus 按状态获取任务
func (s *VideoRecordingTaskService) GetTasksByStatus(status models.VideoRecordingTaskStatus) ([]models.VideoRecordingTask, error) {
    var tasks []models.VideoRecordingTask
    err := s.db.
        Preload("HuaweiConfig").
        Preload("Creator").
        Where("status = ?", status).
        Order("start_time ASC").
        Find(&tasks).Error

    return tasks, err
}

// GetPendingTasks 获取待执行任务
func (s *VideoRecordingTaskService) GetPendingTasks() ([]models.VideoRecordingTask, error) {
    return s.GetTasksByStatus(models.VideoStatusPending)
}
```

### 2.3 UserService

```go
// internal/services/core/user_service.go
type UserService struct {
    common.BaseService
    db     *gorm.DB
    logger *zap.Logger
}

// CreateUser 创建用户
func (s *UserService) CreateUser(req *CreateUserRequest) (*models.User, error) {
    // 检查用户名是否存在
    var count int64
    s.db.Model(&models.User{}).Where("username = ?", req.Username).Count(&count)
    if count > 0 {
        return nil, errors.New("用户名已存在")
    }

    // 创建用户
    user := &models.User{
        Username: req.Username,
        Email:    req.Email,
        FullName: req.FullName,
        RoleID:   req.RoleID,
        IsActive: true,
    }

    // 设置密码
    if err := user.SetPassword(req.Password); err != nil {
        return nil, err
    }

    if err := s.db.Create(user).Error; err != nil {
        return nil, err
    }

    return user, nil
}

// GetUser 获取用户
func (s *UserService) GetUser(userID uint) (*models.User, error) {
    var user models.User
    err := s.db.
        Preload("Role").
        Preload("Role.Permissions").
        First(&user, userID).Error

    return &user, err
}

// Authenticate 认证用户
func (s *UserService) Authenticate(username, password string) (*models.User, error) {
    var user models.User
    err := s.db.
        Preload("Role").
        Preload("Role.Permissions").
        Where("username = ?", username).
        First(&user).Error

    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, errors.New("用户名或密码错误")
        }
        return nil, err
    }

    // 检查密码
    if !user.CheckPassword(password) {
        return nil, errors.New("用户名或密码错误")
    }

    // 检查状态
    if !user.IsActive {
        return nil, errors.New("用户已被禁用")
    }

    return &user, nil
}
```

## 三、服务容器

### 3.1 服务容器设计

```go
// internal/common/service_container.go
type ServiceContainer struct {
    services map[string]common.Service
    mu       sync.RWMutex
    logger   *zap.Logger
}

// NewServiceContainer 创建服务容器
func NewServiceContainer(logger *zap.Logger) *ServiceContainer {
    return &ServiceContainer{
        services: make(map[string]common.Service),
        logger:   logger,
    }
}

// Register 注册服务
func (c *ServiceContainer) Register(service common.Service) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    name := service.GetName()

    if _, exists := c.services[name]; exists {
        return fmt.Errorf("服务已存在: %s", name)
    }

    c.services[name] = service

    c.logger.Info("服务已注册",
        zap.String("name", name),
        zap.String("type", service.GetType()),
    )

    return nil
}

// Get 获取服务
func (c *ServiceContainer) Get(name string) (common.Service, error) {
    c.mu.RLock()
    defer c.mu.RUnlock()

    service, ok := c.services[name]
    if !ok {
        return nil, fmt.Errorf("服务不存在: %s", name)
    }

    return service, nil
}

// GetByType 按类型获取服务
func (c *ServiceContainer) GetByType(sType string) ([]common.Service, error) {
    c.mu.RLock()
    defer c.mu.RUnlock()

    var services []common.Service
    for _, service := range c.services {
        if service.GetType() == sType {
            services = append(services, service)
        }
    }

    return services, nil
}

// InitializeAll 初始化所有服务
func (c *ServiceContainer) InitializeAll() error {
    c.mu.RLock()
    defer c.mu.RUnlock()

    for name, service := range c.services {
        if err := service.Initialize(); err != nil {
            return fmt.Errorf("初始化服务 %s 失败: %w", name, err)
        }
    }

    return nil
}

// StartAll 启动所有服务
func (c *ServiceContainer) StartAll() error {
    c.mu.RLock()
    defer c.mu.RUnlock()

    for name, service := range c.services {
        if err := service.Start(); err != nil {
            return fmt.Errorf("启动服务 %s 失败: %w", name, err)
        }
    }

    return nil
}

// StopAll 停止所有服务
func (c *ServiceContainer) StopAll() error {
    c.mu.RLock()
    defer c.mu.RUnlock()

    var errs []error

    // 逆序停止
    for _, service := range c.services {
        if err := service.Stop(); err != nil {
            errs = append(errs, err)
        }
    }

    if len(errs) > 0 {
        return fmt.Errorf("停止服务时发生错误: %v", errs)
    }

    return nil
}

// HealthCheckAll 检查所有服务健康状态
func (c *ServiceContainer) HealthCheckAll() map[string]error {
    c.mu.RLock()
    defer c.mu.RUnlock()

    results := make(map[string]error)

    for name, service := range c.services {
        results[name] = service.HealthCheck()
    }

    return results
}
```

```go
    "github.com/google/wire"

}
```
