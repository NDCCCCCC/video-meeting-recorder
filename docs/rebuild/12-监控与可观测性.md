## 二、Metrics指标系统

### 2.1 Prometheus集成

```go
// internal/monitoring/metrics.go
package monitoring

import (
    "github.com/prometheus/client_golang/prometheus"
    "github.com/prometheus/client_golang/prometheus/promauto"
)

// Metrics 指标收集器
type Metrics struct {
    // HTTP请求指标
    HTTPRequestsTotal    *prometheus.CounterVec
    HTTPRequestDuration  *prometheus.HistogramVec
    HTTPRequestSize      *prometheus.HistogramVec
    HTTPResponseSize     *prometheus.HistogramVec

    // 任务指标
    TasksTotal           *prometheus.CounterVec
    TasksActive          prometheus.Gauge
    TasksCompleted       *prometheus.CounterVec
    TasksFailed          *prometheus.CounterVec
    TaskDuration         *prometheus.HistogramVec

    // 录制指标
    RecordingsActive     prometheus.Gauge
    RecordingsTotal      *prometheus.CounterVec
    RecordingDuration    *prometheus.HistogramVec
    RecordingSize        *prometheus.HistogramVec

    // FFmpeg指标
    FFmpegProcesses      prometheus.Gauge
    FFmpegErrors         *prometheus.CounterVec

    // 系统指标
    DiskUsage            prometheus.Gauge
    DiskAvailable        prometheus.Gauge
    MemoryUsage          prometheus.Gauge

    // Huawei指标
    HuaweiAPIRequests    *prometheus.CounterVec
    HuaweiAPIErrors      *prometheus.CounterVec
    HuaweiConnections    prometheus.Gauge
}

// NewMetrics 创建指标收集器
func NewMetrics() *Metrics {
    return &Metrics{
        // HTTP请求指标
        HTTPRequestsTotal: promauto.NewCounterVec(
            prometheus.CounterOpts{
                Name: "conference_http_requests_total",
                Help: "Total number of HTTP requests",
            },
            []string{"method", "endpoint", "status"},
        ),

        HTTPRequestDuration: promauto.NewHistogramVec(
            prometheus.HistogramOpts{
                Name:    "conference_http_request_duration_seconds",
                Help:    "HTTP request duration in seconds",
                Buckets: prometheus.DefBuckets,
            },
            []string{"method", "endpoint"},
        ),

        // 任务指标
        TasksTotal: promauto.NewCounterVec(
            prometheus.CounterOpts{
                Name: "conference_tasks_total",
                Help: "Total number of tasks",
            },
            []string{"type", "status"},
        ),

        TasksActive: promauto.NewGauge(
            prometheus.GaugeOpts{
                Name: "conference_tasks_active",
                Help: "Number of active tasks",
            },
        ),

        TaskDuration: promauto.NewHistogramVec(
            prometheus.HistogramOpts{
                Name:    "conference_task_duration_seconds",
                Help:    "Task execution duration in seconds",
                Buckets: prometheus.ExponentialBuckets(60, 2, 10), // 1min, 2min, 4min, ...
            },
            []string{"type"},
        ),

        // 录制指标
        RecordingsActive: promauto.NewGauge(
            prometheus.GaugeOpts{
                Name: "conference_recordings_active",
                Help: "Number of active recordings",
            },
        ),

        RecordingDuration: promauto.NewHistogramVec(
            prometheus.HistogramOpts{
                Name:    "conference_recording_duration_seconds",
                Help:    "Recording duration in seconds",
                Buckets: prometheus.ExponentialBuckets(300, 2, 8), // 5min, 10min, 20min, ...
            },
            []string{"format"},
        ),

        // FFmpeg指标
        FFmpegProcesses: promauto.NewGauge(
            prometheus.GaugeOpts{
                Name: "conference_ffmpeg_processes",
                Help: "Number of FFmpeg processes",
            },
        ),

        FFmpegErrors: promauto.NewCounterVec(
            prometheus.CounterOpts{
                Name: "conference_ffmpeg_errors_total",
                Help: "Total number of FFmpeg errors",
            },
            []string{"error_type"},
        ),

        // 系统指标
        DiskUsage: promauto.NewGauge(
            prometheus.GaugeOpts{
                Name: "conference_disk_usage_bytes",
                Help: "Disk usage in bytes",
            },
        ),

        MemoryUsage: promauto.NewGauge(
            prometheus.GaugeOpts{
                Name: "conference_memory_usage_bytes",
                Help: "Memory usage in bytes",
            },
        ),
    }
}

// RecordHTTPRequest 记录HTTP请求
func (m *Metrics) RecordHTTPRequest(method, endpoint string, status int, duration float64) {
    m.HTTPRequestsTotal.WithLabelValues(method, endpoint, fmt.Sprintf("%d", status)).Inc()
    m.HTTPRequestDuration.WithLabelValues(method, endpoint).Observe(duration)
}

// RecordTaskCreated 记录任务创建
func (m *Metrics) RecordTaskCreated(taskType string) {
    m.TasksTotal.WithLabelValues(taskType, "created").Inc()
}

// RecordTaskCompleted 记录任务完成
func (m *Metrics) RecordTaskCompleted(taskType string, duration float64, success bool) {
    if success {
        m.TasksTotal.WithLabelValues(taskType, "completed").Inc()
        m.TasksCompleted.WithLabelValues(taskType).Inc()
    } else {
        m.TasksTotal.WithLabelValues(taskType, "failed").Inc()
        m.TasksFailed.WithLabelValues(taskType).Inc()
    }
    m.TaskDuration.WithLabelValues(taskType).Observe(duration)
}

// UpdateActiveTasks 更新活跃任务数
func (m *Metrics) UpdateActiveTasks(count int) {
    m.TasksActive.Set(float64(count))
}
```

### 2.2 中间件集成

```go
// internal/middleware/metrics.go
package middleware

import (
    "record_go/internal/monitoring"
    "time"
    "github.com/gin-gonic/gin"
)

// MetricsMiddleware 指标收集中间件
func MetricsMiddleware(metrics *monitoring.Metrics) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()

        // 处理请求
        c.Next()

        // 记录指标
        duration := time.Since(start).Seconds()
        status := c.Writer.Status()
        method := c.Request.Method
        endpoint := c.FullPath()

        if endpoint == "" {
            endpoint = c.Request.URL.Path
        }

        metrics.RecordHTTPRequest(method, endpoint, status, duration)
    }
}
```

## 四、健康检查

### 4.1 健康检查端点

```go
// internal/monitoring/health.go
package monitoring

import (
    "net/http"
    "record_go/internal/common"
)

// HealthChecker 健康检查器
type HealthChecker struct {
    services map[string]common.Service
    db       *gorm.DB
}

// NewHealthChecker 创建健康检查器
func NewHealthChecker(db *gorm.DB, services map[string]common.Service) *HealthChecker {
    return &HealthChecker{
        services: services,
        db:       db,
    }
}

// Check 健康检查
func (h *HealthChecker) Check() *HealthStatus {
    status := &HealthStatus{
        Status:    "healthy",
        Timestamp: time.Now(),
        Services:  make(map[string]ServiceHealth),
    }

    // 1. 数据库检查
    sqlDB, err := h.db.DB()
    if err != nil {
        status.Status = "unhealthy"
        status.Services["database"] = ServiceHealth{
            Status:  "unhealthy",
            Message: err.Error(),
        }
        return status
    }

    if err := sqlDB.Ping(); err != nil {
        status.Status = "degraded"
        status.Services["database"] = ServiceHealth{
            Status:  "unhealthy",
            Message: err.Error(),
        }
    } else {
        status.Services["database"] = ServiceHealth{
            Status: "healthy",
        }
    }

    // 2. 服务检查
    for name, service := range h.services {
        if err := service.HealthCheck(); err != nil {
            status.Status = "degraded"
            status.Services[name] = ServiceHealth{
                Status:  "unhealthy",
                Message: err.Error(),
            }
        } else {
            status.Services[name] = ServiceHealth{
                Status: "healthy",
            }
        }
    }

    return status
}

// HealthStatus 健康状态
type HealthStatus struct {
    Status    string                    `json:"status"`     // healthy | degraded | unhealthy
    Timestamp time.Time                 `json:"timestamp"`
    Services  map[string]ServiceHealth  `json:"services"`
}

// ServiceHealth 服务健康状态
type ServiceHealth struct {
    Status  string `json:"status"`  // healthy | unhealthy
    Message string `json:"message,omitempty"`
}

// Handler HTTP处理器
func (h *HealthChecker) Handler(c *gin.Context) {
    status := h.Check()

    httpStatus := http.StatusOK
    if status.Status == "degraded" {
        httpStatus = http.StatusServiceUnavailable
    } else if status.Status == "unhealthy" {
        httpStatus = http.StatusServiceUnavailable
    }

    c.JSON(httpStatus, status)
}

// ReadyHandler 就绪检查
func (h *HealthChecker) ReadyHandler(c *gin.Context) {
    // 只检查关键服务
    criticalServices := []string{
        "database",
        "video_recording_task_service",
    }

    for _, name := range criticalServices {
        if service, ok := h.services[name]; ok {
            if err := service.HealthCheck(); err != nil {
                c.JSON(http.StatusServiceUnavailable, gin.H{
                    "status":  "not_ready",
                    "service": name,
                    "error":   err.Error(),
                })
                return
            }
        }
    }

    c.JSON(http.StatusOK, gin.H{
        "status": "ready",
    })
}
```

## 六、分布式追踪

### 6.1 OpenTelemetry集成

```go
// internal/tracing/tracer.go
package tracing

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/trace"
)

// Tracer 追踪器
type Tracer struct {
    tracer trace.Tracer
}

// NewTracer 创建追踪器
func NewTracer(serviceName string) (*Tracer, error) {
    // 初始化OpenTelemetry
    tracerProvider := initTracerProvider(serviceName)
    otel.SetTracerProvider(tracerProvider)

    return &Tracer{
        tracer: otel.Tracer(serviceName),
    }, nil
}

// StartSpan 开始一个Span
func (t *Tracer) StartSpan(ctx context.Context, name string) (context.Context, trace.Span) {
    return t.tracer.Start(ctx, name)
}

// RecordError 记录错误
func (t *Tracer) RecordError(span trace.Span, err error) {
    if err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
    }
}

// 使用中间件
func TracingMiddleware(tracer *Tracer) gin.HandlerFunc {
    return func(c *gin.Context) {
        ctx, span := tracer.StartSpan(c.Request.Context(), c.Request.URL.Path)
        defer span.End()

        // 添加属性
        span.SetAttributes(
            attribute.String("http.method", c.Request.Method),
            attribute.String("http.url", c.Request.URL.String()),
            attribute.String("http.host", c.Request.Host),
            attribute.String("http.remote_addr", c.Request.RemoteAddr),
            attribute.String("http.user_agent", c.Request.UserAgent()),
        )

        // 传递上下文
        c.Request = c.Request.WithContext(ctx)
        c.Next()
    }
}
```
