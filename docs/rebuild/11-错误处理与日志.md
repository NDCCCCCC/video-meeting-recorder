```go
// internal/common/errors.go
package common

import (
    "fmt"
    "net/http"

// AppError 应用错误
type AppError struct {
    Code    int                    `json:"code"`
    Message string                 `json:"message"`
    Details map[string]interface{} `json:"details,omitempty"`
}

func (e *AppError) Error() string {
    return e.Message
}

// NewAppError 创建应用错误
func NewAppError(code int, message string, details map[string]interface{}) *AppError {
    return &AppError{
        Code:    code,
        Message: message,
        Details: details,
    }
}

// WrapError 包装错误
func WrapError(err error, message string) *AppError {
    return &AppError{
        Code:    CodeInternalError,
        Message: fmt.Sprintf("%s: %v", message, err),
        Details: map[string]interface{}{
            "original_error": err.Error(),
        },
    }
}
```

### 1.2 错误处理中间件

```go
// internal/middleware/error_handler.go
package middleware

import (
    "record_go/internal/common"
    "github.com/gin-gonic/gin"
)

// ErrorHandler 错误处理中间件
func ErrorHandler() gin.HandlerFunc {
    return func(c *gin.Context) {
        c.Next()

        // 检查是否有错误
        if len(c.Errors) > 0 {
            err := c.Errors.Last().Err

            // 处理应用错误
            if appErr, ok := err.(*common.AppError); ok {
                c.JSON(common.GetHTTPStatus(appErr.Code), gin.H{
                    "code":    appErr.Code,
                    "message": appErr.Message,
                    "data":    nil,
                })
                return
            }

            // 处理其他错误
            c.JSON(common.GetHTTPStatus(common.CodeInternalError), gin.H{
                "code":    common.CodeInternalError,
                "message": "内部服务器错误",
                "data":    nil,
            })
        }
    }
}

// RecoverHandler 恢复中间件
func RecoverHandler() gin.HandlerFunc {
    return func(c *gin.Context) {
        defer func() {
            if err := recover(); err != nil {
                common.GetLogger().Error("panic recovered",
                    zap.Any("error", err),
                    zap.Stack("stack"),
                )

                c.JSON(common.GetHTTPStatus(common.CodeInternalError), gin.H{
                    "code":    common.CodeInternalError,
                    "message": "内部服务器错误",
                    "data":    nil,
                })

                c.Abort()
            }
        }()

        c.Next()
    }
}
```

### 1.3 错误包装最佳实践

```go
// 服务层错误处理
func (s *UserService) CreateUser(req *CreateUserRequest) (*User, error) {
    // 1. 验证请求
    if err := req.Validate(); err != nil {
        return nil, common.NewAppError(
            common.CodeValidationFailed,
            "请求参数验证失败",
            map[string]interface{}{
                "details": err.Error(),
            },
        )
    }

    // 2. 检查用户是否存在
    var count int64
    if err := s.db.Model(&User{}).Where("username = ?", req.Username).Count(&count).Error; err != nil {
        return nil, common.WrapError(err, "查询用户失败")
    }

    if count > 0 {
        return nil, common.NewAppError(
            common.CodeUserExists,
            "用户名已存在",
            map[string]interface{}{
                "username": req.Username,
            },
        )
    }

    // 3. 创建用户
    user := &User{
        Username: req.Username,
        // ...
    }

    if err := s.db.Create(user).Error; err != nil {
        return nil, common.WrapError(err, "创建用户失败")
    }

    return user, nil
}
```

## 二、日志系统

### 2.1 日志配置

```go
// internal/logging/logger.go
package logging

import (
    "go.uber.org/zap"
    "go.uber.org/zap/zapcore"
    "gopkg.in/natefinch/lumberjack.v2"
)

type LoggerConfig struct {
    Level       string `mapstructure:"level"`        // debug | info | warn | error
    Format      string `mapstructure:"format"`       // json | text
    Output      string `mapstructure:"output"`       // stdout | file | both
    Filename    string `mapstructure:"filename"`
    MaxSize     int    `mapstructure:"max_size"`     // MB
    MaxBackups  int    `mapstructure:"max_backups"`
    MaxAge      int    `mapstructure:"max_age"`      // days
    Compress    bool   `mapstructure:"compress"`
    Development bool   `mapstructure:"development"`
}

// NewLogger 创建日志记录器
func NewLogger(config *LoggerConfig) (*zap.Logger, error) {
    // 1. 编码器配置
    encoderConfig := zap.NewProductionEncoderConfig()
    encoderConfig.TimeKey = "timestamp"
    encoderConfig.EncodeTime = zapcore.ISO8601TimeEncoder
    encoderConfig.EncodeLevel = zapcore.CapitalLevelEncoder
    encoderConfig.EncodeCaller = zapcore.ShortCallerEncoder

    // 2. 编码器
    var encoder zapcore.Encoder
    if config.Format == "json" {
        encoder = zapcore.NewJSONEncoder(encoderConfig)
    } else {
        encoder = zapcore.NewConsoleEncoder(encoderConfig)
    }

    // 3. 日志级别
    level, err := zapcore.ParseLevel(config.Level)
    if err != nil {
        level = zapcore.InfoLevel
    }

    // 4. 输出
    var writers []zapcore.WriteSyncer

    if config.Output == "stdout" || config.Output == "both" {
        writers = append(writers, zapcore.AddSync(os.Stdout))
    }

    if config.Output == "file" || config.Output == "both" {
        // 日志轮转
        writer := &lumberjack.Logger{
            Filename:   config.Filename,
            MaxSize:    config.MaxSize,
            MaxBackups: config.MaxBackups,
            MaxAge:     config.MaxAge,
            Compress:   config.Compress,
        }
        writers = append(writers, zapcore.AddSync(writer))
    }

    // 5. 创建Core
    core := zapcore.NewCore(
        encoder,
        zapcore.NewMultiWriteSyncer(writers...),
        level,
    )

    // 6. 创建Logger
    logger := zap.New(core,
        zap.AddCaller(),
        zap.AddCallerSkip(1),
        zap.AddStacktrace(zapcore.ErrorLevel),
    )

    return logger, nil
}
```

```go
// 操作日志
type OperationLog struct {
    Operation string                 `json:"operation"`
    UserID    uint                   `json:"user_id,omitempty"`
    Details   map[string]interface{} `json:"details,omitempty"`
    Error     string                 `json:"error,omitempty"`
    Duration  time.Duration          `json:"duration,omitempty"`
}

// 记录操作日志
func LogOperation(logger *zap.Logger, op *OperationLog) {
    fields := []zap.Field{
        zap.String("operation", op.Operation),
        zap.Duration("duration", op.Duration),
    }

    if op.UserID > 0 {
        fields = append(fields, zap.Uint("user_id", op.UserID))
    }

    for k, v := range op.Details {
        fields = append(fields, zap.Any(k, v))
    }

    if op.Error != "" {
        fields = append(fields, zap.String("error", op.Error))
        logger.Error("操作失败", fields...)
    } else {
        logger.Info("操作成功", fields...)
    }
}

// 使用示例
func (s *UserService) CreateUser(req *CreateUserRequest) (*User, error) {
    start := time.Now()

    // ... 业务逻辑 ...

    LogOperation(s.logger, &OperationLog{
        Operation: "create_user",
        UserID:     req.CreatorID,
        Details: map[string]interface{}{
            "username": req.Username,
            "role_id":  req.RoleID,
        },
        Duration: time.Since(start),
    })

    return user, nil
}
```

### 2.3 操作日志模式

```go
// 服务层统一日志模式
func (s *BaseService) LogOperationStart(operation string, fields ...zap.Field) {
    allFields := append([]zap.Field{
        zap.String("operation", operation),
        zap.String("stage", "start"),
    }, fields...)

    s.logger.Info("操作开始", allFields...)
}

func (s *BaseService) LogOperationSuccess(operation string, fields ...zap.Field) {
    allFields := append([]zap.Field{
        zap.String("operation", operation),
        zap.String("stage", "success"),
    }, fields...)

    s.logger.Info("操作成功", allFields...)
}

func (s *BaseService) LogOperationError(operation string, err error, fields ...zap.Field) {
    allFields := append([]zap.Field{
        zap.String("operation", operation),
        zap.String("stage", "error"),
        zap.Error(err),
    }, fields...)

    s.logger.Error("操作失败", allFields...)
}

// 使用示例
func (s *VideoRecordingTaskService) CreateTask(req *CreateTaskRequest, userID uint) (*VideoRecordingTask, error) {
    s.LogOperationStart("create_task",
        zap.Uint("user_id", userID),
        zap.String("name", req.Name),
    )

    task, err := s.createTaskInternal(req, userID)
    if err != nil {
        s.LogOperationError("create_task", err,
            zap.Uint("user_id", userID),
        )
        return nil, err
    }

    s.LogOperationSuccess("create_task",
        zap.Uint("task_id", task.ID),
        zap.Uint("user_id", userID),
    )

    return task, nil
}
```

## 三、日志分级

### 3.1 日志级别说明

| 级别 | 说明 | 使用场景 |
|------|------|----------|
| **Debug** | 调试信息 | 开发调试，详细输出 |
| **Info** | 一般信息 | 正常业务流程，重要操作 |
| **Warn** | 警告信息 | 潜在问题，可恢复的错误 |
| **Error** | 错误信息 | 错误异常，但不影响运行 |
| **Fatal** | 致命错误 | 严重错误，程序终止 |
| **Panic** | 恐慌错误 | 不可恢复的错误，打印堆栈 |

### 3.2 日志级别示例

```go
// Debug级别 - 详细调试信息
logger.Debug("处理请求",
    zap.String("method", c.Request.Method),
    zap.String("path", c.Request.URL.Path),
    zap.Any("headers", c.Request.Header),
)

// Info级别 - 正常业务流程
logger.Info("任务创建成功",
    zap.Uint("task_id", task.ID),
    zap.String("name", task.Name),
    zap.Time("start_time", task.StartTime),
)

// Warn级别 - 潜在问题
logger.Warn("磁盘空间不足",
    zap.Int64("free_space", freeSpace),
    zap.Int64("min_required", minRequired),
    zap.String("path", "/recordings"),
)

// Error级别 - 错误异常
logger.Error("数据库查询失败",
    zap.Error(err),
    zap.String("query", "SELECT * FROM users"),
)

// Fatal级别 - 致命错误(程序终止)
logger.Fatal("无法连接数据库",
    zap.Error(err),
    zap.String("host", dbHost),
    zap.Int("port", dbPort),
)
```

```go
// internal/middleware/request_logger.go
func RequestLogger(logger *zap.Logger) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()

        // 读取请求ID
        requestID := c.GetHeader("X-Request-ID")
        if requestID == "" {
            requestID = generateRequestID()
        }
        c.Set("request_id", requestID)

        // 记录请求
        logger.Debug("请求开始",
            zap.String("request_id", requestID),
            zap.String("method", c.Request.Method),
            zap.String("path", c.Request.URL.Path),
            zap.String("query", c.Request.URL.RawQuery),
            zap.String("ip", c.ClientIP()),
            zap.String("user_agent", c.Request.UserAgent()),
        )

        // 处理请求
        c.Next()

        // 记录响应
        duration := time.Since(start)
        fields := []zap.Field{
            zap.String("request_id", requestID),
            zap.String("method", c.Request.Method),
            zap.String("path", c.Request.URL.Path),
            zap.Int("status", c.Writer.Status()),
            zap.Duration("duration", duration),
            zap.Int64("response_size", c.Writer.Size()),
        }

        if userID := c.GetUint("user_id"); userID > 0 {
            fields = append(fields, zap.Uint("user_id", userID))
        }

        if c.Writer.Status() >= 400 {
            logger.Error("请求完成(错误)", fields...)
        } else {
            logger.Info("请求完成", fields...)
        }
    }
}
```

```go
// internal/models/audit_log.go
type AuditLog struct {
    ID        uint      `gorm:"primarykey" json:"id"`
    UserID    uint      `gorm:"index" json:"user_id,omitempty"`
    Username  string    `json:"username,omitempty"`
    Action    string    `gorm:"index" json:"action"`       // create, update, delete, etc.
    Resource  string    `json:"resource"`                 // user, task, conference, etc.
    ResourceID *uint    `json:"resource_id,omitempty"`
    Details   string    `gorm:"type:text" json:"details,omitempty"`
    IPAddress string    `json:"ip_address,omitempty"`
    UserAgent string    `json:"user_agent,omitempty"`
    Status    string    `json:"status"`                   // success, failure
    CreatedAt time.Time `json:"created_at"`
}

// 记录审计日志
func LogAudit(db *gorm.DB, userID uint, action, resource string, resourceID *uint, status string, details map[string]interface{}, c *gin.Context) {
    detailsJSON, _ := json.Marshal(details)

    auditLog := &AuditLog{
        UserID:     userID,
        Action:     action,
        Resource:   resource,
        ResourceID: resourceID,
        Details:    string(detailsJSON),
        Status:     status,
    }

    if c != nil {
        auditLog.IPAddress = c.ClientIP()
        auditLog.UserAgent = c.Request.UserAgent()
    }

    if userID > 0 {
        var user User
        db.First(&user, userID)
        auditLog.Username = user.Username
    }

    db.Create(auditLog)
}

// 使用示例
func (h *RecordingHandler) CreateTask(c *gin.Context) {
    // ... 创建任务 ...

    // 记录审计日志
    if err == nil {
        LogAudit(h.db, userID, "create", "recording_task", &task.ID, "success", map[string]interface{}{
            "task_name": task.Name,
            "start_time": task.StartTime,
        }, c)
    } else {
        LogAudit(h.db, userID, "create", "recording_task", nil, "failure", map[string]interface{}{
            "error": err.Error(),
        }, c)
    }
}
```
