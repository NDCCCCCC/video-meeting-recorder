# 部署运维指南

## 概述

本文档描述了视频会议录制系统的部署和运维流程，包括环境准备、构建部署、监控告警、故障排查等内容。

## 系统要求

### 硬件要求

| 组件 | 最低配置 | 推荐配置 |
|------|----------|----------|
| CPU | 4 核 | 8 核+ |
| 内存 | 8 GB | 16 GB+ |
| 存储 | 100 GB SSD | 500 GB SSD |
| 网络 | 100 Mbps | 1 Gbps |

### 软件要求

| 软件 | 版本 |
|------|------|
| 操作系统 | Linux (Ubuntu 20.04+, CentOS 7+) |
| Go | 1.21+ |
| FFmpeg | 4.0+ |
| SQLite | 3.35+ |

## 部署方式

### 方式一：二进制部署

#### 1. 构建应用

```bash
# 克隆代码
git clone https://github.com/example/record_go.git
cd record_go

# 构建二进制
make build

# 或手动构建
CGO_ENABLED=1 go build \
    -ldflags="-s -w -X main.version=v1.0.0" \
    -o bin/server \
    ./cmd/server
```

#### 2. 创建目录

```bash
# 创建目录结构
sudo mkdir -p /opt/record_go/{bin,config,data,logs,storage/{videos,ppts,temp},scripts}
sudo mkdir -p /var/run/record_go

# 设置权限
sudo chown -R appuser:appuser /opt/record_go
sudo chown -R appuser:appuser /var/run/record_go
```

#### 3. 复制文件

```bash
# 复制二进制
sudo cp bin/server /opt/record_go/bin/

# 复制配置
sudo cp config.yaml.example /opt/record_go/config/config.yaml

# 复制脚本
sudo cp scripts/*.sh /opt/record_go/scripts/
sudo chmod +x /opt/record_go/scripts/*.sh
```

#### 4. 配置 systemd 服务

```bash
# 创建服务文件
sudo tee /etc/systemd/system/record-go.service > /dev/null <<EOF
[Unit]
Description=Video Conference Recording Server
Documentation=https://github.com/example/record_go
After=network.target

[Service]
Type=simple
User=appuser
Group=appuser
WorkingDirectory=/opt/record_go
ExecStart=/opt/record_go/bin/server --config=/opt/record_go/config/config.yaml
ExecStop=/opt/record_go/scripts/stop.sh
Restart=always
RestartSec=5
StandardOutput=append:/opt/record_go/logs/app.log
StandardError=append:/opt/record_go/logs/error.log

# 安全设置
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/opt/record_go/data /opt/record_go/logs /opt/record_go/storage

# 资源限制
LimitNOFILE=65536
LimitNPROC=4096

[Install]
WantedBy=multi-user.target
EOF

# 重载并启动
sudo systemctl daemon-reload
sudo systemctl enable record-go
sudo systemctl start record-go
```

## 启动脚本

**文件**: `/opt/record_go/scripts/start.sh`

```bash
#!/bin/bash

set -e

# 配置
APP_NAME="record-go"
APP_DIR="/opt/record_go"
BIN_DIR="${APP_DIR}/bin"
CONFIG_DIR="${APP_DIR}/config"
LOG_DIR="${APP_DIR}/logs"
PID_FILE="/var/run/${APP_NAME}/${APP_NAME}.pid"
LOCK_FILE="/var/lock/subsys/${APP_NAME}"

# 颜色
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $(date '+%Y-%m-%d %H:%M:%S') $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $(date '+%Y-%m-%d %H:%M:%S') $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $(date '+%Y-%m-%d %H:%M:%S') $1"
}

# 检查是否已运行
is_running() {
    if [ -f "$PID_FILE" ]; then
        PID=$(cat "$PID_FILE")
        if ps -p "$PID" > /dev/null 2>&1; then
            return 0
        else
            rm -f "$PID_FILE"
        fi
    fi
    return 1
}

# 创建必要目录
create_dirs() {
    mkdir -p "$LOG_DIR"
    mkdir -p "$(dirname $PID_FILE)"
    mkdir -p "${APP_DIR}/data"
    mkdir -p "${APP_DIR}/storage/videos"
    mkdir -p "${APP_DIR}/storage/ppts"
    mkdir -p "${APP_DIR}/storage/temp"
}

# 启动应用
start() {
    if is_running; then
        log_warn "$APP_NAME 已在运行 (PID: $(cat $PID_FILE))"
        return 1
    fi

    log_info "启动 $APP_NAME..."

    create_dirs

    # 加载环境变量
    if [ -f "${APP_DIR}/.env" ]; then
        export $(cat "${APP_DIR}/.env" | grep -v '^#' | xargs)
    fi

    # 启动服务
    cd "$APP_DIR"
    nohup "${BIN_DIR}/server" \
        --config="${CONFIG_DIR}/config.yaml" \
        >> "${LOG_DIR}/app.log" 2>&1 &

    PID=$!
    echo $PID > "$PID_FILE"
    touch "$LOCK_FILE"

    # 等待启动
    sleep 2

    if ps -p "$PID" > /dev/null; then
        log_info "$APP_NAME 启动成功 (PID: $PID)"
        return 0
    else
        log_error "$APP_NAME 启动失败"
        rm -f "$PID_FILE"
        return 1
    fi
}

# 主函数
main() {
    start
    exit $?
}

main "$@"
```

## 停止脚本

**文件**: `/opt/record_go/scripts/stop.sh`

```bash
#!/bin/bash

set -e

APP_NAME="record-go"
PID_FILE="/var/run/${APP_NAME}/${APP_NAME}.pid"
LOCK_FILE="/var/lock/subsys/${APP_NAME}"

log_info() {
    echo "[INFO] $(date '+%Y-%m-%d %H:%M:%S') $1"
}

stop() {
    if [ ! -f "$PID_FILE" ]; then
        echo "$APP_NAME 未运行"
        return 0
    fi

    PID=$(cat "$PID_FILE")

    log_info "停止 $APP_NAME (PID: $PID)..."

    # 发送 TERM 信号
    kill -TERM "$PID" 2>/dev/null || true

    # 等待进程退出
    TIMEOUT=30
    while [ $TIMEOUT -gt 0 ]; do
        if ! ps -p "$PID" > /dev/null 2>&1; then
            break
        fi
        sleep 1
        TIMEOUT=$((TIMEOUT - 1))
    done

    # 如果还没退出，强制杀死
    if ps -p "$PID" > /dev/null 2>&1; then
        log_warn "进程未响应，强制终止..."
        kill -KILL "$PID" 2>/dev/null || true
        sleep 1
    fi

    # 清理
    rm -f "$PID_FILE" "$LOCK_FILE"

    log_info "$APP_NAME 已停止"
}

stop
```

## 备份策略

### 数据库备份

**文件**: `/opt/record_go/scripts/backup.sh`

```bash
#!/bin/bash

set -e

# 配置
APP_NAME="record-go"
BACKUP_DIR="/backup/${APP_NAME}"
DATA_DIR="/opt/record_go/data"
RETENTION_DAYS=30

# 创建备份目录
mkdir -p "$BACKUP_DIR/daily"
mkdir -p "$BACKUP_DIR/weekly"
mkdir -p "$BACKUP_DIR/monthly"

# 日期
DATE=$(date +%Y%m%d)
DAY_OF_WEEK=$(date +%u)  # 1-7 (Monday-Sunday)
DAY_OF_MONTH=$(date +%d)

# 备份函数
backup_database() {
    local backup_file="$BACKUP_DIR/daily/app_${DATE}.db"

    log_info "备份数据库..."

    # 使用 SQLite 备份命令
    sqlite3 "${DATA_DIR}/app.db" ".backup '${backup_file}'"

    # 压缩
    gzip "$backup_file"

    # 保留策略
    find "$BACKUP_DIR/daily" -name "*.db.gz" -mtime +${RETENTION_DAYS} -delete

    log_info "数据库备份完成: ${backup_file}.gz"
}

# 周备份
backup_weekly() {
    if [ "$DAY_OF_WEEK" -eq 7 ]; then
        local backup_file="$BACKUP_DIR/weekly/app_${DATE}.db.gz"
        cp "$BACKUP_DIR/daily/app_${DATE}.db.gz" "$backup_file"

        # 保留 12 周
        find "$BACKUP_DIR/weekly" -name "*.db.gz" -mtime +84 -delete
    fi
}

# 月备份
backup_monthly() {
    if [ "$DAY_OF_MONTH" -eq 1 ]; then
        local backup_file="$BACKUP_DIR/monthly/app_${DATE}.db.gz"
        cp "$BACKUP_DIR/daily/app_${DATE}.db.gz" "$backup_file"

        # 保留 12 个月
        find "$BACKUP_DIR/monthly" -name "*.db.gz" -mtime +365 -delete
    fi
}

# 主函数
main() {
    backup_database
    backup_weekly
    backup_monthly
}

main "$@"
```

```nginx
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```
