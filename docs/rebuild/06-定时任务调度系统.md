# 定时任务调度系统

## 一、调度系统概述

### 1.1 功能概述

定时任务调度系统负责管理视频录制任务的定时执行，基于Cron表达式实现精确的时间控制。

**核心功能**：
- **任务调度** - 基于Cron表达式的定时触发
- **任务执行** - 完整的任务执行流程管理
- **任务监控** - 实时监控任务状态
- **失败重试** - 支持失败任务的重试
- **并发控制** - 控制同时执行的任务数量

## 二、Cron表达式生成

### 2.1 表达式格式

```
格式: 秒 分 时 日 月 周
      *  *  *  *  *  *

字段说明:
- 秒   (0-59)
- 分   (0-59)
- 时   (0-23)
- 日   (1-31)
- 月   (1-12)
- 周   (0-6, 0=周日)

特殊符号:
*     : 所有值
/     : 步长
-     : 范围
,     : 列表
?     : 不指定(仅用于日和周)
```

### 2.2 表达式生成

```go
// internal/services/video_recording/video_simple_scheduler.go
func generateCronExpression(triggerTime time.Time) string {
    // Cron 格式: 秒 分 时 日 月 周
    return fmt.Sprintf("%d %d %d %d %d *",
        triggerTime.Second(),
        triggerTime.Minute(),
        triggerTime.Hour(),
        triggerTime.Day(),
        int(triggerTime.Month()),
    )
}

// 示例:
// 触发时间: 2024-01-15 14:30:00
// Cron 表达式: "0 30 14 15 1 *"
// 含义: 在1月15日14:30:00执行

// 触发时间: 2024-02-20 09:00:00
// Cron 表达式: "0 0 9 20 2 *"
// 含义: 在2月20日09:00:00执行
```

### 2.3 触发时间计算

```go
// 计算触发时间
func (t *VideoRecordingTask) GetTriggerTime() time.Time {
    // 触发时间 = 开始时间 - 提前进入时间
    return t.StartTime.Add(-time.Duration(t.PreJoinMinutes) * time.Minute)
}

// 示例:
// StartTime: 2024-01-15 14:00:00
// PreJoinMinutes: 5
// TriggerTime: 2024-01-15 13:55:00

// 任务执行时间线:
// 13:55:00 ──> 任务触发, 开始连接会议
// 14:00:00 ──> 会议正式开始
// 14:00:00 ──> 延迟0分钟后开始录制
// 15:00:00 ──> 会议结束
// 15:00:00 ──> 停止录制, 断开连接
```

## 三、VideoSimpleScheduler 实现

### 3.1 调度器结构

```go
// internal/services/video_recording/video_simple_scheduler.go
type VideoSimpleScheduler struct {
    cron              *cron.Cron
    taskService       VideoRecordingTaskServiceInterface
    huaweiConnector   HuaweiConferenceConnectorInterface
    coordinator       CoordinatorInterface
    taskEntries       map[uint]cron.EntryID  // 任务ID -> Cron条目ID
    entryTasks        map[cron.EntryID]uint  // Cron条目ID -> 任务ID
    executingTasks    map[uint]bool         // 正在执行的任务
    logger            *zap.Logger
    config            *config.Config
    mu                sync.RWMutex
}

// NewVideoSimpleScheduler 创建调度器
func NewVideoSimpleScheduler(
    taskService VideoRecordingTaskServiceInterface,
    huaweiConnector HuaweiConferenceConnectorInterface,
    coordinator CoordinatorInterface,
    logger *zap.Logger,
    config *config.Config,
) *VideoSimpleScheduler {
    // 创建Cron调度器 (秒级精度)
    c := cron.New(cron.WithSeconds())

    scheduler := &VideoSimpleScheduler{
        cron:            c,
        taskService:     taskService,
        huaweiConnector: huaweiConnector,
        coordinator:     coordinator,
        taskEntries:     make(map[uint]cron.EntryID),
        entryTasks:      make(map[cron.EntryID]uint),
        executingTasks:  make(map[uint]bool),
        logger:          logger,
        config:          config,
    }

    // 启动调度器
    c.Start()

    return scheduler
}
```

### 3.2 添加任务

```go
// AddTask 添加任务到调度器
func (s *VideoSimpleScheduler) AddTask(task *models.VideoRecordingTask) error {
    s.mu.Lock()
    defer s.mu.Unlock()

    // 1. 检查任务是否已存在
    if _, exists := s.taskEntries[task.ID]; exists {
        return fmt.Errorf("任务已在调度器中: %d", task.ID)
    }

    // 2. 计算触发时间
    triggerTime := task.GetTriggerTime()

    // 3. 检查触发时间是否在未来
    if triggerTime.Before(time.Now()) {
        return fmt.Errorf("触发时间已过期: %s", triggerTime.Format(time.RFC3339))
    }

    // 4. 生成Cron表达式
    cronExpr := generateCronExpression(triggerTime)

    // 5. 添加到Cron调度器
    entryID, err := s.cron.AddFunc(cronExpr, func() {
        s.executeTask(task.ID)
    })
    if err != nil {
        return fmt.Errorf("添加Cron任务失败: %w", err)
    }

    // 6. 保存映射关系
    s.taskEntries[task.ID] = entryID
    s.entryTasks[entryID] = task.ID

    s.logger.Info("任务已添加到调度器",
        zap.Uint("task_id", task.ID),
        zap.String("cron_expr", cronExpr),
        zap.Time("trigger_time", triggerTime),
    )

    return nil
}

// RemoveTask 从调度器移除任务
func (s *VideoSimpleScheduler) RemoveTask(taskID uint) error {
    s.mu.Lock()
    defer s.mu.Unlock()

    // 1. 获取Cron条目ID
    entryID, exists := s.taskEntries[taskID]
    if !exists {
        return fmt.Errorf("任务不在调度器中: %d", taskID)
    }

    // 2. 从Cron调度器移除
    s.cron.Remove(entryID)

    // 3. 删除映射关系
    delete(s.taskEntries, taskID)
    delete(s.entryTasks, entryID)

    s.logger.Info("任务已从调度器移除",
        zap.Uint("task_id", taskID),
    )

    return nil
}
```

### 3.3 任务执行

```go
// executeTask 执行任务
func (s *VideoSimpleScheduler) executeTask(taskID uint) {
    s.logger.Info("开始执行任务",
        zap.Uint("task_id", taskID),
    )

    // 1. 检查是否正在执行
    s.mu.Lock()
    if s.executingTasks[taskID] {
        s.mu.Unlock()
        s.logger.Warn("任务正在执行中", zap.Uint("task_id", taskID))
        return
    }
    s.executingTasks[taskID] = true
    s.mu.Unlock()

    defer func() {
        s.mu.Lock()
        delete(s.executingTasks, taskID)
        s.mu.Unlock()
    }()

    // 2. 加载任务
    task, err := s.taskService.GetTask(taskID)
    if err != nil {
        s.logger.Error("加载任务失败",
            zap.Uint("task_id", taskID),
            zap.Error(err),
        )
        return
    }

    // 3. 验证任务状态
    if task.Status != models.VideoStatusPending {
        s.logger.Warn("任务状态不正确",
            zap.Uint("task_id", taskID),
            zap.String("status", string(task.Status)),
        )
        return
    }

    // 4. 检查任务是否过期
    if task.IsExpired() {
        s.logger.Warn("任务已过期",
            zap.Uint("task_id", taskID),
            zap.Time("end_time", task.EndTime),
        )
        s.updateTaskStatus(taskID, models.VideoStatusFailed, "任务已过期")
        return
    }

    // 5. 更新状态为连接中
    if err := s.updateTaskStatus(taskID, models.VideoStatusConnecting, ""); err != nil {
        s.logger.Error("更新任务状态失败", zap.Error(err))
        return
    }

    // 6. 连接华为会议
    if err := s.huaweiConnector.ConnectToConference(task); err != nil {
        s.logger.Error("连接华为会议失败",
            zap.Uint("task_id", taskID),
            zap.Error(err),
        )
        s.updateTaskStatus(taskID, models.VideoStatusFailed, err.Error())
        return
    }

    // 7. 等待录制延迟
    if task.RecordDelayMinutes > 0 {
        s.logger.Info("等待录制延迟",
            zap.Uint("task_id", taskID),
            zap.Int("delay_minutes", task.RecordDelayMinutes),
        )
        time.Sleep(time.Duration(task.RecordDelayMinutes) * time.Minute)
    }

    // 8. 加载华为配置
    huaweiConfig, err := s.taskService.GetHuaweiConfig(task.HuaweiConfigID)
    if err != nil {
        s.logger.Error("加载华为配置失败", zap.Error(err))
        s.updateTaskStatus(taskID, models.VideoStatusFailed, err.Error())
        s.huaweiConnector.DisconnectFromConference(task)
        return
    }

    // 9. 更新状态为录制中
    if err := s.updateTaskStatus(taskID, models.VideoStatusRecording, ""); err != nil {
        s.logger.Error("更新任务状态失败", zap.Error(err))
        s.huaweiConnector.DisconnectFromConference(task)
        return
    }

    // 10. 启动录制
    if err := s.coordinator.StartRecording(task, huaweiConfig); err != nil {
        s.logger.Error("启动录制失败",
            zap.Uint("task_id", taskID),
            zap.Error(err),
        )
        s.updateTaskStatus(taskID, models.VideoStatusFailed, err.Error())
        s.huaweiConnector.DisconnectFromConference(task)
        return
    }

    // 11. 启动监控
    go s.monitorTask(task)

    s.logger.Info("任务执行完成, 录制中",
        zap.Uint("task_id", taskID),
    )
}

// monitorTask 监控任务
func (s *VideoSimpleScheduler) monitorTask(task *models.VideoRecordingTask) {
    // 计算剩余时间
    remaining := time.Until(task.EndTime)
    if remaining < 0 {
        remaining = 0
    }

    s.logger.Info("开始监控任务",
        zap.Uint("task_id", task.ID),
        zap.Duration("remaining", remaining),
    )

    // 等待会议结束
    timer := time.NewTimer(remaining)
    <-timer.C

    // 完成任务
    s.completeTask(task.ID)
}
```

```go
// completeTask 完成任务
func (s *VideoSimpleScheduler) completeTask(taskID uint) {
    s.logger.Info("开始完成任务",
        zap.Uint("task_id", taskID),
    )

    // 1. 加载任务
    task, err := s.taskService.GetTask(taskID)
    if err != nil {
        s.logger.Error("加载任务失败", zap.Error(err))
        return
    }

    // 2. 停止录制
    if err := s.coordinator.StopRecording(taskID); err != nil {
        s.logger.Error("停止录制失败",
            zap.Uint("task_id", taskID),
            zap.Error(err),
        )
    }

    // 3. 断开华为会议
    if err := s.huaweiConnector.DisconnectFromConference(task); err != nil {
        s.logger.Error("断开华为会议失败",
            zap.Uint("task_id", taskID),
            zap.Error(err),
        )
    }

    // 4. 更新任务状态
    if err := s.updateTaskStatus(taskID, models.VideoStatusCompleted, ""); err != nil {
        s.logger.Error("更新任务状态失败", zap.Error(err))
    }

    // 5. 从调度器移除
    s.RemoveTask(taskID)

    s.logger.Info("任务完成",
        zap.Uint("task_id", taskID),
    )
}
```

### 4.1 并发限制

```go
// internal/services/video_recording/video_simple_scheduler.go
type ConcurrencyLimiter struct {
    semaphore chan struct{}
    logger    *zap.Logger
}

// NewConcurrencyLimiter 创建并发限制器
func NewConcurrencyLimiter(maxConcurrent int, logger *zap.Logger) *ConcurrencyLimiter {
    return &ConcurrencyLimiter{
        semaphore: make(chan struct{}, maxConcurrent),
        logger:    logger,
    }
}

// Acquire 获取执行许可
func (l *ConcurrencyLimiter) Acquire(ctx context.Context) error {
    select {
    case l.semaphore <- struct{}{}:
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

// Release 释放执行许可
func (l *ConcurrencyLimiter) Release() {
    select {
    case <-l.semaphore:
    default:
    }
}

// 使用示例
func (s *VideoSimpleScheduler) executeTaskWithLimit(taskID uint) {
    // 获取许可
    if err := s.limiter.Acquire(context.Background()); err != nil {
        s.logger.Error("获取执行许可失败", zap.Error(err))
        return
    }
    defer s.limiter.Release()

    // 执行任务
    s.executeTask(taskID)
}
```

```go
// PriorityQueue 优先级队列
type PriorityQueue struct {
    items []*models.VideoRecordingTask
    less  func(i, j int) bool
}

func (pq *PriorityQueue) Push(task *models.VideoRecordingTask) {
    pq.items = append(pq.items, task)
    pq.up(len(pq.items) - 1)
}

func (pq *PriorityQueue) Pop() *models.VideoRecordingTask {
    if len(pq.items) == 0 {
        return nil
    }

    item := pq.items[0]
    pq.items[0] = pq.items[len(pq.items)-1]
    pq.items = pq.items[:len(pq.items)-1]
    pq.down(0)

    return item
}
```

## 五、持久化与恢复

### 5.1 状态持久化

```go
// SchedulerState 调度器状态
type SchedulerState struct {
    ScheduledTasks []uint    `json:"scheduled_tasks"`
    ExecutingTasks []uint    `json:"executing_tasks"`
    LastSync       time.Time `json:"last_sync"`
}

// SaveState 保存状态
func (s *VideoSimpleScheduler) SaveState() error {
    s.mu.RLock()
    defer s.mu.RUnlock()

    state := &SchedulerState{
        ScheduledTasks: make([]uint, 0, len(s.taskEntries)),
        ExecutingTasks: make([]uint, 0, len(s.executingTasks)),
        LastSync:       time.Now(),
    }

    for taskID := range s.taskEntries {
        state.ScheduledTasks = append(state.ScheduledTasks, taskID)
    }

    for taskID := range s.executingTasks {
        state.ExecutingTasks = append(state.ExecutingTasks, taskID)
    }

    // 保存到数据库或文件
    data, err := json.Marshal(state)
    if err != nil {
        return err
    }

    return os.WriteFile("scheduler_state.json", data, 0644)
}

// LoadState 加载状态
func (s *VideoSimpleScheduler) LoadState() error {
    data, err := os.ReadFile("scheduler_state.json")
    if err != nil {
        return err
    }

    var state SchedulerState
    if err := json.Unmarshal(data, &state); err != nil {
        return err
    }

    // 恢复调度任务
    for _, taskID := range state.ScheduledTasks {
        task, err := s.taskService.GetTask(taskID)
        if err != nil {
            s.logger.Warn("加载任务失败",
                zap.Uint("task_id", taskID),
                zap.Error(err),
            )
            continue
        }

        // 检查任务是否仍然有效
        if task.Status == models.VideoStatusPending && task.GetTriggerTime().After(time.Now()) {
            s.AddTask(task)
        }
    }

    // 处理正在执行的任务
    for _, taskID := range state.ExecutingTasks {
        // 标记任务为失败，需要人工处理
        s.updateTaskStatus(taskID, models.VideoStatusFailed, "系统重启，任务中断")
    }

    return nil
}
```

### 5.2 启动恢复

```go
// Start 启动调度器
func (s *VideoSimpleScheduler) Start() error {
    s.logger.Info("启动调度器")

    // 1. 加载持久化状态
    if err := s.LoadState(); err != nil {
        s.logger.Warn("加载调度器状态失败", zap.Error(err))
    }

    // 2. 从数据库加载待执行的任务
    tasks, err := s.taskService.GetPendingTasks()
    if err != nil {
        return fmt.Errorf("加载待执行任务失败: %w", err)
    }

    // 3. 添加到调度器
    for _, task := range tasks {
        // 检查触发时间
        if task.GetTriggerTime().After(time.Now()) {
            if err := s.AddTask(&task); err != nil {
                s.logger.Error("添加任务失败",
                    zap.Uint("task_id", task.ID),
                    zap.Error(err),
                )
            }
        } else {
            // 触发时间已过，标记为失败
            s.updateTaskStatus(task.ID, models.VideoStatusFailed, "触发时间已过期")
        }
    }

    // 4. 启动状态保存定时器
    go s.periodicSaveState()

    s.logger.Info("调度器启动完成",
        zap.Int("scheduled_tasks", len(s.taskEntries)),
    )

    return nil
}

// Stop 停止调度器
func (s *VideoSimpleScheduler) Stop() {
    s.logger.Info("停止调度器")

    // 1. 保存状态
    s.SaveState()

    // 2. 停止Cron调度器
    s.cron.Stop()

    // 3. 等待正在执行的任务
    s.mu.RLock()
    executingCount := len(s.executingTasks)
    s.mu.RUnlock()

    if executingCount > 0 {
        s.logger.Info("等待执行中的任务完成",
            zap.Int("count", executingCount),
        )
        // 等待最多5分钟
        deadline := time.Now().Add(5 * time.Minute)
        for time.Now().Before(deadline) {
            s.mu.RLock()
            executingCount = len(s.executingTasks)
            s.mu.RUnlock()

            if executingCount == 0 {
                break
            }
            time.Sleep(1 * time.Second)
        }
    }

    s.logger.Info("调度器已停止")
}

// periodicSaveState 定期保存状态
func (s *VideoSimpleScheduler) periodicSaveState() {
    ticker := time.NewTicker(1 * time.Minute)
    defer ticker.Stop()

    for range ticker.C {
        if err := s.SaveState(); err != nil {
            s.logger.Error("保存调度器状态失败", zap.Error(err))
        }
    }
}
```
