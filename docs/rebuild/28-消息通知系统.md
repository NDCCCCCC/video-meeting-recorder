# 消息通知系统

## 一、系统概述

### 1.1 设计目标

消息通知系统提供统一的消息推送服务，支持多种通知渠道：

- **站内消息** - 系统内置消息中心
- **邮件通知** - 支持SMTP邮件发送
- **短信通知** - 支持阿里云/腾讯云短信
- **企业IM** - 支持钉钉、企业微信、飞书

### 1.2 核心特性

| 特性 | 说明 |
|------|------|
| **多渠道支持** | 统一接口，支持多种通知渠道 |
| **异步发送** | 消息队列异步处理，不阻塞主流程 |
| **失败重试** | 自动重试机制，提高送达率 |
| **模板管理** | 消息模板支持，参数化配置 |
| **用户偏好** | 用户可自定义通知渠道和频率 |
| **免打扰** | 支持免打扰时段设置 |

```go
// internal/services/notification/notification_service.go
package notification

import (
    "context"
    "encoding/json"
    "fmt"
    "time"

    "go.uber.org/zap"
    "gorm.io/gorm"
)

// NotificationService 通知服务
type NotificationService struct {
    db      *gorm.DB
    logger  *zap.Logger
    config  *config.Config

    // 渠道发送器
    senders map[NotificationChannel]ChannelSender

    // 消息队列
    queue chan *SendTask
}

// SendTask 发送任务
type SendTask struct {
    Message  *NotificationMessage
    Channels []NotificationChannel
    Retry    int
}

// ChannelSender 渠道发送器接口
type ChannelSender interface {
    Send(ctx context.Context, msg *NotificationMessage) error
    Name() NotificationChannel
}

// NewNotificationService 创建通知服务
func NewNotificationService(db *gorm.DB, logger *zap.Logger, config *config.Config) *NotificationService {
    service := &NotificationService{
        db:      db,
        logger:  logger,
        config:  config,
        senders: make(map[NotificationChannel]ChannelSender),
        queue:   make(chan *SendTask, 1000),
    }

    // 初始化渠道发送器
    service.initSenders()

    // 启动处理goroutine
    go service.processQueue()

    return service
}

// initSenders 初始化渠道发送器
func (s *NotificationService) initSenders() {
    // 站内消息（默认启用）
    s.senders[ChannelSystem] = NewSystemSender(s.db, s.logger)

    // 邮件发送器
    if s.config.Notification.Email.Enabled {
        s.senders[ChannelEmail] = NewEmailSender(
            s.config.Notification.Email.SMTPHost,
            s.config.Notification.Email.SMTPPort,
            s.config.Notification.Email.Username,
            s.config.Notification.Email.Password,
            s.config.Notification.Email.From,
        )
    }

    // 短信发送器
    if s.config.Notification.SMS.Enabled {
        s.senders[ChannelSMS] = NewSMSSender(
            s.config.Notification.SMS.Provider,
            s.config.Notification.SMS.AccessKey,
            s.config.Notification.SMS.SecretKey,
        )
    }

    // 钉钉发送器
    if s.config.Notification.DingTalk.Enabled {
        s.senders[ChannelDingTalk] = NewDingTalkSender(
            s.config.Notification.DingTalk.WebhookURL,
            s.config.Notification.DingTalk.Secret,
        )
    }

    // 企业微信发送器
    if s.config.Notification.WeChat.Enabled {
        s.senders[ChannelWeChat] = NewWeChatSender(
            s.config.Notification.WeChat.CorpID,
            s.config.Notification.WeChat.AgentID,
            s.config.Notification.WeChat.AgentSecret,
        )
    }
}

// processQueue 处理发送队列
func (s *NotificationService) processQueue() {
    for task := range s.queue {
        go s.handleSendTask(task)
    }
}

// handleSendTask 处理发送任务
func (s *NotificationService) handleSendTask(task *SendTask) {
    ctx := context.Background()
    channelStatus := make(map[string]string)

    for _, channel := range task.Channels {
        sender, ok := s.senders[channel]
        if !ok {
            s.logger.Warn("未找到发送器",
                zap.String("channel", string(channel)),
            )
            channelStatus[string(channel)] = string(ChannelStatusFailed)
            continue
        }

        // 发送消息
        err := sender.Send(ctx, task.Message)
        if err != nil {
            s.logger.Error("发送失败",
                zap.String("channel", string(channel)),
                zap.Uint("message_id", task.Message.ID),
                zap.Error(err),
            )
            channelStatus[string(channel)] = string(ChannelStatusFailed)

            // 重试逻辑
            if task.Retry < 3 {
                task.Retry++
                time.AfterFunc(time.Minute*time.Duration(task.Retry), func() {
                    s.queue <- task
                })
            }
        } else {
            channelStatus[string(channel)] = string(ChannelStatusSent)
        }
    }

    // 更新消息状态
    statusJSON, _ := json.Marshal(channelStatus)
    s.db.Model(task.Message).Update("channel_status", statusJSON)
}

// SendNotification 发送通知
func (s *NotificationService) SendNotification(ctx context.Context, req *SendNotificationRequest) error {
    // 1. 获取用户通知配置
    setting, err := s.getUserSetting(req.UserID)
    if err != nil {
        return err
    }

    // 2. 检查免打扰时段
    if s.isQuietHours(setting) {
        // 延迟到免打扰结束后发送
        return s.scheduleNotification(req, setting)
    }

    // 3. 检查频率限制
    if !s.checkRateLimit(req.UserID, setting) {
        return ErrRateLimitExceeded
    }

    // 4. 渲染消息内容
    message, err := s.renderMessage(req)
    if err != nil {
        return err
    }

    // 5. 保存到数据库
    if err := s.db.Create(message).Error; err != nil {
        return err
    }

    // 6. 确定发送渠道
    channels := s.determineChannels(req.Channels, setting, req.Type)

    // 7. 放入发送队列
    s.queue <- &SendTask{
        Message:  message,
        Channels: channels,
        Retry:    0,
    }

    return nil
}

// getUserSetting 获取用户通知配置
func (s *NotificationService) getUserSetting(userID uint) (*models.UserNotificationSetting, error) {
    var setting models.UserNotificationSetting
    err := s.db.Where("user_id = ?", userID).First(&setting).Error
    if err != nil {
        if err == gorm.ErrRecordNotFound {
            // 返回默认配置
            return s.getDefaultSetting(userID), nil
        }
        return nil, err
    }
    return &setting, nil
}

// getDefaultSetting 获取默认配置
func (s *NotificationService) getDefaultSetting(userID uint) *models.UserNotificationSetting {
    return &models.UserNotificationSetting{
        UserID:                  userID,
        EmailEnabled:            true,
        SMSEnabled:              false,
        DingTalkEnabled:         false,
        WeChatEnabled:           false,
        FeiShuEnabled:           false,
        TaskEnabled:             true,
        SystemEnabled:           true,
        WarningEnabled:          true,
        ReminderEnabled:         true,
        EnableQuietHours:        false,
        QuietHoursStart:         "22:00:00",
        QuietHoursEnd:           "08:00",
        MaxNotificationsPerHour: 10,
    }
}

// isQuietHours 检查是否在免打扰时段
func (s *NotificationService) isQuietHours(setting *models.UserNotificationSetting) bool {
    if !setting.EnableQuietHours {
        return false
    }

    now := time.Now()
    currentTime := now.Format("15:04:05")

    // 处理跨天情况
    if setting.QuietHoursStart < setting.QuietHoursEnd {
        return currentTime >= setting.QuietHoursStart && currentTime <= setting.QuietHoursEnd
    } else {
        return currentTime >= setting.QuietHoursStart || currentTime <= setting.QuietHoursEnd
    }
}

// checkRateLimit 检查频率限制
func (s *NotificationService) checkRateLimit(userID uint, setting *models.UserNotificationSetting) bool {
    // 使用Redis实现滑动窗口限流
    // key: notification:rate_limit:{user_id}:{hour}
    // 检查当前小时已发送数量

    // 这里简化实现，实际应使用Redis
    var count int64
    s.db.Model(&models.NotificationMessage{}).
        Where("user_id = ? AND created_at >= ?", userID, time.Now().Truncate(time.Hour)).
        Count(&count)

    return int(count) < setting.MaxNotificationsPerHour
}

// renderMessage 渲染消息
func (s *NotificationService) renderMessage(req *SendNotificationRequest) (*models.NotificationMessage, error) {
    // 如果指定了模板，使用模板渲染
    if req.TemplateCode != "" {
        return s.renderFromTemplate(req)
    }

    // 直接使用提供的内容
    message := &models.NotificationMessage{
        UserID:       req.UserID,
        Type:         req.Type,
        Title:        req.Title,
        Content:      req.Content,
        RelatedID:    req.RelatedID,
        RelatedType:  req.RelatedType,
        RelatedURL:   req.RelatedURL,
        IsRead:       false,
        CreatedAt:    time.Now(),
    }

    if req.Data != nil {
        dataJSON, _ := json.Marshal(req.Data)
        message.Data = string(dataJSON)
    }

    return message, nil
}

// renderFromTemplate 从模板渲染
func (s *NotificationService) renderFromTemplate(req *SendNotificationRequest) (*models.NotificationMessage, error) {
    var template models.NotificationTemplate
    err := s.db.Where("code = ? AND is_active = ?", req.TemplateCode, true).
        First(&template).Error
    if err != nil {
        return nil, fmt.Errorf("模板不存在: %s", req.TemplateCode)
    }

    // 渲染标题和内容
    title := s.renderTemplate(template.TitleTemplate, req.Data)
    content := s.renderTemplate(template.ContentTemplate, req.Data)

    message := &models.NotificationMessage{
        UserID:       req.UserID,
        Type:         template.Type,
        Title:        title,
        Content:      content,
        RelatedID:    req.RelatedID,
        RelatedType:  req.RelatedType,
        RelatedURL:   req.RelatedURL,
        IsRead:       false,
        CreatedAt:    time.Now(),
    }

    return message, nil
}

// renderTemplate 渲染模板
func (s *NotificationService) renderTemplate(templateStr string, data map[string]interface{}) string {
    result := templateStr
    for key, value := range data {
        placeholder := fmt.Sprintf("{{.%s}}", key)
        result = fmt.Sprintf(result, placeholder, value)
    }
    return result
}

// determineChannels 确定发送渠道
func (s *NotificationService) determineChannels(
    requestedChannels []NotificationChannel,
    setting *models.UserNotificationSetting,
    msgType models.NotificationType,
) []NotificationChannel {
    channels := make([]NotificationChannel, 0)

    // 检查通知类型是否启用
    var typeEnabled bool
    switch msgType {
    case models.TypeTask:
        typeEnabled = setting.TaskEnabled
    case models.TypeSystem:
        typeEnabled = setting.SystemEnabled
    case models.TypeWarning:
        typeEnabled = setting.WarningEnabled
    case models.TypeReminder:
        typeEnabled = setting.ReminderEnabled
    default:
        typeEnabled = true
    }

    if !typeEnabled {
        return channels
    }

    // 检查渠道是否启用
    for _, channel := range requestedChannels {
        enabled := true
        switch channel {
        case ChannelEmail:
            enabled = setting.EmailEnabled
        case ChannelSMS:
            enabled = setting.SMSEnabled
        case ChannelDingTalk:
            enabled = setting.DingTalkEnabled
        case ChannelWeChat:
            enabled = setting.WeChatEnabled
        case ChannelFeiShu:
            enabled = setting.FeiShuEnabled
        }

        if enabled {
            channels = append(channels, channel)
        }
    }

    // 站内消息始终发送
    channels = append(channels, ChannelSystem)

    return channels
}
```

```go
// Send 发送邮件
func (s *EmailSender) Send(ctx context.Context, msg *models.NotificationMessage) error {
    // 构建邮件内容
    subject := msg.Title
    body := s.buildEmailBody(msg)

    // 发送邮件
    auth := smtp.PlainAuth("", s.username, s.password, s.smtpHost)
    addr := fmt.Sprintf("%s:%d", s.smtpHost, s.smtpPort)

    // 获取用户邮箱
    to := s.getUserEmail(ctx, msg.UserID)
    if to == "" {
        return fmt.Errorf("用户邮箱为空")
    }

    msgStr := fmt.Sprintf("To: %s\r\nSubject: %s\r\nContent-Type: text/html; charset=UTF-8\r\n\r\n%s",
        to, subject, body)

    err := smtp.SendMail(addr, auth, s.from, []string{to}, []byte(msgStr))
    if err != nil {
        s.logger.Error("发送邮件失败",
            zap.Uint("message_id", msg.ID),
            zap.String("to", to),
            zap.Error(err),
        )
        return err
    }

    return nil
}

    tmpl.Parse(tmpl)
```

### 3.4 钉钉发送器

```go
// internal/services/notification/senders/dingtalk_sender.go
package senders

import (
    "bytes"
    "crypto/hmac"
    "crypto/sha256"
    "encoding/base64"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"

    "go.uber.org/zap"
)

// DingTalkSender 钉钉发送器
type DingTalkSender struct {
    webhookURL string
    secret     string
    httpClient *http.Client
    logger     *zap.Logger
}

// NewDingTalkSender 创建钉钉发送器
func NewDingTalkSender(webhookURL, secret string) *DingTalkSender {
    return &DingTalkSender{
        webhookURL: webhookURL,
        secret:     secret,
        httpClient: &http.Client{Timeout: 30 * time.Second},
    }
}

// Send 发送钉钉消息
func (s *DingTalkSender) Send(ctx context.Context, msg *models.NotificationMessage) error {
    // 构建钉钉消息
    dingMsg := map[string]interface{}{
        "msgtype": "text",
        "text": map[string]interface{}{
            "content": fmt.Sprintf("%s\n\n%s", msg.Title, msg.Content),
        },
    }

    // 添加签名
    timestamp := time.Now().UnixNano() / 1e6
    sign := s.generateSign(timestamp)

    // 构建请求体
    payload := map[string]interface{}{
        "timestamp": timestamp,
        "sign":      sign,
        "msg":       dingMsg,
    }

    body, _ := json.Marshal(payload)
    req, _ := http.NewRequestWithContext(ctx, "POST", s.webhookURL, bytes.NewBuffer(body))
    req.Header.Set("Content-Type", "application/json")

    resp, err := s.httpClient.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        body, _ := io.ReadAll(resp.Body)
        return fmt.Errorf("钉钉发送失败: %s", string(body))
    }

    return nil
}

// generateSign 生成签名
func (s *DingTalkSender) generateSign(timestamp int64) string {
    stringToSign := fmt.Sprintf("%d\n%s", timestamp, s.secret)
    h := hmac.New(sha256.New, []byte(s.secret))
    h.Write([]byte(stringToSign))
    signData := h.Sum(nil)
    return base64.StdEncoding.EncodeToString(signData)
}

// Name 返回渠道名称
func (s *DingTalkSender) Name() models.NotificationChannel {
    return models.ChannelDingTalk
}
```

```go
// ListNotifications 获取通知列表
// @Summary 获取通知列表
// @Tags 通知
// @Produce json
// @Param type query string false "通知类型"
// @Param is_read query bool false "是否已读"
// @Param page query int false "页码" default(1)
// @Param page_size query int false "页大小" default(20)
// @Success 200 {object} Response
// @Router /api/v1/notifications [get]
func (h *NotificationHandler) ListNotifications(c *gin.Context) {
    var req struct {
        Type     string `form:"type"`
        IsRead   *bool  `form:"is_read"`
        Page     int    `form:"page" binding:"min=1"`
        PageSize int    `form:"page_size" binding:"min=1,max=100"`
    }
    if err := c.ShouldBindQuery(&req); err != nil {
        h.ErrorResponse(c, 400, "INVALID_REQUEST", "请求参数错误", err)
        return
    }

    userID := h.GetUserID(c)
    result, err := h.notificationService.Query(c.Request.Context(), userID, &req)
    if err != nil {
        h.ErrorResponse(c, 500, "INTERNAL_ERROR", "查询失败", err)
        return
    }

    h.SuccessResponse(c, result)
}

// @Summary 标记通知为已读
// @Tags 通知
// @Produce json
// @Param id path int true "通知ID"
// @Success 200 {object} Response
// @Router /api/v1/notifications/:id/read [put]
```
