# 统一验证中间件

## 概述

统一验证中间件提供了一套完整的请求参数验证框架，确保所有API请求在到达业务逻辑层之前都经过严格的参数校验。本文档详细描述了验证中间件的架构、实现和最佳实践。

## 核心设计原则

1. **声明式验证** - 使用结构体标签定义验证规则
2. **统一错误响应** - 验证失败返回标准格式错误
3. **可扩展性** - 支持自定义验证器和验证规则
4. **性能优化** - 验证缓存和短路机制
5. **国际化支持** - 多语言错误消息

## 架构设计

```
internal/
├── middleware/
│   ├── validator.go          # 验证中间件核心
│   ├── validation_rules.go   # 通用验证规则
│   └── custom_validators.go  # 自定义验证器
├── common/
│   └── validation.go         # 验证错误类型定义
└── handlers/
    └── base.go               # BaseHandler验证集成
```

```go
package middleware

import (
    "reflect"
    "strings"
    "unicode"

    "github.com/gin-gonic/gin"
    "github.com/go-playground/locales/zh"
    ut "github.com/go-playground/universal-translator"
    "github.com/go-playground/validator/v10"
    zh_translations "github.com/go-playground/validator/v10/translations/zh"
    "go.uber.org/zap"
)

// Validator 验证器实例
type Validator struct {
    validate     *validator.Validate
    translator   ut.Translator
    logger       *zap.Logger
}

// singleton instance
var globalValidator *Validator

// NewValidator 创建新的验证器实例
func NewValidator(logger *zap.Logger) *Validator {
    v := validator.New()

    // 注册自定义标签名称的函数
    v.RegisterTagNameFunc(func(fld reflect.StructField) string {
        name := strings.SplitN(fld.Tag.Get("json"), ",", 2)[0]
        if name == "-" {
            return ""
        }
        return name
    })

    // 设置中文翻译器
    uni := ut.New(zh.New())
    trans, _ := uni.GetTranslator("zh")
    zh_translations.RegisterDefaultTranslations(v, trans)

    validator := &Validator{
        validate:   v,
        translator: trans,
        logger:     logger,
    }

    // 注册自定义验证器
    validator.registerCustomValidators()

    return validator
}

// GetGlobalValidator 获取全局验证器实例
func GetGlobalValidator() *Validator {
    return globalValidator
}

// InitGlobalValidator 初始化全局验证器
func InitGlobalValidator(logger *zap.Logger) {
    globalValidator = NewValidator(logger)
}

// Validate 验证结构体
func (v *Validator) Validate(data interface{}) map[string]string {
    err := v.validate.Struct(data)
    if err == nil {
        return nil
    }

    errors := make(map[string]string)
    for _, err := range err.(validator.ValidationErrors) {
        fieldName := err.Field()
        translatedErr := err.Translate(v.translator)

        // 使用自定义错误消息（如果定义）
        if customMsg := v.getCustomMessage(fieldName, err.Tag()); customMsg != "" {
            errors[fieldName] = customMsg
        } else {
            errors[fieldName] = translatedErr
        }
    }

    return errors
}

// ValidateJSON 验证JSON请求体中间件
func (v *Validator) ValidateJSON(obj interface{}) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 绑定JSON
        if err := c.ShouldBindJSON(obj); err != nil {
            v.logger.Warn("JSON绑定失败",
                zap.String("path", c.Request.URL.Path),
                zap.Error(err),
            )
            c.JSON(400, gin.H{
                "code":    "INVALID_JSON",
                "message": "请求体格式错误",
                "details": err.Error(),
            })
            c.Abort()
            return
        }

        // 验证
        if errors := v.Validate(obj); errors != nil {
            v.logger.Warn("参数验证失败",
                zap.String("path", c.Request.URL.Path),
                zap.Any("errors", errors),
            )
            c.JSON(400, gin.H{
                "code":    "VALIDATION_ERROR",
                "message": "参数验证失败",
                "details": errors,
            })
            c.Abort()
            return
        }

        // 将验证后的对象保存到上下文
        c.Set("validated", obj)
        c.Next()
    }
}

// getCustomMessage 获取自定义错误消息

    customMessages := map[string]map[string]string{
        "password": {
            "required":  "密码不能为空",
            "min":       "密码长度不能少于8位",
            "complexity": "密码必须包含大小写字母、数字和特殊字符",
        },
        "username": {
            "required": "用户名不能为空",
            "alphanum": "用户名只能包含字母和数字",
        },
        "email": {
            "required": "邮箱不能为空",
            "email":    "邮箱格式不正确",
        },
    }

    if fieldMsgs, ok := customMessages[field]; ok {
        if msg, ok := fieldMsgs[tag]; ok {
            return msg
        }
    }
    return ""
}
```

### 2. 自定义验证器

**文件位置**: `internal/middleware/custom_validators.go`

```go
package middleware

import (
    "fmt"
    "net/url"
    "regexp"
    "strings"
    "unicode"

    "github.com/go-playground/validator/v10"
)

// registerCustomValidators 注册所有自定义验证器
func (v *Validator) registerCustomValidators() {
    // 1. 密码强度验证
    v.validate.RegisterValidation("complexity", v.validatePasswordComplexity)

    // 2. RTSP URL验证
    v.validate.RegisterValidation("rtsp_url", v.validateRTSPURL)

    // 3. IP地址验证（支持端口）
    v.validate.RegisterValidation("ip_port", v.validateIPWithPort)

    // 4. 电话号码验证（中国手机号）
    v.validate.RegisterValidation("phone_cn", v.validateChinesePhone)

    // 5. 用户名验证（字母开头，允许字母数字下划线）
    v.validate.RegisterValidation("username", v.validateUsername)

    // 6. Cron表达式验证
    v.validate.RegisterValidation("cron", v.validateCron)

    // 7. 端口范围验证
    v.validate.RegisterValidation("port", v.validatePort)

    // 8. 文件路径验证
    v.validate.RegisterValidation("filepath", v.validateFilePath)

    // 9. 枚举值验证
    v.validate.RegisterValidation("enum", v.validateEnum)

    // 10. 字符串数组验证（所有元素非空）
    v.validate.RegisterValidation("notblank", v.validateNotBlank)
}

// validatePasswordComplexity 密码复杂度验证
// 必须包含：大写字母、小写字母、数字、特殊字符，长度8-32位
func (v *Validator) validatePasswordComplexity(fl validator.FieldLevel) bool {
    password := fl.Field().String()
    if len(password) < 8 || len(password) > 32 {
        return false
    }

    var (
        hasUpper   = false
        hasLower   = false
        hasNumber  = false
        hasSpecial = false
    )

    for _, char := range password {
        switch {
        case unicode.IsUpper(char):
            hasUpper = true
        case unicode.IsLower(char):
            hasLower = true
        case unicode.IsNumber(char):
            hasNumber = true
        case unicode.IsPunct(char) || unicode.IsSymbol(char):
            hasSpecial = true
        }
    }

    return hasUpper && hasLower && hasNumber && hasSpecial
}

// validateRTSPURL RTSP URL验证
func (v *Validator) validateRTSPURL(fl validator.FieldLevel) bool {
    rtspURL := fl.Field().String()
    if rtspURL == "" {
        return false
    }

    // 必须以rtsp://或rtsps://开头
    if !strings.HasPrefix(rtspURL, "rtsp://") && !strings.HasPrefix(rtspURL, "rtsps://") {
        return false
    }

    // 验证URL格式
    u, err := url.Parse(rtspURL)
    if err != nil {
        return false
    }

    // 必须有主机地址
    if u.Host == "" {
        return false
    }

    return true
}

// validateIPWithPort IP:端口格式验证
func (v *Validator) validateIPWithPort(fl validator.FieldLevel) bool {
    value := fl.Field().String()
    if value == "" {
        return false
    }

    // 正则匹配 IP:端口 格式
    // 支持 IPv4 和 IPv6
    ipv4Pattern := `^(\d{1,3}\.){3}\d{1,3}:\d{1,5}$`
    ipv6Pattern := `^\[([0-9a-fA-F:]+)\]:\d{1,5}$`

    matched, _ := regexp.MatchString(ipv4Pattern, value)
    if matched {
        // 验证IP地址范围
        parts := strings.Split(value, ":")
        ip := strings.Join(parts[:4], ".")
        return isValidIP(ip)
    }

    matched, _ = regexp.MatchString(ipv6Pattern, value)
    return matched
}

// validateChinesePhone 中国手机号验证
func (v *Validator) validateChinesePhone(fl validator.FieldLevel) bool {
    phone := fl.Field().String()
    if phone == "" {
        return false
    }

    // 中国手机号正则：1开头，第二位3-9，共11位数字
    pattern := `^1[3-9]\d{9}$`
    matched, _ := regexp.MatchString(pattern, phone)
    return matched
}

// validateUsername 用户名验证
// 规则：4-32位，字母开头，只允许字母数字下划线
func (v *Validator) validateUsername(fl validator.FieldLevel) bool {
    username := fl.Field().String()
    if len(username) < 4 || len(username) > 32 {
        return false
    }

    // 第一个字符必须是字母
    if !unicode.IsLetter(rune(username[0])) {
        return false
    }

    // 只允许字母数字下划线
    pattern := `^[a-zA-Z][a-zA-Z0-9_]*$`
    matched, _ := regexp.MatchString(pattern, username)
    return matched
}

// validateCron Cron表达式验证
func (v *Validator) validateCron(fl validator.FieldLevel) bool {
    cronExpr := fl.Field().String()
    if cronExpr == "" {
        return false
    }

    // 基本的Cron表达式格式验证
    // 标准5段或6段格式
    parts := strings.Fields(cronExpr)
    if len(parts) != 5 && len(parts) != 6 {
        return false
    }

    // 这里可以添加更详细的验证逻辑
    // 或使用github.com/robfig/cron/v3包进行验证
    return true
}

// validatePort 端口范围验证
func (v *Validator) validatePort(fl validator.FieldLevel) bool {
    port := fl.Field().Int()
    return port > 0 && port <= 65535
}

// validateFilePath 文件路径验证
func (v *Validator) validateFilePath(fl validator.FieldLevel) bool {
    path := fl.Field().String()
    if path == "" {
        return false
    }

    // 检查路径是否包含非法字符
    illegalChars := []string{"<", ">", ":", "\"", "|", "?", "*"}
    for _, char := range illegalChars {
        if strings.Contains(path, char) {
            return false
        }
    }

    return true
}

// validateEnum 枚举值验证
// 使用方法：validate:"enum=value1,value2,value3"
func (v *Validator) validateEnum(fl validator.FieldLevel) bool {
    value := fl.Field().String()
    param := fl.Param()

    if value == "" {
        return true // 空值由required标签处理
    }

    allowedValues := strings.Split(param, ",")
    for _, allowed := range allowedValues {
        if value == strings.TrimSpace(allowed) {
            return true
        }
    }

    return false
}

// validateNotBlank 字符串数组非空验证
func (v *Validator) validateNotBlank(fl validator.FieldLevel) bool {
    field := fl.Field()

    // 处理字符串数组/切片
    if field.Kind() == reflect.Slice || field.Kind() == reflect.Array {
        for i := 0; i < field.Len(); i++ {
            elem := field.Index(i)
            if elem.Kind() == reflect.String {
                if strings.TrimSpace(elem.String()) == "" {
                    return false
                }
            }
        }
        return true
    }

    // 处理单个字符串
    if field.Kind() == reflect.String {
        return strings.TrimSpace(field.String()) != ""
    }

    return true
}

// 辅助函数
func isValidIP(ip string) bool {
    parts := strings.Split(ip, ".")
    if len(parts) != 4 {
        return false
    }

    for _, part := range parts {
        num, err := fmt.Sscanf(part, "%d")
        if err != nil || num < 0 || num > 255 {
            return false
        }
    }

    return true
}
```

```go
package middleware

// CommonRequest 通用请求参数
type CommonRequest struct {
    Page     int    `form:"page" json:"page" validate:"min=1"`
    PageSize int    `form:"page_size" json:"page_size" validate:"min=1,max=100"`
    OrderBy  string `form:"order_by" json:"order_by" validate:"omitempty,oneof=id created_at updated_at"`
    Order    string `form:"order" json:"order" validate:"omitempty,oneof=asc desc"`
}

// IDRequest ID参数请求
type IDRequest struct {
    ID uint `uri:"id" json:"id" validate:"required"`
}

// IDBatchRequest 批量ID请求
type IDBatchRequest struct {
    IDs []uint `json:"ids" validate:"required,min=1,dive,min=1"`
}

// DateRangeRequest 日期范围查询
type DateRangeRequest struct {
    StartDate string `form:"start_date" json:"start_date" validate:"required,datetime=2006-01-02"`
    EndDate   string `form:"end_date" json:"end_date" validate:"required,datetime=2006-01-02"`
}

// KeywordRequest 关键词搜索请求
type KeywordRequest struct {
    Keyword string `form:"keyword" json:"keyword" validate:"omitempty,min=1,max=100"`
}

// UsernameRequest 用户名请求
type UsernameRequest struct {
    Username string `json:"username" validate:"required,username,min=4,max=32"`
}

// PasswordRequest 密码请求
type PasswordRequest struct {
    Password string `json:"password" validate:"required,complexity"`
}

// EmailRequest 邮箱请求
type EmailRequest struct {
    Email string `json:"email" validate:"required,email"`
}

// PhoneRequest 电话请求
```

```go
// routes.go
func SetupRoutes(r *gin.Engine, app *MinimalApp) {
    validator := middleware.GetGlobalValidator()
    userHandler := handlers.NewUserHandler(app.Logger, app.Config)

    api := r.Group("/api/v1")
    {
        users := api.Group("/users")
        {
            // 使用验证中间件

                validator.ValidateJSON(CreateUserRequest{}),
                userHandler.CreateUser,

            users.GET("/:id",
                validator.ValidateURI(IDRequest{}),

            users.PUT("/:id",

                userHandler.UpdateUser,

                validator.ValidateURI(IDRequest{}),

            )

            users.GET("",
```

```go
// UpdatePasswordRequest 修改密码请求
type UpdatePasswordRequest struct {
    OldPassword string `json:"old_password" validate:"required"`
    NewPassword string `json:"new_password" validate:"required,complexity"`
}
```

```go
// ConferenceCreateRequest 创建会议请求
type ConferenceCreateRequest struct {
    Name        string    `json:"name" validate:"required,min=1,max=100"`
    StartTime   time.Time `json:"start_time" validate:"required"`
    EndTime     time.Time `json:"end_time" validate:"required"`
    MaxAttendees int      `json:"max_attendees" validate:"min=1,max=500"`
}

// CustomValidation 自定义验证方法
func (r *ConferenceCreateRequest) CustomValidation() error {
    // 结束时间必须大于开始时间
    if r.EndTime.Before(r.StartTime) || r.EndTime.Equal(r.StartTime) {
        return fmt.Errorf("结束时间必须大于开始时间")
    }

    // 会议时长不能超过24小时
    duration := r.EndTime.Sub(r.StartTime)
    if duration > 24*time.Hour {
        return fmt.Errorf("会议时长不能超过24小时")
    }

    return nil
}
```
