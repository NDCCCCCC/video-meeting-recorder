# 前端路由与权限

## 一、路由设计概述

### 1.1 路由结构

```
┌─────────────────────────────────────────────────────────────┐
│                       路由层次结构                            │
└─────────────────────────────────────────────────────────────┘
    │
    ├── 公开路由（无需认证）
    │   ├── /auth/login
    │   ├── /auth/register
    │   └── /auth/callback
    │
    ├── 主应用路由（需要认证）
    │   ├── /
    │   ├── /dashboard
    │   ├── /tasks
    │   ├── /conferences
    │   ├── /recordings
    │   └── /system
    │
    └── 错误路由
        ├── /403 (Forbidden)
        ├── /404 (NotFound)
        └── /500 (ServerError)
```

```typescript
// 路由配置接口
interface RouteConfig {
  path: string;
  element: React.ReactNode;
  children?: RouteConfig[];
  // 权限相关
  auth?: boolean;           // 是否需要认证
  permission?: string;       // 需要的权限

  // 其他配置

  loader?: () => Promise<any> | any;

  title?: string;
  icon?: React.ReactNode;
  hidden?: boolean;         // 是否在菜单中隐藏
  keepAlive?: boolean;      // 是否缓存组件
  affix?: boolean;          // 是否固定在标签栏
  noCache?: boolean;        // 不缓存
}
```

### 2.2 路由守卫实现

```typescript
// src/router/guards.tsx
import { redirect, LoaderFunction } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';
import { NProgress } from '@/utils/nprogress';

// 认证守卫
export const authGuard: LoaderFunction = async () => {
  NProgress.start();
  const isAuthenticated = useAuthStore.getState().isAuthenticated;

  if (!isAuthenticated) {
    // 未登录，跳转到登录页
    const returnUrl = location.pathname;
    return redirect(`/auth/login?redirect=${encodeURIComponent(returnUrl)}`);
  }

  return null;
};

// 权限守卫
export const permissionGuard = (requiredPermission: string): LoaderFunction => {
  return async () => {
    const hasPermission = useAuthStore.getState().hasPermission(requiredPermission);

    if (!hasPermission) {
      return redirect('/403');
    }

    return null;
  };
};

// 角色守卫
export const roleGuard = (allowedRoles: string[]): LoaderFunction => {
  return async () => {
    const user = useAuthStore.getState().user;
    const hasRole = user?.roles?.some(role => allowedRoles.includes(role));

    if (!hasRole) {
      return redirect('/403');
    }

    return null;
  };
};

// 进度条守卫
export const progressGuard: LoaderFunction = async () => {
  NProgress.start();
  return () => NProgress.done();
};
```

### 2.3 路由生成器

```typescript
// src/router/generator.tsx
import { Navigate, RouteObject, useRoutes } from 'react-router-dom';
import { Suspense } from 'react';
import { authGuard, permissionGuard, roleGuard, progressGuard } from './guards';
import { PageLoading } from '@/components/PageLoading';
import type { RouteConfig } from '@/types/router';

// 将路由配置转换为React Router的RouteObject
export function generateRoutes(routes: RouteConfig[]): RouteObject[] {
  return routes.map(route => {
    // 处理children
    const children = route.children ? generateRoutes(route.children) : undefined;

    // 创建RouteObject
    const routeObject: RouteObject = {
      path: route.path,
      element: route.element,
      children,
    };

    // 添加loader
    const loaders: LoaderFunction[] = [];

    // 添加进度条loader
    loaders.push(progressGuard);

    // 添加认证loader
    if (route.auth) {
      loaders.push(authGuard);
    }

    // 添加权限loader
    if (route.permission) {
      loaders.push(permissionGuard(route.permission));
    }

    // 添加角色loader
    if (route.roles) {
      loaders.push(roleGuard(route.roles));
    }

    // 组合所有loader
    if (loaders.length > 0) {
      routeObject.loader = async (...args) => {
        for (const loader of loaders) {
          const result = await loader(...args);
          if (result) return result;
        }
        return null;
      };
    }

    return routeObject;
  });
}

// 包装Suspense
export function wrapSuspense(element: React.ReactNode) {
  return (
    <Suspense fallback={<PageLoading />}>
      {element}
    </Suspense>
  );
}

// 创建应用路由
export function createAppRouter(routes: RouteConfig[]) {
  const routeObjects = generateRoutes(routes);

  const wrappedRoutes = routeObjects.map(route => ({
    ...route,
    element: wrapSuspense(route.element!),
    children: route.children?.map(child => ({
      ...child,
      element: wrapSuspense(child.element!),
    })),
  }));

  // 添加404重定向
  wrappedRoutes.push({
    path: '*',
    element: <Navigate to="/404" replace />,
  });

  return useRoutes(wrappedRoutes);
}
```

## 三、权限控制实现

### 3.1 权限指令组件

```typescript
// src/components/Permission/Permission.tsx
import { ReactNode } from 'react';
import { useAuthStore } from '@/stores/authStore';
import type { PermissionMode } from './types';

interface PermissionProps {
  permission?: string | string[];
  mode?: PermissionMode;
  role?: string | string[];
  fallback?: ReactNode;
  children: ReactNode;
}

type PermissionMode = 'oneOf' | 'allOf';

// 权限控制组件
export function Permission({
  permission,
  mode = 'oneOf',
  role,
  fallback = null,
  children,
}: PermissionProps) {
  const { hasPermission, hasAnyPermission, hasRole } = useAuthStore();

  // 检查权限
  let hasPermissionAccess = true;
  if (permission) {
    const permissions = Array.isArray(permission) ? permission : [permission];
    hasPermissionAccess = mode === 'oneOf'
      ? hasAnyPermission(permissions)
      : permissions.every(p => hasPermission(p));
  }

  // 检查角色
  let hasRoleAccess = true;
  if (role) {
    const roles = Array.isArray(role) ? role : [role];
    hasRoleAccess = roles.some(r => hasRole(r));
  }

  const hasAccess = hasPermissionAccess && hasRoleAccess;

  if (!hasAccess) {
    return <>{fallback}</>;
  }

  return <>{children}</>;
}

// 权限按钮组件
interface PermissionButtonProps extends ButtonProps {
  permission?: string | string[];
  mode?: PermissionMode;
}

export function PermissionButton({
  permission,
  mode = 'oneOf',
  children,
  ...props
}: PermissionButtonProps) {
  return (
    <Permission permission={permission} mode={mode} fallback={null}>
      <Button {...props}>{children}</Button>
    </Permission>
  );
}

// 使用示例
function TaskActions() {
  return (
    <Space>
      <PermissionButton permission="task:create" type="primary">
        创建任务
      </PermissionButton>
      <PermissionButton permission="task:delete" type="primary" danger>
        删除任务
      </PermissionButton>
      <Permission
        permission={['task:edit', 'admin']}
        mode="oneOf"
        fallback={<span>无权限</span>}
      >
        <Button>编辑</Button>
      </Permission>
    </Space>
  );
}
```

### 3.2 菜单权限过滤

```typescript
// src/router/menuFilter.ts
import { useMemo } from 'react';
import { useAuthStore } from '@/stores/authStore';
import { routes } from './routes';
import type { MenuItem } from 'antd/es/menu/interface';

// 将路由配置转换为菜单项
export function useMenuItems(): MenuItem[] {
  const { hasPermission, hasRole, user } = useAuthStore();

  return useMemo(() => {
    return filterRoutesToMenuItems(routes, { hasPermission, hasRole, user });
  }, [hasPermission, hasRole, user]);
}

function filterRoutesToMenuItems(
  routes: RouteConfig[],
  context: AuthContext
): MenuItem[] {
  const menuItems: MenuItem[] = [];

  for (const route of routes) {
    // 跳过隐藏路由和错误路由
    if (route.meta?.hidden || route.path === '*' || route.path.startsWith('/auth')) {
      continue;
    }

    // 检查权限
    if (!checkRoutePermission(route, context)) {
      continue;
    }

    // 有子路由的处理
    if (route.children && route.children.length > 0) {
      const childItems = filterRoutesToMenuItems(route.children, context);
      if (childItems.length > 0) {
        menuItems.push({
          key: route.path,
          icon: route.meta?.icon,
          label: route.meta?.title,
          children: childItems,
        });
      }
    } else {
      // 叶子菜单项
      menuItems.push({
        key: route.path,
        icon: route.meta?.icon,
        label: route.meta?.title,
      });
    }
  }

  return menuItems;
}

function checkRoutePermission(route: RouteConfig, context: AuthContext): boolean {
  // 检查认证要求
  if (route.auth && !context.isAuthenticated) {
    return false;
  }

  // 检查权限
  if (route.permission && !context.hasPermission(route.permission)) {
    return false;
  }

  // 检查角色
  if (route.roles && route.roles.length > 0) {
    const userRoles = context.user?.roles || [];
    if (!route.roles.some(role => userRoles.includes(role))) {
      return false;
    }
  }

  return true;
}
```

### 3.3 路由级权限Hook

```typescript
// src/hooks/useRoutePermission.ts
import { useLocation } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';
import { routes } from '@/router/routes';
import type { RouteConfig } from '@/types/router';

export function useRoutePermission() {
  const location = useLocation();
  const { hasPermission, hasRole, isAuthenticated, user } = useAuthStore();

  const currentRoute = findRouteByPath(routes, location.pathname);
  const hasAccess = checkRouteAccess(currentRoute, {
    hasPermission,
    hasRole,
    isAuthenticated,
    user,
  });

  return {
    hasAccess,
    requiredPermission: currentRoute?.permission,
    requiredRoles: currentRoute?.roles,
  };
}

function findRouteByPath(routes: RouteConfig[], path: string): RouteConfig | null {
  for (const route of routes) {
    if (route.path === path) {
      return route;
    }
    if (route.children) {
      const found = findRouteByPath(route.children, path);
      if (found) return found;
    }
  }
  return null;
}

function checkRouteAccess(
  route: RouteConfig | null,
  context: AuthContext
): boolean {
  if (!route) return false;

  // 检查认证
  if (route.auth && !context.isAuthenticated) {
    return false;
  }

  // 检查权限
  if (route.permission && !context.hasPermission(route.permission)) {
    return false;
  }

  // 检查角色
  if (route.roles && route.roles.length > 0) {
    const userRoles = context.user?.roles || [];
    if (!route.roles.some(role => userRoles.includes(role))) {
      return false;
    }
  }

  return true;
}
```

```typescript
// src/router/breadcrumb.tsx
import { Breadcrumb } from 'antd';
import { Link, useLocation } from 'react-router-dom';
import { HomeOutlined } from '@ant-design/icons';
import { useRouteBreadcrumbs } from '@/hooks/useRouteBreadcrumbs';

export function AppBreadcrumb() {
  const breadcrumbs = useRouteBreadcrumbs();

  if (breadcrumbs.length === 0) {
    return null;
  }

  const items = [
    {
      title: <Link to="/"><HomeOutlined /></Link>,
    },
    ...breadcrumbs.map((bc, index) => ({
      title: bc.href ? (
        <Link to={bc.href}>{bc.title}</Link>
      ) : (
        <span>{bc.title}</span>
      ),
    })),
  ];

  return <Breadcrumb items={items} />;
}

// src/hooks/useRouteBreadcrumbs.ts
import { useMemo } from 'react';
import { useLocation } from 'react-router-dom';
import { routes } from '@/router/routes';

interface BreadcrumbItem {
  title: string;
  href?: string;
}

export function useRouteBreadcrumbs(): BreadcrumbItem[] {
  const location = useLocation();

  return useMemo(() => {
    const breadcrumbs: BreadcrumbItem[] = [];
    const pathSegments = location.pathname.split('/').filter(Boolean);

    let currentPath = '';
    let currentRoutes = routes;

    for (let i = 0; i < pathSegments.length; i++) {
      const segment = pathSegments[i];
      currentPath += '/' + segment;

      // 在当前路由层级中查找匹配的路由
      const matchedRoute = findRouteByPath(currentRoutes, currentPath);
      if (matchedRoute?.meta?.title) {
        breadcrumbs.push({
          title: matchedRoute.meta.title,
          href: i < pathSegments.length - 1 ? currentPath : undefined,
        });

        // 进入子路由继续查找
        if (matchedRoute.children) {
          currentRoutes = matchedRoute.children;
        }
      }
    }

    return breadcrumbs;
  }, [location.pathname]);
}
```

### 5.1 标签页Store

```typescript
// src/stores/tabsStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface Tab {
  key: string;
  title: string;
  path: string;
  closable: boolean;
}

interface TabsState {
  tabs: Tab[];
  activeKey: string;

  addTab: (tab: Tab) => void;
  removeTab: (key: string) => void;
  setActiveKey: (key: string) => void;
  closeOtherTabs: (key: string) => void;
  closeAllTabs: () => void;
}

export const useTabsStore = create<TabsState>()(
  persist(
    (set, get) => ({
      tabs: [],
      activeKey: '/',

      addTab: (tab) => {
        set((state) => {
          // 检查标签是否已存在
          const existingTab = state.tabs.find(t => t.key === tab.key);
          if (existingTab) {
            return {
              ...state,
              activeKey: tab.key,
            };
          }

          return {
            tabs: [...state.tabs, tab],
            activeKey: tab.key,
          };
        });
      },

      removeTab: (key) => {
        set((state) => {
          const tabs = state.tabs.filter(t => t.key !== key);

          // 如果关闭的是当前激活的标签，切换到前一个
          let activeKey = state.activeKey;
          if (state.activeKey === key) {
            const index = state.tabs.findIndex(t => t.key === key);
            activeKey = index > 0 ? state.tabs[index - 1].key : '/';
          }

          return { tabs, activeKey };
        });
      },

      setActiveKey: (activeKey) => set({ activeKey }),

      closeOtherTabs: (key) => {
        set((state) => ({
          tabs: state.tabs.filter(t => t.key === key || !t.closable),
        }));
      },

      closeAllTabs: () => {
        set((state) => ({
          tabs: state.tabs.filter(t => !t.closable),
          activeKey: '/',
        }));
      },
    }),
    {
      name: 'tabs-storage',
    }
  )
);
```

### 5.2 自动标签管理

```typescript
// src/components/AutoTabs/index.tsx
import { useEffect } from 'react';
import { useLocation, useNavigate } from 'react-router-dom';
import { useTabsStore } from '@/stores/tabsStore';
import { routes } from '@/router/routes';

export function AutoTabs() {
  const location = useLocation();
  const navigate = useNavigate();
  const { tabs, activeKey, addTab, removeTab, setActiveKey } = useTabsStore();

  // 监听路由变化，自动添加标签
  useEffect(() => {
    const route = findRouteByPath(routes, location.pathname);
    if (route?.meta?.title) {
      addTab({
        key: location.pathname,
        title: route.meta.title,
        path: location.pathname,
        closable: location.pathname !== '/', // 首页不可关闭
      });
    }
    setActiveKey(location.pathname);
  }, [location.pathname]);

  // 监听标签点击
  useEffect(() => {
    if (activeKey && activeKey !== location.pathname) {
      navigate(activeKey);
    }
  }, [activeKey]);

  return null; // 仅做逻辑处理，UI在AppLayout中渲染
}
```
