```typescript
// ==================== 字幕加载 ====================
useEffect(() => {
  if (!visible || !file?.id) {
    setVttContent(null)
    setSubtitleAvailable(false)
    return
  }

  let cancelled = false

  checkSubtitleExists(file.id)
    .then(res => {
      if (cancelled) return
      if (res.data?.exists) {
        setSubtitleAvailable(true)
        return getSubtitleContent(file.id)
      }
      setSubtitleAvailable(false)
      return null
    })
    .then(content => {
      if (cancelled || !content) return
      setVttContent(content)
    })
    .catch(err => {
      if (!cancelled) {
        console.warn('字幕加载失败:', err)
        setSubtitleAvailable(false)
      }
    })

  return () => { cancelled = true }
}, [visible, file?.id])
```
