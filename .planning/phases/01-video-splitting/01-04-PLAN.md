   ```typescript
   import { PERMISSIONS } from '../../utils/permissions'
   ```
   (PERMISSIONS may already be imported; just ensure RECORDING_SNAPSHOT is available.)

2. Add snapshot state per task. Since tasks is an array, track generating state with a Set:
   ```typescript
   const [snapshotGenerating, setSnapshotGenerating] = useState<Set<number>>(new Set())
   ```

3. Add handler function:
   ```typescript
   const handleGenerateSnapshot = async (taskId: number) => {
     setSnapshotGenerating(prev => new Set(prev).add(taskId))
     try {
       const result = await apiRequest<{ snapshot_file_id: number; file_name: string }>(
         `/api/v1/tasks/${taskId}/snapshot`,
         { method: 'POST' }
       )
       if (result.data) {
         message.success(`快照已生成：${result.data.file_name}`)
       }
     } catch (error) {
       message.error(error instanceof Error ? error.message : '生成快照失败')
     } finally {
       setSnapshotGenerating(prev => {
         const next = new Set(prev)
         next.delete(taskId)
         return next
       })
     }
   }
   ```

<threat_model>
## Attack Surface
- Snapshot button triggers server-side FFmpeg operation
- File list auto-refresh polls backend every 5 seconds
- Source column displays data from backend (no user input)

## Threats
| Threat | Severity | Mitigation |
|--------|----------|------------|
| Snapshot generation DoS — user clicks snapshot button repeatedly | MEDIUM | Loading state disables button during generation. Backend should check task status and return error if not recording. Single snapshot per task at a time enforced by the loading state |
| File list polling overhead | LOW | 5-second interval is reasonable. Silent refresh without loading spinner. Polling cancels on unmount via useEffect cleanup |
| IDOR via "查看原视频" link | LOW | Link navigates to file list, not directly to file. Backend enforces ownership check on file detail API |

## Security Decisions
- Auto-refresh is polling-based (5s interval) rather than WebSocket for simplicity
- Snapshot button disabled during generation to prevent duplicate requests
- "查看原视频" link navigates to file list page (user still needs permission to view)
</threat_model>
