   ```go
   func (s *VideoFileService) CreateSegmentFile(segmentPath string, parentVideoID *uint, sourceType string, createdBy uint) (*models.VideoFile, error) {
       // 1. Check if file already exists (duplicate check)
       existing, err := s.findExistingFile(segmentPath)
       if err == nil && existing != nil {
           s.logger.Warn("文件记录已存在", zap.String("path", segmentPath), zap.Uint("existing_id", existing.ID))
           return existing, nil
       }

       // 2. Get file info
       fileInfo, err := os.Stat(segmentPath)
       if err != nil {
           return nil, fmt.Errorf("文件不存在: %w", err)
       }

       // 3. Extract metadata using existing method
       metadata, err := s.extractVideoMetadata(segmentPath)
       if err != nil {
           s.logger.Warn("提取视频元数据失败，使用默认值", zap.Error(err))
           metadata = s.getDefaultMetadata(segmentPath)
       }

       // 4. Generate file name from path
       fileName := filepath.Base(segmentPath)

       // 5. Determine format from extension
       format := s.inferFormat(segmentPath)

       // 6. Create VideoFile record
       videoFile := &models.VideoFile{
           FileName:   fileName,
           FilePath:   segmentPath,
           FileSize:   fileInfo.Size(),
           Duration:   metadata.Duration,
           Format:     format,
           Resolution: metadata.Resolution,
           Bitrate:    metadata.Bitrate,
           Codec:      metadata.Codec,
           ParentID:   parentVideoID,
           SourceType: sourceType,
           CreatedBy:  createdBy,
           Status:     models.FileStatusReady,
           RecordedAt: nil,
       }

       // 7. Set RecordedAt from parent if available
       if parentVideoID != nil {
           var parent models.VideoFile
           if err := s.db.First(&parent, *parentVideoID).Error; err == nil {
               videoFile.RecordedAt = parent.RecordedAt
           }
       }

       // 8. Save to database
       if err := s.createWithDuplicateCheck(videoFile); err != nil {
           return nil, fmt.Errorf("创建文件记录失败: %w", err)
       }

       return videoFile, nil
   }
   ```

- Database migration — ALTER TABLE operations

## Threats
| Threat | Severity | Mitigation |
|--------|----------|------------|
| Unauthorized parent_id manipulation — user could set parent_id to another user's video | MEDIUM | Handler layer must verify user owns parent video before creating segment (enforced in Plan 02 handler) |
| source_type injection — unexpected source_type values stored | LOW | source_type validated at handler level against allowed constants (recording/snapshot/split) |

## Security Decisions
- source_type field uses VARCHAR(20) with no CHECK constraint (validation at application layer)
- parent_id is nullable — NULL means the file is an original recording (not a segment/snapshot)
