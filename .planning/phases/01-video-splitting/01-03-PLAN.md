```typescript
import type { ApiResponse } from '../types/auth'
import type { VideoFile } from '../types/video-file'
import { apiRequest } from './apiClient'

export interface SplitRequest {
  markers: number[]
  re_encode?: boolean
}

export interface SplitResponse {
  video_file_id: number
  status: string
  segment_count: number
}

export interface SplitStatusResponse {
  status: string
  segments?: VideoFile[]
}

// Submit a split task
export async function submitSplit(
  videoFileId: number,
  markers: number[],
  reEncode: boolean = false
): Promise<ApiResponse<SplitResponse>> {
  return apiRequest<SplitResponse>(`/api/v1/videos/${videoFileId}/split`, {
    method: 'POST',
    body: JSON.stringify({ markers, re_encode: reEncode }),
  })
}

// Get split status for a video file
export async function getSplitStatus(
  videoFileId: number
): Promise<ApiResponse<SplitStatusResponse>> {
  return apiRequest<SplitStatusResponse>(`/api/v1/videos/${videoFileId}/split-status`)
}

// Get segments for a video file
export async function getSegments(
  videoFileId: number
): Promise<ApiResponse<VideoFile[]>> {
  return apiRequest<VideoFile[]>(`/api/v1/videos/${videoFileId}/segments`)
}
```

```typescript
interface TimelineWithMarkersProps {

  onMarkerRemove: (time: number) => void
  onSeek: (time: number) => void
}
```

- Below the Slider, render a row of marker chips/tags. Each shows the timestamp and has an X button to remove.
- Add an `Input` field with `onPressEnter` that parses the timestamp (MM:SS or seconds) and calls `onMarkerAdd`
- Helper function `formatTime(seconds: number): string` returns `MM:SS` format
- Validate: timestamp must be > 0 and < duration, minimum 2 seconds between markers, max 20 markers

<threat_model>
## Attack Surface
- Video player loads video from backend download endpoint (token-authenticated)
- User-provided timestamps in manual input field
- Split API call with markers array

## Threats
| Threat | Severity | Mitigation |
|--------|----------|------------|
| XSS via marker labels | LOW | Ant Design Slider marks accept ReactNode, timestamps are numbers rendered through formatTime — no raw HTML injection possible |
| Marker timestamp manipulation | LOW | Frontend validates markers (range, count, dedup), backend validates independently. No security impact from crafted markers (FFmpeg uses exec.Command with array args) |
| Unauthorized video access via URL manipulation | LOW | Backend enforces auth middleware on all /api/v1 routes. Frontend passes token in download URL |

## Security Decisions
- Video download URL includes token query parameter (existing pattern from downloadVideoFile)
- All API calls go through apiRequest which handles auth tokens
- Marker validation happens on both frontend (UX) and backend (security)
</threat_model>
