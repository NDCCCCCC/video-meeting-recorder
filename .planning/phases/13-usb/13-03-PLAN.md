    ```go
    // ValidateConfig 根据配置类型验证必填字段
    func (s *InputConfigService) ValidateConfig(config *models.InputConfig) error {
        // First run model-level validation
        if err := config.Validate(); err != nil {
            return err
        }

        // Additional service-level validation based on config type
        switch config.ConfigType {
        case models.ConfigTypeHuaweiAuto:
            if config.HuaweiEnabled {
                // Huawei terminal fields are required
                return s.validateHuaweiFields(config)
            } else {
                // Must have USB or Stream source
                hasUSB := config.USBCameraDevice != ""
                hasStream := config.StreamURL != ""
                if !hasUSB && !hasStream {
                    return errors.New("华为控制关闭时，必须选择USB或流媒体录制源")
                }
            }

        case models.ConfigTypeUSB:
            // USB device is required
            if config.USBCameraDevice == "" {
                return errors.New("USB配置必须指定摄像头设备")
            }

        case models.ConfigTypeStream:
            // Stream URL is required
            if config.StreamURL == "" {
                return errors.New("流媒体配置必须指定流地址")
            }

        default:
            return fmt.Errorf("无效的配置类型: %s", config.ConfigType)
        }

        return nil
    }

    // validateHuaweiFields 验证华为终端必填字段
    func (s *InputConfigService) validateHuaweiFields(config *models.InputConfig) error {
        if config.Server == "" {
            return errors.New("华为服务器地址不能为空")
        }
        if config.Username == "" {
            return errors.New("用户名不能为空")
        }
        if config.TerminalNumber == "" {
            return errors.New("终端号码不能为空")
        }
        // Note: Password validation is handled at HTTP handler level (binding:"required")
        return nil
    }
    ```

       ```go
       func (s *InputConfigService) testStreamConnection(req *TestConnectionRequest) error {
           s.logger.Info("测试流媒体连接",
               zap.String("protocol", req.StreamProtocol),
               zap.String("url", req.StreamURL),
           )

           ffprobePath := s.config.FFmpeg.FFProbePath
           if ffprobePath == "" {
               ffprobePath = "ffprobe"
           }

           var inputArgs []string
           switch req.StreamProtocol {
           case "rtmp":
               inputArgs = []string{"-i", req.StreamURL}
           case "rtsp":
               inputArgs = []string{"-rtsp_transport", "tcp", "-i", req.StreamURL}
           case "srt":
               inputArgs = []string{"-i", fmt.Sprintf("srt://%s", req.StreamURL)}
           case "hls":
               inputArgs = []string{"-i", req.StreamURL}
           default:
               return errors.New("不支持的流媒体协议")
           }

           ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
           defer cancel()

           args := []string{"-v", "error"}
           args = append(args, inputArgs...)
           args = append(args, "-f", "null", "-")

           cmd := exec.CommandContext(ctx, ffprobePath, args...)
           output, err := cmd.CombinedOutput()

           if ctx.Err() == context.DeadlineExceeded {
               return fmt.Errorf("连接超时（15秒），请检查网络和流媒体地址是否正确")
           }

           if err != nil {
               s.logger.Error("FFprobe failed",
                   zap.Error(err),
                   zap.String("output", string(output)),
               )
               return fmt.Errorf("流媒体连接测试失败: %w", err)
           }

           s.logger.Info("流媒体连接测试成功")
           return nil
       }
       ```
