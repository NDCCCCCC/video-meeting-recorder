               if c.HuaweiEnabled {
                   if c.Server == "" { return errors.New("华为服务器地址不能为空") }
                   if c.Username == "" { return errors.New("用户名不能为空") }
                   if c.TerminalNumber == "" { return errors.New("终端号码不能为空") }
               } else {
                   hasUSB := c.USBCameraDevice != ""
                   hasStream := c.StreamURL != ""
                   if !hasUSB && !hasStream { return errors.New("华为控制关闭时，必须选择USB或流媒体录制源") }
