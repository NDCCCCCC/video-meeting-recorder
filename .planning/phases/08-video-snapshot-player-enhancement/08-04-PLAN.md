2. Add handleMuteToggle callback:
```typescript
const handleMuteToggle = useCallback(() => {
  const video = videoRef.current
  if (!video) return

  if (muted) {
    // Unmute: restore previous volume
    const restoreVolume = volume > 0 ? volume : 0.5
    setMuted(false)
    setActualVolume(restoreVolume)
    video.volume = restoreVolume
    video.muted = false
  } else {
    // Mute: store current volume and set to 0
    setMuted(true)
    setVolume(video.volume)
    setActualVolume(0)
    video.muted = true
  }
}, [muted, volume])
```

3. Add handleSeek with Infinity support:
```typescript
const handleSeekWithInfinity = useCallback((seconds: number) => {
  const video = videoRef.current
  if (!video || !duration) return

  if (seconds === Infinity) {
    video.currentTime = video.duration
  } else if (seconds === -Infinity) {
    video.currentTime = 0
  } else {
    video.currentTime = Math.max(0, Math.min(duration, video.currentTime + seconds))
  }
}, [duration])
```

  }
}, [])
